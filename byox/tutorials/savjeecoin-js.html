بلاکچین‌ها (Blockchains)

بلاکچین یک پایگاه‌داده عمومی است که از بلاک‌هایی تشکیل‌شده که هرکسی می‌تواند آن‌ها را بخواند. هیچ چیز خاصی نیست، اما یک خاصیت جالب دارند: تغییرناپذیر هستند (immutable). به محض اینکه یک بلاک به زنجیره اضافه شود، دیگر نمی‌توان آن را تغییر داد — مگر اینکه بقیه زنجیره را هم باطل کنید.

به همین دلیل است که ارزهای دیجیتال بر پایه بلاکچین ساخته می‌شوند. شما نمی‌خواهید مردم بعد از انجام تراکنش‌هایشان آن‌ها را تغییر دهند!

ساخت یک بلاک (Building a Block)

یک بلاکچین از بلاک‌های زیادی تشکیل‌شده که به یکدیگر زنجیرشده‌اند. زنجیرکردن بلاک‌ها به شکلی انجام می‌شود که به ما اجازه می‌دهد تشخیص دهیم آیا کسی یکی از بلاک‌های قبلی را دستکاری کرده است یا نه.

پس چگونه یکپارچگی (integrity) را تضمین می‌کنیم؟ هر بلاک یک هش (hash) دارد که بر اساس محتوایش محاسبه می‌شود. همچنین هش بلاک قبلی را هم در خود نگه می‌دارد.

یک کلاس Block در جاوااسکریپت می‌تواند این‌گونه باشد:

js// کتابخانه crypto-js را وارد می‌کنیم؛
// چون تابع هش sha256 به‌صورت داخلی در جاوااسکریپت وجود ندارد
const SHA256 = require("crypto-js/sha256");

class Block {
	// سازنده بلاک: ایندکس، زمان‌سنج، داده و هش بلاک قبلی را می‌گیرد
	constructor(index, timestamp, data, previousHash = '') {
		this.index = index;                 // جایگاه بلاک روی زنجیره
		this.previousHash = previousHash;   // هش بلاک قبل
		this.timestamp = timestamp;         // زمان ایجاد بلاک
		this.data = data;                   // داده‌هایی که در بلاک ذخیره می‌شود
		this.hash = this.calculateHash();   // هش این بلاک
	}

	// محاسبه هش: ایندکس + هش قبلی + زمان + داده‌ها
	calculateHash() {
		return SHA256(this.index + this.previousHash + this.timestamp + JSON.stringify(this.data)).toString();
	}
}

ابتدا کتابخانه crypto-js را وارد می‌کنم، چون تابع هش sha256 در جاوااسکریپت موجود نیست. بعد از آن یک سازنده تعریف می‌کنم که خصوصیات بلاک من را مقداردهی می‌کند. به هر بلاک یک ایندکس داده می‌شود که می‌گوید بلاک در چه جایگاهی از زنجیره قرار دارد. همچنین یک زمان‌سنج، مقداری داده برای ذخیره در بلاک و در نهایت هش بلاک قبلی را هم اضافه می‌کنیم.

ساخت زنجیره (Building the chain)

حالا می‌توانیم بلاک‌ها را در یک کلاس Blockchain به هم زنجیر کنیم! در جاوااسکریپت این‌گونه خواهد بود:

jsclass Blockchain {
	// سازنده: زنجیره را با ساخت بلاک جنسیس (بلاک اول) مقداردهی می‌کند
	constructor() {
		this.chain = [this.createGenesisBlock()];
	}

	// بلاک جنسیس: بلاک اول زنجیره که نمی‌تواند به بلاک قبلی اشاره کند
	createGenesisBlock() {
		return new Block(0, "01/01/2017", "Genesis block", "0");
	}

	// آخرین بلاک زنجیره را برمی‌گرداند
	getLatestBlock() {
		return this.chain[this.chain.length - 1];
	}

	// افزودن بلاک جدید به زنجیره
	addBlock(newBlock) {
		// هش آخرین بلاک را به‌عنوان previousHash بلاک جدید تنظیم می‌کنیم
		newBlock.previousHash = this.getLatestBlock().hash;
		// چون محتوای بلاک عوض شده، هش آن را دوباره محاسبه می‌کنیم
		newBlock.hash = newBlock.calculateHash();
		// بلاک را به زنجیره اضافه می‌کنیم
		this.chain.push(newBlock);
	}

	// بررسی صحت زنجیره: آیا کسی بلاکچین را دستکاری کرده است؟
	isChainValid() {
		for (let i = 1; i < this.chain.length; i++) {
			const currentBlock = this.chain[i];   // بلاک فعلی
			const previousBlock = this.chain[i - 1]; // بلاک قبلی

			// اگر هش فعلی با هش بازمحاسبه‌شده فرق داشت، زنجیره باطل است
			if (currentBlock.hash !== currentBlock.calculateHash()) {
				return false;
			}

			// اگر بلاک فعلی به بلاک قبلی درست اشاره نکند، زنجیره باطل است
			if (currentBlock.previousHash !== previousBlock.hash) {
				return false;
			}
		}
		// اگر همه چیز درست بود، زنجیره معتبر است
		return true;
	}
}

در سازنده، زنجیره را با ساخت یک آرایه که حاوی بلاک جنسیس (genesis block) است مقداردهی می‌کنم. بلاک اول ویژه است؛ چون نمی‌تواند به بلاک قبلی اشاره کند. همچنین دو متد اضافه کرده‌ام:

  • getLatestBlock() آخرین بلاک بلاکچین ما را برمی‌گرداند.
  • addBlock() مسئول افزودن بلاک جدید به زنجیره است. برای این کار، هش بلاک قبلی را به بلاک جدید اضافه می‌کنیم تا یکپارچگی زنجیره حفظ شود. چون محتوای بلاک جدید را تغییر داده‌ایم، باید هش آن را دوباره محاسبه کنیم. وقتی تمام شد، بلاک را روی زنجیره (آرایه) push می‌کنیم.

در نهایت متد isChainValid() را ساخته‌ام تا مطمئن شویم کسی دستکاری نکرده است. روی همه بلاک‌ها پیمایش می‌کند و بررسی می‌کند هش هر بلاک درست است. همچنین بررسی می‌کند هر بلاک با مقایسه مقدار previousHash به بلاک قبلیِ درست اشاره می‌کند. اگر همه چیز درست باشد true و اگر مشکلی باشد false برمی‌گرداند.

استفاده از بلاکچین

حالا که کلاس Blockchain تمام شده، واقعاً می‌توانیم از آن استفاده کنیم!

js// ساخت یک نمونه از بلاکچین به نام ساوجی‌کوین
let savjeeCoin = new Blockchain();

// افزودن چند بلاک آزمایشی به زنجیره
savjeeCoin.addBlock(new Block(1, "20/07/2017", { amount: 4 }));
savjeeCoin.addBlock(new Block(2, "20/07/2017", { amount: 8 }));

اینجا فقط یک نمونه جدید از Blockchain ساخته و نامش را SavjeeCoin گذاشته‌ام. بعد چند بلاک آزمایشی روی زنجیره اضافه می‌کنم. بلاک‌ها می‌توانند هر داده‌ای که می‌خواهید داشته باشند، اما در این مورد یک آبجکت با خصوصیت amount انتخاب کردم.

تلاش برای دستکاری

در مقدمه گفتم بلاکچین‌ها تغییرناپذیر هستند. با این تست مطمئن می‌شویم که این واقعاً درست است!

js// بررسی صحت زنجیره (خروجی: true)
console.log('Blockchain valid? ' + savjeeCoin.isChainValid());

// حالا داده‌ها را دستکاری می‌کنیم
savjeeCoin.chain[1].data = { amount: 100 };

// دوباره زنجیره را بررسی می‌کنیم (خروجی: false)
console.log("Blockchain valid? " + savjeeCoin.isChainValid());

با اجرای isChainValid() شروع می‌کنم تا یکپارچگی زنجیره را بررسی کنم. هیچ بلاکی را دستکاری نکرده‌ام پس true برمی‌گرداند.

بعد از آن بلاک اول زنجیره (ایندکس ۱) را برمی‌دارم و مقدار amount آن را دستکاری می‌کنم. سپس دوباره یکپارچگی زنجیره را بررسی می‌کنم و این بار تشخیص می‌دهد که مشکلی وجود دارد. زنجیره ما دیگر معتبر نیست.

بخش ۲: پیاده‌سازی اثبات کار (Proof-of-work)

مقاله اصلی: «Implementing proof-of-work (part 2)» — ویدیو: Adding proof-of-work (part 2)

در مقاله قبلی، یک بلاکچین ساده در جاوااسکریپت ساختیم تا نشان دهیم بلاکچین چگونه کار می‌کند. اما خیلی از افراد کامنت گذاشتند و گفتند که پیاده‌سازی کامل نیست و هنوز می‌توان سیستم را گول زد. آن‌ها درست می‌گویند! بلاکچین ما به سازوکار دیگری برای محافظت از خود در برابر حملات نیاز دارد. ببینیم این کار را چطور انجام دهیم!

مسئله (The problem)

در حال حاضر می‌توانیم خیلی سریع بلاک بسازیم و به بلاکچین اضافه کنیم. این سه مشکل ایجاد می‌کند:

  • اول: مردم می‌توانند با سرعت بسیار زیاد بلاک بسازند و بلاکچین ما را هرزنگاری (spam) کنند. سیل بلاک‌ها بلاکچین را غرق می‌کند و غیرقابل‌استفاده می‌شود.
  • دوم: چون ساختن بلاک معتبر خیلی آسان است، مردم می‌توانند یکی از بلاک‌های زنجیره را دستکاری کنند و سپس به‌سادگی هش همه بلاک‌های بعدی را دوباره محاسبه کنند. نتیجه یک بلاکچین «معتبر» می‌شود، حتی با وجود دستکاری!
  • سوم: می‌توانید دو مشکل بالا را ترکیب کنید تا عملاً کنترل بلاکچین را به دست بگیرید. بلاکچین‌ها با یک شبکه همتا‌به‌همتا (peer-to-peer) کار می‌کنند که در آن نودها بلاک‌هایشان را به طولانی‌ترین زنجیره اضافه می‌کنند. پس می‌توانید یک بلاک را دستکاری کنید، بقیه بلاک‌ها را بازمحاسبه کنید و بعد هر چقدر می‌خواهید بلاک اضافه کنید. در نهایت طولانی‌ترین زنجیره را خواهید داشت و همه همتاها آن را می‌پذیرند و شروع به اضافه‌کردن بلاک‌های خودشان به آن می‌کنند.

به‌وضوح برای این مشکلات به یک راه‌حل نیاز داریم. نام آن: اثبات کار (proof-of-work).

اثبات کار چیست؟

اثبات کار، سازوکاری است که قبل از ساخت اولین بلاکچین هم وجود داشت. یک تکنیک ساده است که با مطالبه مقدار مشخصی «کار محاسباتی» از سوءاستفاده جلوگیری می‌کند. همین مقدار کار، کلید جلوگیری از هرزنگاری و دستکاری است. وقتی هرزنگاری نیازمند قدرت محاسباتی زیاد باشد، دیگر به‌صرفه نیست.

بیت‌کوین اثبات کار را این‌گونه پیاده‌سازی می‌کند که هش هر بلاک باید با تعداد مشخصی صفر شروع شود. به این تعداد، سختی (difficulty) گفته می‌شود.

اما یک لحظه صبر کنید! هش یک بلاک چطور می‌تواند تغییر کند؟ در مورد بیت‌کوین، هر بلاک جزئیات یک تراکنش مالی است. مطمئناً نمی‌خواهیم فقط برای گرفتن یک هش درست، آن داده‌ها را خراب کنیم!

برای حل این مشکل، بلاکچین‌ها یک مقدار nonce اضافه می‌کنند. این یک عدد است که آن‌قدر افزایش می‌یابد تا یک هش خوب پیدا شود. و چون نمی‌توان خروجی یک تابع هش را پیش‌بینی کرد، باید ترکیب‌های زیادی را امتحان کنید تا به هشی برسید که سختی را راضی کند. جست‌وجوی یک هش معتبر (برای ساخت بلاک جدید) در دنیای ارز دیجیتال «ماینینگ» نامیده می‌شود.

در مورد بیت‌کوین، سازوکار اثبات کار تضمین می‌کند که فقط هر ۱۰ دقیقه یک بلاک اضافه شود. می‌توانید تصور کنید هرزنگارها چه سختی خواهند کشید تا شبکه را گول بزنند، وقتی برای ساخت هر بلاک جدید به این‌همه قدرت محاسباتی نیاز دارند — چه برسد به دستکاری کل زنجیره!

پیاده‌سازی اثبات کار

پس چطور آن را پیاده‌سازی می‌کنیم؟ با تغییر کلاس Block شروع می‌کنیم و متغیر nonce را به سازنده آن اضافه می‌کنیم. مقدار اولیه آن را ۰ تنظیم می‌کنم:

js// سازنده جدید بلاک با متغیر nonce
constructor(index, timestamp, data, previousHash = '') {
	this.index = index;
	this.previousHash = previousHash;
	this.timestamp = timestamp;
	this.data = data;
	this.hash = this.calculateHash();
	this.nonce = 0; // عددی که تا پیدا شدن هش معتبر افزایش می‌یابد
}

به یک متد جدید هم نیاز داریم که nonce را آن‌قدر افزایش دهد تا به یک هش معتبر برسیم. باز هم این مقدار توسط سختی تعیین می‌شود، پس سختی را به‌عنوان پارامتر دریافت می‌کنیم:

js// استخراج بلاک: nonce را افزایش می‌دهیم تا هش با تعداد صفرهای لازم شروع شود
mineBlock(difficulty) {
	// تا وقتی هش با «difficulty» عدد صفر شروع نشده، nonce را زیاد می‌کنیم
	while (this.hash.substring(0, difficulty) !== Array(difficulty + 1).join("0")) {
		this.nonce++;
		this.hash = this.calculateHash();
	}
	// وقتی هش معتبر پیدا شد، آن را چاپ می‌کنیم
	console.log("BLOCK MINED: " + this.hash);
}

و در نهایت باید تابع calculateHash() را هم تغییر دهیم؛ چون الان از متغیر nonce برای محاسبه هش استفاده نمی‌کند.

js// هش جدید: این بار nonce هم در محاسبه شرکت می‌کند
calculateHash() {
	return SHA256(this.index +
		this.previousHash +
		this.timestamp +
		JSON.stringify(this.data) +
		this.nonce
	).toString();
}

اگر همه این‌ها را کنار هم بگذارید، کلاس Block این‌شکلی می‌شود:

jsclass Block {
	// سازنده بلاک: همه خصوصیات به‌جز nonce با مقداردهی اولیه ۰
	constructor(index, timestamp, data, previousHash = '') {
		this.index = index;
		this.previousHash = previousHash;
		this.timestamp = timestamp;
		this.data = data;
		this.hash = this.calculateHash();
		this.nonce = 0;
	}

	// محاسبه هش با درنظرگرفتن nonce
	calculateHash() {
		return SHA256(this.index + this.previousHash + this.timestamp + JSON.stringify(this.data) + this.nonce).toString();
	}

	// استخراج بلاک با تنظیم سختی
	mineBlock(difficulty) {
		while (this.hash.substring(0, difficulty) !== Array(difficulty + 1).join("0")) {
			this.nonce++;
			this.hash = this.calculateHash();
		}
		console.log("BLOCK MINED: " + this.hash);
	}
}

تغییرات بلاکچین

حالا که بلاک‌های ما nonce دارند و قابل استخراج‌اند، باید مطمئن شویم بلاکچین هم از این رفتار جدید پشتیبانی می‌کند. اول یک خصوصیت جدید به بلاکچین اضافه می‌کنیم تا سختی زنجیره را نگه دارد. مقدار اولیه آن را ۲ می‌گذارم (یعنی هش بلاک‌ها باید با ۲ صفر شروع شوند):

js// سازنده بلاکچین با سختی اولیه ۲
constructor() {
	this.chain = [this.createGenesisBlock()];
	this.difficulty = 2; // هش بلاک‌ها باید با ۲ صفر شروع شوند
}

حالا فقط کافی است متد addBlock() را طوری تغییر دهیم که قبل از افزودن بلاک به زنجیره، واقعاً آن را استخراج کند. اینجا سختی را به بلاک پاس می‌دهیم:

js// افزودن بلاک: ابتدا استخراج (ماینینگ) سپس push به زنجیره
addBlock(newBlock) {
	newBlock.previousHash = this.getLatestBlock().hash;
	newBlock.mineBlock(this.difficulty); // استخراج بلاک با سختی زنجیره
	this.chain.push(newBlock);
}

همین! بلاکچین ما حالا اثبات کار دارد و در برابر هرزنگاری و دستکاری محافظت می‌شود.

این بخش سوم از سری مقالات ساخت بلاکچین با جاوااسکریپت است. در مقاله‌های قبلی یک بلاکچین ساده ساختیم و برای محافظت در برابر هرزنگارها و مهاجمان، اثبات کار پیاده‌سازی کردیم. اما در طول راه میان‌برهایی زدیم: بلاکچین ما فقط می‌تواند ۱ تراکنش در هر بلاک ذخیره کند و پاداشی هم برای ماینرها وجود ندارد. بیایید آن را درست کنیم!

بازسازی کلاس Block

در حال حاضر یک بلاک خصوصیات index، previousHash، timestamp، data، hash و nonce دارد. خصوصیت index واقعاً کاربردی نیست؛ راستش حتی نمی‌دانم اول چرا اضافه‌اش کردم. پس بیایید آن را حذف کنیم و نام data را به transactions تغییر دهیم که منطقی‌تر است:

jsclass Block {
	// سازنده جدید: به‌جای index و data، زمان و لیست تراکنش‌ها را می‌گیرد
	constructor(timestamp, transactions, previousHash = '') {
		this.previousHash = previousHash;
		this.timestamp = timestamp;
		this.transactions = transactions; // لیست تراکنش‌های داخل بلاک
		this.hash = this.calculateHash();
		this.nonce = 0;
	}
}

وقتی کلاس Block را تغییر می‌دهیم، باید تابع calculateHash() را هم عوض کنیم؛ چون هنوز از خصوصیات قدیمی index و data استفاده می‌کند که حذف کردیم:

js// محاسبه هش: این بار با تراکنش‌ها به‌جای data
calculateHash() {
	return SHA256(this.previousHash + this.timestamp + JSON.stringify(this.transactions) + this.nonce).toString();
}

کلاس Transaction

داخل هر بلاک می‌توانیم چند تراکنش ذخیره کنیم. پس بیایید یک کلاس Transaction هم تعریف کنیم تا مشخص کنیم یک تراکنش باید چه خصوصیاتی داشته باشد:

jsclass Transaction {
	// یک تراکنش: فرستنده، گیرنده و مقدار
	constructor(fromAddress, toAddress, amount) {
		this.fromAddress = fromAddress; // آدرس فرستنده
		this.toAddress = toAddress;     // آدرس گیرنده
		this.amount = amount;           // مقدار انتقالی
	}
}

در این مثال یک تراکنش خیلی ساده است و فقط شامل فرستنده (fromAddress)، گیرنده (toAddress) و یک مقدار است. در صورت نیاز می‌توانید فیلدهای بیشتری به تراکنش اضافه کنید، اما این‌ها حداقل مطلق هستند.

سازگارکردن بلاکچین

حالا بزرگ‌ترین کار: هماهنگ‌کردن بلاکچین با همه این تغییرات. اولین چیزی که نیاز داریم جایی برای ذخیره تراکنش‌های در انتظار (pending) است.

همان‌طور که می‌دانید بلاکچین‌ها با کمک الگوریتم اثبات کار در بازه‌های زمانی منظم بلاک می‌سازند. در مورد بیت‌کوین، سختی طوری تنظیم می‌شود که تقریباً هر ۱۰ دقیقه یک بلاک جدید ساخته شود. اما باید بتوان بین ساخت دو بلاک، تراکنش‌های جدیدی ثبت کرد.

برای این کار، سازنده Blockchain را طوری تغییر می‌دهیم که جایی برای ذخیره تراکنش‌های در انتظار داشته باشد. همچنین یک خصوصیت تعریف می‌کنیم که مشخص می‌کند ماینر بابت تلاشش چند سکه پاداش می‌گیرد:

jsclass Blockchain {
	// سازنده: زنجیره + سختی + تراکنش‌های در انتظار + پاداش ماینر
	constructor() {
		this.chain = [this.createGenesisBlock()];
		this.difficulty = 5;

		// جایی برای ذخیره تراکنش‌ها بین ساخت دو بلاک
		this.pendingTransactions = [];

		// پاداشی که ماینر بابت تلاشش دریافت می‌کند (سکه)
		this.miningReward = 100;
	}
}

قدم بعدی: متد addBlock() را باید تغییر دهیم — و با «تغییر» یعنی کاملاً حذفش کنیم و از نو بنویسیم! دیگر اجازه نمی‌دهیم مردم مستقیماً بلاک به زنجیره اضافه کنند. در عوض باید تراکنش اضافه کنند که در بلاک بعدی گنجانده می‌شود. پس متد addBlock() را با createTransaction() جایگزین کنیم که منطقی‌تر است:

js// ثبت تراکنش: تراکنش را به لیست در انتظار اضافه می‌کند
createTransaction(transaction) {
	// اینجا باید بررسی‌هایی هم انجام شود!

	// اضافه‌کردن به آرایه «تراکنش‌های در انتظار»
	this.pendingTransactions.push(transaction);
}

استخراج بلاک (ماینینگ)

مردم حالا می‌توانند تراکنش‌های جدید به لیست در انتظار اضافه کنند. اما به هر حال باید آن‌ها را از لیست خارج و داخل بلاک‌های واقعی قرار دهیم. برای این کار متدی به نام minePendingTransactions() می‌سازیم. این متد نه‌تنها یک بلاک جدید با همه تراکنش‌های در انتظار استخراج می‌کند، بلکه پاداش ماینینگ را هم به ماینر می‌فرستد:

js// استخراج یک بلاک شامل همه تراکنش‌های در انتظار + ارسال پاداش به ماینر
minePendingTransactions(miningRewardAddress) {
	// بلاک جدید با همه تراکنش‌های در انتظار می‌سازیم و استخراجش می‌کنیم
	let block = new Block(Date.now(), this.pendingTransactions);
	block.mineBlock(this.difficulty);

	// بلاک استخراج‌شده را به زنجیره اضافه می‌کنیم
	this.chain.push(block);

	// لیست تراکنش‌های در انتظار را خالی می‌کنیم و پاداش ماینر را ثبت می‌کنیم
	this.pendingTransactions = [
		new Transaction(null, miningRewardAddress, this.miningReward)
	];
}

توجه کنید که این متد یک آرگومان miningRewardAddress می‌گیرد. اگر شروع به ماینینگ کنید، می‌توانید آدرس کیف پولتان را به این متد بدهید. وقتی با موفقیت یک بلاک را استخراج کنید، سیستم یک تراکنش جدید برای دادن پاداش ماینینگ (در این مورد ۱۰۰ سکه) می‌سازد.

یک نکته: در این پیاده‌سازی همه تراکنش‌های در انتظار را گرفته و داخل یک بلاک می‌گذاریم. در واقعیت اما این کار نمی‌کند؛ چون اندازه بلاک محدود است. در مورد بیت‌کوین سقف اندازه بلاک ۲ مگابایت است. اگر تراکنش‌ها بیشتر از ظرفیت یک بلاک باشند، ماینر انتخاب می‌کند کدام تراکنش را بگنجاند و کدام را نه (معمولاً تراکنش‌هایی با کارمزد بالاتر برنده‌اند).

موجودی یک آدرس (Balance)

قبل از تست کد، یک کار دیگر انجام دهیم! خوب است بتوانیم موجودی آدرس‌های بلاکچین‌مان را بررسی کنیم:

js// محاسبه موجودی یک آدرس با پیمایش همه بلاک‌ها و تراکنش‌ها
getBalanceOfAddress(address) {
	let balance = 0; // از صفر شروع می‌کنیم!

	// روی هر بلاک و هر تراکنش داخل بلاک پیمایش می‌کنیم
	for (const block of this.chain) {
		for (const trans of block.transactions) {

			// اگر آدرس بدهد فرستنده بود، موجودی کم می‌شود
			if (trans.fromAddress === address) {
				balance -= trans.amount;
			}

			// اگر آدرس گیرنده بود، موجودی زیاد می‌شود
			if (trans.toAddress === address) {
				balance += trans.amount;
			}
		}
	}

	return balance;
}

محدودیت‌ها و جمع‌بندی

الان بلاکچین کوچک ما می‌تواند چند تراکنش را در یک بلاک ذخیره کند و به ماینرها پاداش بدهد.

ساخت یک تولیدکننده کلید (Key generator)

برای حل این مشکل باید تراکنش‌ها را امضا کنیم. این کار را با یک جفت‌کلید عمومی/خصوصی انجام می‌دهیم. کلید عمومی به‌عنوان آدرس کیف پول ما استفاده می‌شود و آزادانه قابل اشتراک‌گذاری است، در حالی که کلید خصوصی برای امضای تراکنش‌ها استفاده می‌شود. این یعنی فقط در صورتی می‌توانیم از کیف پول پول خرج کنیم که کلید خصوصی مرتبط با آن را داشته باشیم. همین سیستم در بیت‌کوین، اتریوم و سایر ارزهای دیجیتال استفاده می‌شود.

پس با تولید یک کلید عمومی و خصوصی برای خودمان شروع می‌کنیم. برای این کار فایلی به نام keygenerator.js می‌سازم:

const publicKey = key.getPublic('hex');
const privateKey = key.getPrivate('hex');

// کلیدها را در کنسول چاپ می‌کنیم

تغییر کلاس Transaction

حالا باید کلاس Transaction خود را تغییر دهیم. متدهایی اضافه می‌کنیم که به ما اجازه می‌دهند یک تراکنش را امضا کنیم، امضا را بررسی کنیم و هش تراکنش‌ها را هم محاسبه می‌کنیم.

با هش شروع کنیم:

js// محاسبه هش تراکنش از روی فرستنده، گیرنده و مبلغ
calculateHash() {
	return SHA256(this.fromAddress + this.toAddress + this.amount)
			.toString();
}

چرا باید برای هر تراکنش هش محاسبه کنیم؟ چون نمی‌خواهیم همه داده‌های داخل آبجکت Transaction را امضا کنیم. فقط هش تراکنش را امضا می‌کنیم.

بعد، متدی به نام signTransaction می‌سازیم که باید یک signingKey به‌عنوان پارامتر بگیرد:

js// امضای تراکنش با کلید خصوصی فرستنده
signTransaction(signingKey) {
	// فقط می‌توان از کیف پولی که کلیدش را داریم تراکنش فرستاد؛
	// پس بررسی می‌کنیم که fromAddress با کلید عمومی مطابقت دارد
	if (signingKey.getPublic('hex') !== this.fromAddress) {
		throw new Error('You cannot sign transactions for other wallets!');
	}

	// هش تراکنش را محاسبه و با کلید امضا می‌کنیم
	const hashTx = this.calculateHash();
	const sig = signingKey.sign(hashTx, 'base64');
	this.signature = sig.toDER('hex');
}

همچنین یک بررسی اضافه کرده‌ایم که مطمئن شود کلید داده‌شده با فرستنده تراکنش مطابقت دارد. یادتان باشد: فقط زمانی می‌توانید تراکنش‌ها را امضا کنید که از جیب خودتان بیرون آمده باشند. نمی‌توانید سکه‌های دیگری را خرج کنید.

در نهایت یک متد isValid اضافه می‌کنیم که امضای روی تراکنش را بررسی می‌کند و اگر همه چیز درست بود true برمی‌گرداند و اگر کسی سعی کرده باشد تراکنش را دستکاری کند و امضا نامعتبر باشد false:

js// بررسی صحت امضای تراکنش با کلید عمومی فرستنده
isValid() {
	// اگر تراکنش فرستنده ندارد (مثل پاداش ماینینگ)، معتبر فرض می‌کنیم
	if (this.fromAddress === null) return true;

	// اگر امضایی روی تراکنش نیست، خطا پرتاب می‌کنیم
	if (!this.signature || this.signature.length === 0) {
		throw new Error('No signature in this transaction');
	}

	// امضا را با کلید عمومی فرستنده اعتبارسنجی می‌کنیم
	const publicKey = ec.keyFromPublic(this.fromAddress, 'hex');
	return publicKey.verify(this.calculateHash(), this.signature);
}

کلاس Block

برای قراردادن یک تراکنش در بلاکچین، باید آن را به یک Block اضافه کنید. یک بلاک می‌تواند چند تراکنش داشته باشد؛ پس برای راحتی کارمان، یک متد ساده می‌سازم که بررسی می‌کند همه تراکنش‌های داخل بلاک معتبر و درست امضاشده‌اند:

js// بررسی معتبربودن همه تراکنش‌های داخل بلاک
hasValidTransactions() {
	// روی همه تراکنش‌ها پیمایش و متد isValid آن‌ها را صدا می‌زنیم
	for (const tx of this.transactions) {
		if (!tx.isValid()) {
			return false;
		}
	}

	return true;
}

قدم بعدی، مطمئن‌شدن از این است که کاربر فقط می‌تواند یک تراکنش معتبر به بلاکچین اضافه کند. یعنی بررسی کنیم آدرس‌های from و to پر شده‌اند و تراکنش به‌درستی امضاشده است:

js// افزودن تراکنش معتبر به لیست در انتظار
addTransaction(transaction) {

		throw new Error('Transaction must include from and to address');
	}

	// اگر تراکنش معتبر نباشد (امضا اشتباه/دستکاری)، رد می‌کنیم
	if (!transaction.isValid()) {
		throw new Error('Cannot add invalid transaction to chain');
	}

	// تراکنش معتبر را به لیست در انتظار اضافه می‌کنیم
	this.pendingTransactions.push(transaction);
}

src/blockchain.js — کلاس‌های Transaction، Block و Blockchain

js'use strict';
const crypto = require('crypto');
const EC = require('elliptic').ec;
const ec = new EC('secp256k1');
const debug = require('debug')('savjeecoin:blockchain');

class Transaction {
  /**
   * @param {string} fromAddress آدرس فرستنده
   * @param {string} toAddress آدرس گیرنده
   * @param {number} amount مبلغ تراکنش
   */
  constructor(fromAddress, toAddress, amount) {
    this.fromAddress = fromAddress;
    this.toAddress = toAddress;
    this.amount = amount;
    this.timestamp = Date.now();
  }

  /**
   * هش SHA256 تراکنش را می‌سازد
   *
   * @returns {string}
   */
  calculateHash() {
    return crypto
      .createHash('sha256')
      .update(this.fromAddress + this.toAddress + this.amount + this.timestamp)
      .digest('hex');
  }

  /**
   * تراکنش را با کلید امضا (signingKey که یک جفت‌کلید elliptic شامل کلید خصوصی است)
   * امضا می‌کند. امضا داخل آبجکت تراکنش ذخیره و بعداً روی بلاکچین ثبت می‌شود.
   *
   * @param {string} signingKey
   */
  sign(signingKey) {
    // فقط می‌توان از کیف پولی که کلیدش را دارید تراکنش فرستاد؛
    // پس بررسی می‌کنیم fromAddress با کلید عمومی مطابقت دارد
    if (signingKey.getPublic('hex') !== this.fromAddress) {
      throw new Error('You cannot sign transactions for other wallets!');
    }

    // هش تراکنش را محاسبه، با کلید امضا
    // و داخل آبجکت تراکنش ذخیره می‌کنیم
    const hashTx = this.calculateHash();
    const sig = signingKey.sign(hashTx, 'base64');

    this.signature = sig.toDER('hex');
  }

  /**
   * بررسی می‌کند امضا معتبر است (تراکنش دستکاری نشده).
   * از fromAddress به‌عنوان کلید عمومی استفاده می‌کند.
   *
   * @returns {boolean}
   */
  isValid() {
    // اگر تراکنش آدرس فرستنده ندارد، فرض می‌کنیم پاداش ماینینگ
    // است و معتبر است (می‌توانید به شکل دیگری هم بررسی کنید)
    if (this.fromAddress === null) return true;

    if (!this.signature || this.signature.length === 0) {
      throw new Error('No signature in this transaction');
    }

    const publicKey = ec.keyFromPublic(this.fromAddress, 'hex');
    return publicKey.verify(this.calculateHash(), this.signature);
  }
}

class Block {
  /**
   * @param {number} timestamp
   * @param {Transaction[]} transactions
   * @param {string} previousHash
   */
  constructor(timestamp, transactions, previousHash = '') {
    this.previousHash = previousHash;
    this.timestamp = timestamp;
    this.transactions = transactions;
    this.nonce = 0;
    this.hash = this.calculateHash();
  }

  /**
   * هش SHA256 این بلاک را برمی‌گرداند
   * (با پردازش همه داده‌های ذخیره‌شده داخل بلاک)
   *
   * @returns {string}
   */
  calculateHash() {
    return crypto
      .createHash('sha256')
      .update(
        this.previousHash +
          this.timestamp +
          JSON.stringify(this.transactions) +
          this.nonce
      )
      .digest('hex');
  }

  /**
   * فرایند ماینینگ را روی بلاک شروع می‌کند. مقدار «nonce» را آن‌قدر تغییر
   * می‌دهد تا هش بلاک با تعداد کافی صفر (= سختی) شروع شود.
   *
   * @param {number} difficulty
   */
  mineBlock(difficulty) {
    while (
      this.hash.substring(0, difficulty) !== Array(difficulty + 1).join('0')
    ) {
      this.nonce++;
      this.hash = this.calculateHash();
    }

    debug(`Block mined: ${this.hash}`);
  }

  /**
   * همه تراکنش‌های داخل این بلاک را (امضا + هش) اعتبارسنجی می‌کند.
   * اگر همه چیز درست باشد true و اگر بلاک نامعتبر باشد false.
   *
   * @returns {boolean}
   */
  hasValidTransactions() {
    for (const tx of this.transactions) {
      if (!tx.isValid()) {
        return false;
      }
    }

    return true;
  }
}

class Blockchain {
  constructor() {
    this.chain = [this.createGenesisBlock()];
    this.difficulty = 2;
    this.pendingTransactions = [];
    this.miningReward = 100;
  }

  /**
   * @returns {Block}
   */
  createGenesisBlock() {
    return new Block(Date.parse('2017-01-01'), [], '0');
  }

  /**
   * آخرین بلاک زنجیره را برمی‌گرداند. وقتی می‌خواهید بلاک جدید بسازید
   * و به هش بلاک قبلی نیاز دارید، مفید است.
   *
   * @returns {Block[]}
   */
  getLatestBlock() {
    return this.chain[this.chain.length - 1];
  }

  /**
   * همه تراکنش‌های در انتظار را می‌گیرد، داخل یک بلاک قرار می‌دهد و فرایند
   * ماینینگ را شروع می‌کند. همچنین یک تراکنش برای ارسال پاداش ماینینگ
   * به آدرس داده‌شده اضافه می‌کند.
   *
   * @param {string} miningRewardAddress
   */
  minePendingTransactions(miningRewardAddress) {
    const rewardTx = new Transaction(
      null,
      miningRewardAddress,
      this.miningReward
    );
    this.pendingTransactions.push(rewardTx);

    const block = new Block(
      Date.now(),
      this.pendingTransactions,
      this.getLatestBlock().hash
    );
    block.mineBlock(this.difficulty);

    debug('Block successfully mined!');
    this.chain.push(block);

    this.pendingTransactions = [];
  }

  /**
   * یک تراکنش جدید به لیست در انتظار اضافه می‌کند (تا دفعه بعد که ماینینگ
   * شروع شد اضافه شود). بررسی می‌کند تراکنش به‌درستی امضاشده است.
   *
   * @param {Transaction} transaction
   */
  addTransaction(transaction) {
    if (!transaction.fromAddress || !transaction.toAddress) {
      throw new Error('Transaction must include from and to address');
    }

    // بررسی معتبربودن تراکنش
    if (!transaction.isValid()) {
      throw new Error('Cannot add invalid transaction to chain');
    }

    if (transaction.amount <= 0) {
      throw new Error('Transaction amount should be higher than 0');
    }

    // مطمئن می‌شویم مقدار ارسال‌شده از موجودی فعلی بیشتر نیست
    const walletBalance = this.getBalanceOfAddress(transaction.fromAddress);
    if (walletBalance < transaction.amount) {
      throw new Error('Not enough balance');
    }

    // دریافت همه تراکنش‌های در انتظار دیگر برای کیف پول «from»
    const pendingTxForWallet = this.pendingTransactions.filter(
      tx => tx.fromAddress === transaction.fromAddress
    );

    // اگر کیف پول تراکنش‌های در انتظار بیشتری دارد، مجموع سکه‌های هزینه‌شده
    // تا این لحظه را محاسبه می‌کنیم. اگر از موجودی بیشتر شد، از اضافه‌کردن
    // این تراکنش خودداری می‌کنیم.
    if (pendingTxForWallet.length > 0) {
      const totalPendingAmount = pendingTxForWallet
        .map(tx => tx.amount)
        .reduce((prev, curr) => prev + curr);

      const totalAmount = totalPendingAmount + transaction.amount;
      if (totalAmount > walletBalance) {
        throw new Error(
          'Pending transactions for this wallet is higher than its balance.'
        );
      }
    }

    this.pendingTransactions.push(transaction);
    debug('transaction added: %s', transaction);
  }

  /**
   * موجودی یک آدرس کیف پول را برمی‌گرداند.
   *
   * @param {string} address
   * @returns {number} موجودی کیف پول
   */
  getBalanceOfAddress(address) {
    let balance = 0;

    for (const block of this.chain) {
      for (const trans of block.transactions) {
        if (trans.fromAddress === address) {
          balance -= trans.amount;
        }

        if (trans.toAddress === address) {
          balance += trans.amount;
        }
      }
    }

    debug('getBalanceOfAdrees: %s', balance);
    return balance;
  }

  /**
   * لیست همه تراکنش‌هایی که به یک آدرس کیف پول ارسال یا از آن دریافت
   * شده‌اند را برمی‌گرداند.
   *
   * @param  {string} address
   * @return {Transaction[]}
   */
  getAllTransactionsForWallet(address) {
    const txs = [];

    for (const block of this.chain) {
      for (const tx of block.transactions) {
        if (tx.fromAddress === address || tx.toAddress === address) {
          txs.push(tx);
        }
      }
    }

    debug('get transactions for wallet count: %s', txs.length);
    return txs;
  }

  /**
   * روی همه بلاک‌های زنجیره پیمایش می‌کند و بررسی می‌کند درست به هم
   * متصل شده‌اند و کسی هش‌ها را دستکاری نکرده. با بررسی بلاک‌ها،
   * تراکنش‌های (امضاشده) داخل آن‌ها را هم اعتبارسنجی می‌کند.
   *
   * @returns {boolean}
   */
  isChainValid() {
    // بررسی دستکاری‌نشدن بلاک جنسیس با مقایسه خروجی
    // createGenesisBlock با اولین بلاک زنجیره
    const realGenesis = JSON.stringify(this.createGenesisBlock());

    if (realGenesis !== JSON.stringify(this.chain[0])) {
      return false;
    }

    // بررسی بلاک‌های باقی‌مانده زنجیره از نظر صحت هش و امضا
    for (let i = 1; i < this.chain.length; i++) {
      const currentBlock = this.chain[i];
      const previousBlock = this.chain[i - 1];

      if (previousBlock.hash !== currentBlock.previousHash) {
        return false;
      }

      if (!currentBlock.hasValidTransactions()) {
        return false;
      }

      if (currentBlock.hash !== currentBlock.calculateHash()) {
        return false;
      }
    }

    return true;
  }
}

module.exports.Blockchain = Blockchain;
module.exports.Block = Block;
module.exports.Transaction = Transaction;
