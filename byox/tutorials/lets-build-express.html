lets-build-express — کتابخانه اکسپرس را از صفر بسازیم

⚙ زبان: Node.js ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمه فارسی سری «lets-build-express» از Anto Aravinth است — ساخت کتابخانه Express از صفر فصل به فصل، با همان کد، نام متغیرها و ساختار خودِ express، تا علاوه بر یادگیری، بتوانید مستقیم در پروژه express مشارکت کنید.
↗ مشاهده منبع اصلی

۱. چرا این سری؟

بسیاری از توسعه‌دهنده‌ها در فهمیدن اینکه next چگونه کار می‌کند یا ترتیب مسیرها در express به چه شکل است گیج می‌شوند. با ساختن خود کتابخانه express، درک محکم‌تری به دست می‌آوریم — و چه لذتی دارد! فهرست فصل‌ها (فصل ۶ — مدیریت خطا — در حال نوشتن است):

  1. ساخت انتزاع‌های express
  2. مدیریت مسیرها از جعبه
  3. پیاده‌سازی تابع next
  4. توسعه شیء Response
  5. پیاده‌سازی send / json روی response
  6. مدیریت خطا (WIP)

۲. فصل ۱ — ساخت انتزاع‌های express

در این فصل انتزاع‌های لازم برای ساختن express ساخته می‌شود و در پایان یک کتابخانه بسیار کوچک داریم که فقط یک مسیر را مطابقت می‌دهد. اول فایل express.js ما — با export:

javascriptexports = module.exports = createApplication;

function createApplication() {
    var app = function(req,res,next) {
    };

    return app;
}

هیچ‌چیز فانتزی‌ای این‌جا نیست؛ فقط تابع createApplication را export می‌کنیم. این تابع ارجاعی به تابعِ app می‌سازد و آن را برمی‌گرداند. همان app برگشتی است که کاربر موقع صدا زدن express() می‌گیرد. توجه کنید که تابع انتظار req و res و حتی next را دارد (req درخواست ورودی، res شیء پاسخ http و next تابع مخصوص express است — اینکه چه کسی این‌ها را می‌فرستد را بعداً می‌بینیم). پس کاربر موقع express()، app ما را می‌گیرد؛ پس باید متدهای لازم مثل get و post را به شیء app اضافه کنیم.

Application.js — جایی که متدهایی مثل get و use زندگی می‌کنند

فایل app.js را می‌سازیم که متدهای get، post و... را در معرض می‌گذارد. کد اولیه:

javascriptvar app = exports = module.exports = {};

app.init = function() {
    this.cache = {};
    this.engines = {};
    this.settings = {}

    // برای نگه‌داشتن روتر برنامه
    this._router = undefined;
};

رویکرد کد منبع express بسیار ساده است؛ از کتابخانه‌ای به نام methods استفاده می‌کنند که فقط همه متدهای HTTP را به‌صورت حروف کوچک برمی‌گرداند:

javascriptvar methods = require('methods');
var slice = Array.prototype.slice;

methods.forEach(function (method){
    app[method] = function(path) {
        this.lazyrouter()

        var route = this._router.route(path);

        route[method].apply(route, slice.call(arguments, 1));
        return this;
    }
});

داخل هر متد، اول this.lazyrouter() صدا زده می‌شود — یعنی برای برنامه فقط یک Router بساز (و اگر ساخته شده، دیگر نساز):

javascriptapp.lazyrouter = function lazyrouter() {
    if(!this._router) {
        this._router = new Router({})
    }
};
به قطعه var route = this._router.route(path); route[method].apply(route, slice.call(arguments, 1)); بعداً در همین فصل برمی‌گردیم.

Router.js — منطق مسیریابی؛ فقط یکی برای هر برنامه

javascriptvar proto = module.exports = function(options) {
    var opts = options || {}

    function router(req,res,next) {
        router.handle(req,res,next)
    }

    setPrototypeOf(router, proto)

    /* مخصوص express؛ در فصل‌های بعدی به آن‌ها می‌رسیم */
    router.params = {};
    router._params = [];
    router.caseSensitive = opts.caseSensitive;
    router.mergeParams = opts.mergeParams;
    router.strict = opts.strict;
    router.stack = []; // خاصیت بسیار مهم

    return router;
};

خاصیت مهم روتر، stack است — جایی که پیکربندی مسیرها می‌رود. تابع route به کاربر اجازه می‌دهد مسیرهای خاصی به stack اضافه کند:

javascriptproto.route = function route(path) {
    var route = new Route(path)

    var layer = new Layer(path,{},route.dispatch.bind(route))

    layer.route = route;

    this.stack.push(layer);

    return route;
};

برای هر path یک Route و یک Layer می‌سازیم؛ layer حاوی route است و در stack روتر push می‌شود. (کد route.dispatch.bind(route) فعلاً تابعی خالی است — فصل‌های بعدی.)

Route.js — اطلاعات مسیر و لایه موردنظر

javascriptfunction Route(path) {
    this.path = path;
    this.stack = [];

    this.methods = {}
}

Route هم مثل application همه متدهای HTTP را پیاده می‌کند:

javascriptmethods.forEach(function(method){
    Route.prototype[method] = function(){
        var handles = flatten(Array.prototype.slice.call(arguments));

        for (var i = 0; i < handles.length; i++) {
            var handle = handles[i];

            if (typeof handle !== 'function') {
                var type = toString.call(handle);
                var msg = 'Route.' + method + '() requires a callback function but got a ' + type
                throw new Error(msg);
            }

            var layer = Layer('/', {}, handle);
            layer.method = method;

            this.methods[method] = true;
            this.stack.push(layer);
        }

        return this;
    };
});

توجه کنید route هم stack خودش را دارد با نمونه‌های Layer خودش.

Layer.js — مسیر و ارجاع تابعی که باید اجرا شود

javascriptmodule.exports = Layer

function Layer(path, options, fn) {
    if (!(this instanceof Layer)) {
        return new Layer(path, options, fn);
    }

    this.handle = fn;
    this.name = fn.name || '<anonymous>';
    this.params = undefined;
    this.path = undefined;
}

فقط handle و نام تابع را نگه می‌دارد.

برگردیم به application.js

حالا قطعه حل‌نشده قبلی را می‌فهمیم: بعد از ساخت lazy router، مشخصات router با فراخوانی route تنظیم می‌شود. چون application فقط یک روتر دارد و Route هم همه متدهای HTTP را دارد، این خط route[method].apply(route, slice.call(arguments, 1)) در واقع یک Layer در Route می‌سازد که handleاش همان آرگومان ورودی است (همان هندلر که موقع مطابقت مسیر باید اجرا شود).

حالا تابع listen را هم روی application در معرض می‌گذاریم:

javascriptapp.listen = function listen() {
    var server = http.createServer(this);
    return server.listen.apply(server, arguments);
};

سپس express.js را کامل می‌کنیم — mixin همه متدهای app.js را به تابعِ app کپی می‌کند (وقتی ./app import می‌شود، بلوک متدها اجرا شده و همه متدهای HTTP آماده‌اند):

javascriptvar proto = require("./app")

function createApplication() {
    let app = function(req,res,next) {
        app.handle(req,res,next)
    };

    mixin(app,proto,false);

    app.init();
    return app;
}

نکته ظریف: http.createServer(this) یک تابع callback می‌خواهد که با req و res صدا زده می‌شود؛ this همان app است و app.handle کار را به روتر می‌سپارد:

javascriptapp.handle = function handle(req, res, callback) {
    var router = this._router;

    router.handle(req, res);
};

اجرا: اولین اپلیکیشن ما

javascriptlet express = require('./src/express')
const app = express()

app.get('/', (req, res) => {
    res.writeHead(200)
    res.write('Hello World');
    res.end();
});


app.listen(3000, () => console.log('Example app listening on port 3000!'))

برای اینکه پاسخ واقعاً برسد، Router باید handle را پیاده کند — نویسنده اول فقط stack را چاپ می‌کند: وقتی درخواست / شلیک می‌شود، خروجی مثل این است: [{"name":"bound dispatch","route":{"path":"/","stack":[{"name":"<anonymous>","method":"get"}],"methods":{"get":true}}}] — یعنی روی / یک get تعریف شده است! (رویکرد موقتی: تکه‌ای که فقط هندلر ساخته‌شده در route.stack[0] را صدا می‌زند.) برای اجرای هندلر، Layer.prototype.handle_request اضافه می‌شود:

javascriptLayer.prototype.handle_request = function handle(req,res,next) {
    var fn = this.handle;

    try {
        fn(req, res, next);
    } catch (err) {
        console.error(err)
    }
}

حالا دوباره سرور را اجرا کنید و / را بزنید — callback ما با درخواست و پاسخ جاری صدا زده می‌شود و «Hello World» با write و end فرستاده می‌شود. (نکته: فعلاً res.send نداریم چون هنوز شیء پاسخ را توسعه نداده‌ایم؛ پس حتماً res.end() را فراموش نکنید.)

۳. فصل ۲ — مدیریت مسیرها از جعبه

در این فصل handle روتر بهبود می‌یابد تا مسیرها را با URL درخواست مطابقت دهد. لایه‌ها یک تابع match می‌گیرند: اگر path مسیر درخواست به الگوی layer بخورد، layer انتخاب می‌شود. الگوها به‌صورت عبارت باقاعده (regex) کامپایل می‌شوند تا پارامترهای مسیر (مثل /user/:id) هم کار کنند:

javascriptLayer.prototype.match = function match(path) {
    var match = path.match(this.regexp);
    if (match) {
        this.params = {};
        // پارامترهای ستاره‌دار مسیر در this.params می‌نشینند
    }
    return !!match;
};
javascriptproto.handle = function handle(req, res, out) {
    var self = this;
    var stack = self.stack;
    var path = req.url;

    // روی همه لایه‌های stack بچرخ تا اولین تطبیق پیدا شود
    for (var i = 0; i < stack.length; i++) {
        var layer = stack[i];
        if (layer.match(path)) {
            var route = layer.route;
            if (route && route.handle_method(method, req)) {
                // مسیر و متد هر دو تطبیق‌اند: هندلرهای route را اجرا کن
                route.stack[0].handle_request(req, res);
            }
        }
    }
}

در انتهای فصل، برنامه می‌تواند چند مسیر (/، /about،...) و حتی مسیرهای با پارامتر را درست جواب دهد و برای مسیرهای ناشناخته 404 بدهد.

۴. فصل ۳ — پیاده‌سازی تابع next

این فصل مهم‌ترین مفهوم express را می‌سازد: زنجیره اجرای میدل‌ورها. متد use به برنامه اضافه می‌شود تا میدل‌ورها (بدون مسیر یا با مسیر) را ثبت کند. تابع next به هر هندلر پاس داده می‌شود و کنترل را به «هندلر بعدی در صف» می‌دهد. در داخل handle_request، فراخوانی fn(req, res, next) اجرا می‌شود و next یعنی self.handle_request برای لایه بعدی:

javascriptfunction next(err) {
    // هندلر بعدی را در stack روتر پیدا کن و اجرا کن
    var layer = stack[i++];
    if (!layer) return done(err);

    if (layer.match(path)) {
        layer.handle_request(req, res, next);
    } else {
        next(err); // رد شو
    }
}

اینجاست که مشخص می‌شود «ترتیب ثبت» در express چرا اهمیت دارد: اگر یک میدل‌ور روی مسیر /foo قبل از مسیر / ثبت شود، به ترتیب صف اجرا می‌شوند و فقط وقتی هندلری next() صدا بزند، سراغ بعدیِ صف می‌رویم. این همان رفتاری است که به برنامه‌نویسان تازه‌کار express گیج می‌کند.

۵. فصل ۴ — توسعه شیء Response

در این فصل، responsive اصلی express به شیء پاسخ Node متصل می‌شود: متدهای res.status(code) برای تنظیم res.statusCode، res.set(field, value) برای تنظیم سربرگ از طریق res.setHeader و res.get(field) برای خواندن از res.getHeader. این متدها برای زنجیره‌ای بودن (chainable) خود را return this می‌کنند:

javascriptres.status = function status(code) {
    this.statusCode = code;
    return this;
};

res.set = function set(field, value) {
    if (typeof field === 'object') {
        for (var key in field) this.setHeader(key, field[key]);
    } else {
        this.setHeader(field, value);
    }
    return this;
};

res.get = function get(field) {
    return this.getHeader(field);
};

این متدها با app.response (شیء اولیه پاسخ که همه نمونه‌ها از آن می‌سازند) به همه پاسخ‌ها تزریق می‌شوند — همان الگویی که خود express برای افزودن امکانات روی res و req استفاده می‌کند.

۶. فصل ۵ — پیاده‌سازی send و json

حالا نوبت راحت‌ترین APIهاست: res.send() که بر اساس نوع ورودی تصمیم می‌گیرد — اگر رشته به هم برسد، Content-Type: text/html می‌گذارد؛ اگر آبجکت یا آرایه باشد، JSON می‌فرستد با Content-Type: application/json؛ و همیشه Content-Length را بر اساس طول بدنه تنظیم می‌کند (چون بدون آن، بافر شدن و اتصال‌های نگه‌داشته‌شده در Node باگ می‌سازد):

javascriptres.send = function send(body) {
    var chunk = body;

    switch (typeof chunk) {
        case 'string':
            if (!this.get('Content-Type')) {
                this.set('Content-Type', 'text/html');
            }
            break;
        case 'boolean':
        case 'number':
        case 'object':
            if (chunk === null) chunk = '';
            else if (Buffer.isBuffer(chunk)) {
                if (!this.get('Content-Type')) {
                    this.set('Content-Type', 'application/octet-stream');
                }
            } else {
                return this.json(chunk);
            }
            break;
    }

    if (typeof chunk === 'string') {
        this.set('Content-Length', Buffer.byteLength(chunk));
    }
    this.end(chunk);
};

res.json = function json(obj) {
    if (!this.get('Content-Type')) {
        this.set('Content-Type', 'application/json');
    }
    var body = JSON.stringify(obj);
    this.set('Content-Length', Buffer.byteLength(body));
    this.end(body);
};

حالا اپلیکیشن «قدیمی» فصل ۱ به شکل مدرن express بازنویسی می‌شود:

javascriptconst express = require('./src/express');
const app = express();

app.get('/', (req, res) => {
    res.send('Hello World!');
});

app.get('/user/:id', (req, res) => {
    res.json({ id: req.params.id });
});

app.listen(3000);
نکته: فصل ۶ (مدیریت خطا با میدل‌ورهای چهارآرگومانی (err, req, res, next)) در دست نوشتن است؛ همین‌جا می‌بینید خطاها چطور از next(err) به هندلر خطا منتقل می‌شوند. سری حداقلی اما کاملاً کاربردی است — و چون از ساختار واقعی express تقلید می‌کند، بعداً خواندن کد اصلی express برایتان آشنا خواهد بود.
