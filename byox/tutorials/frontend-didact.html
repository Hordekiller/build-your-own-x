Didact: راهنمای DIY برای ساخت React خودت

⚙ زبان: JavaScript ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمهٔ فارسی و مرور مجموعه‌مقاله‌های «Didact — Build your own React» از رودریگو پومبو (Rodrigo Pombo) است که در وبلاگش منتشر شده و در مخزن hexacta/didact جمع‌آوری شده. این سری، React را از پایه — تا سطح Fiber — بازسازی می‌کند.
↗ مشاهده منبع اصلی

۱. مقدمه: چرا Didact؟

هدف رودریگو پومبو این است که نشان دهد React هیچ «جادویی» ندارد: با چند قطعهٔ سادهٔ جاوااسکریپت می‌توان نسخه‌ای کار از آن ساخت. Didact در شش مرحله ساخته می‌شود و در نهایت شامل: createElement، رندر، رندر همزمان (Concurrent Rendering) با Fiber، مقایسه و به‌روزرسانی (Reconciliation)، کامپوننت‌های تابعی و Hook ها است.

نکتهٔ جالب طراحی: هر مرحله یک پوشهٔ جداگانه در مخزن دارد و در هر مرحله کد قبلی بازنویسی می‌شود تا مفاهیم هم‌زمان با پیاده‌سازی روشن شوند.

۲. گام ۱ و ۲: createElement و رندر

اولین قدم مثل همیشه تولید گره‌های درخت مجازی است. به‌جای ساخت مستقیم DOM، هر عنصر به‌صورت یک شیء ساده بازنمایی می‌شود:

jsfunction createElement(type, props, ...children) {
  return {
    type,
    props: {
      ...props,
      children: children.map((child) =>
        // متن‌ها هم نود هستند: نوعشان "TEXT_ELEMENT"
        typeof child === "object" ? child : createTextElement(child)
      ),
    },
  };
}

function createTextElement(text) {
  return { type: "TEXT_ELEMENT", props: { nodeValue: text, children: [] } };
}

سپس تابع render این درخت را به DOM واقعی تبدیل می‌کند؛ به‌صورت بازگشتی برای هر نود، عنصر DOM می‌سازد، props (مانند onClick و className) را روی آن می‌گذارد و بچه‌ها را الحاق می‌کند. همین دو تابع کافی است تا یک برنامهٔ سادهٔ «سلام دنیا» با JSX کار کند.

۳. گام ۳: رندر همزمان با Fiber

مشکل رندر بازگشتی: اگر درخت بزرگ باشد، کل فرایند رندر یک‌باره و بدون توقف اجرا می‌شود و ترد اصلی مسدود می‌شود (فریم‌ها از دست می‌روند). راه‌حل: کار رندر را به واحدهای کوچک کار تقسیم کنید و اجازه دهید در بین آن‌ها مرورگر فرصت اجرا پیدا کند. به هر واحد کار یک Fiber می‌گوییم. ساختار هر fiber:

js// هر fiber یک اشاره‌گر به بچه، خواهر و پدر خود دارد
{
  type,        // نوع نود ("div"، "TEXT_ELEMENT" یا تابع کامپوننت)
  props,
  parent,      // پدر در درخت
  child,       // اولین بچه
  sibling,     // بچهٔ بعدی (زیر هم بودن)
  dom,         // نود DOM ساخته‌شده
  alternate,   // fiber قبلی (برای مقایسه)
  effectTag,   // "UPDATE" | "PLACEMENT" | "DELETION"
}

// تابع کار واحد: درخت را به‌صورت مبتنی بر حلقه طی می‌کند
function performUnitOfWork(fiber) {
  // ساخت نود DOM اگر نبود
  if (!fiber.dom) fiber.dom = createDom(fiber);
  // ساختن fiber بچه‌ها و ساختن روابط
  const elements = fiber.props.children;
  let index = 0;
  let prevSibling = null;
  while (index < elements.length) {
    const element = elements[index];
    const newFiber = { type: element.type, props: element.props, parent: fiber };
    if (index === 0) fiber.child = newFiber;
    else prevSibling.sibling = newFiber;
    prevSibling = newFiber;
    index++;
  }
  // انتخاب واحد کار بعدی: بچه → خواهر → صعود به پدر
  if (fiber.child) return fiber.child;
  let nextFiber = fiber;
  while (nextFiber) {
    if (nextFiber.sibling) return nextFiber.sibling;
    nextFiber = nextFiber.parent;
  }
}

این کار با requestIdleCallback زمان‌بندی می‌شود تا مرورگر هر زمان که بیکار بود قطعهٔ بعدی کار را انجام دهد.

۴. گام ۴: مقایسه و به‌روزرسانی

برای اینکه رندر بعدی فقط «تفاوت‌ها» را اعمال کند، درخت fiber قبلی نگه داشته می‌شود (alternate) و هنگام ساخت فایبرهای جدید، نودها مقایسه می‌شوند:

jsfunction reconcileChildren(wipFiber, elements) {
  let index = 0;
  let oldFiber = wipFiber.alternate && wipFiber.alternate.child;
  let prevSibling = null;

  while (index < elements.length || oldFiber != null) {
    const element = elements[index];
    let newFiber = null;

    const sameType = oldFiber && element && element.type === oldFiber.type;

    if (sameType) {
      // نوع یکسان: فقط props را به‌روز کن
      newFiber = { type: oldFiber.type, props: element.props, dom: oldFiber.dom, parent: wipFiber, alternate: oldFiber, effectTag: "UPDATE" };
    } else if (element) {
      // عنصر جدید
      newFiber = { type: element.type, props: element.props, dom: null, parent: wipFiber, alternate: null, effectTag: "PLACEMENT" };
    } else if (oldFiber) {
      // نود حذف‌شده: علامت‌گذاری برای پاک‌سازی
      oldFiber.effectTag = "DELETION";
      deletions.push(oldFiber);
    }
    /* ... حرکت روی بچه‌ها و خواهرها ... */
  }
}

در انتها مرحلهٔ commit انجام می‌شود: یک بار درخت را طی می‌کنیم و بر اساس effectTag نودها را به DOM اضافه، به‌روزرسانی یا حذف می‌کنیم.

۵. گام ۵ و ۶: کامپوننت‌های تابعی و Hook ها

کامپوننت تابعی یعنی type خودش یک تابع است؛ باید آن را صدا بزنیم تا عنصرهای بچه را بگیریم. برای Hook ها هم یک آرایهٔ جهانی به‌همراه شمارندهٔ تعریف نگه داشته می‌شود؛ useState با استفاده از alternate مقدار قبلی state را پیدا می‌کند:

jslet wipFiber = null;
let hookIndex = null;

function useState(initial) {
  const oldHook =
    wipFiber.alternate && wipFiber.alternate.hooks &&
    wipFiber.alternate.hooks[hookIndex];

  const hook = {
    state: oldHook ? oldHook.state : initial,
    queue: [],
  };

  const actions = oldHook ? oldHook.queue : [];
  actions.forEach((action) => {
    hook.state = typeof action === "function" ? action(hook.state) : action;
  });

  const setState = (action) => {
    hook.queue.push(action);
    // رندر دوباره را از ریشه آغاز کن
    wipRoot = { dom: currentRoot.dom, props: currentRoot.props, alternate: currentRoot };
    nextUnitOfWork = wipRoot;
    deletions = [];
  };

  wipFiber.hooks.push(hook);
  hookIndex++;
  return [hook.state, setState];
}

با همین حدود ۴۰۰ خط کد، برنامه‌های واقعی React (با state، props و به‌روزرسانی کارآمد) اجرا می‌شوند!

۶. جمع‌بندی

  • درخت مجازی یک نمایش دادهٔ ساده است؛ پیچیدگی‌ها در زمان‌بندی و مقایسه است.
  • Fiber ساختاری حلقه‌پذیر است که رندر را قابل تکه‌تکه‌کردن می‌کند؛ پایهٔ رندر همزمان و Suspense در React واقعی.
  • Hook ها بدون هیچ جادویی، فقط با آرایه و شاخص پیاده می‌شوند — که علت قانون «همیشه در سطح بالا صدا زدن» را هم توضیح می‌دهد.
