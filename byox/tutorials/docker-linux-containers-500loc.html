مقدمه

نویسنده سال‌ها مستقیم و غیرمستقیم از کانتینرهای لینوکس استفاده کرده، اما می‌خواسته بیشتر با آن‌ها آشنا شود؛ پس دست به کدنویسی زده است. این کد زمانی ۵۰۰ خط بود (قول می‌دهد!)، اما از زمان انتشار چند بار بازبینی شده و حالا حدود ۷۰ خط بیشتر دارد.

هدف خاص او یافتن مجموعهٔ حداقلی از محدودیت‌ها برای اجرای کدِ غیرقابل‌اعتماد است. این روشی نیست که باید برای کانتینرهای در معرض خطر استفاده کنید — باید همه‌چیز قابل محدودشدن را محدود کنید — اما مهم است بدانیم کدام مجوزها به‌طور قطعی ناامن هستند!

این نوشته یک قطعه کد «ادبی» (literate code) به سبک noweb است؛ ارجاع‌های <<x>> با بلوک کدی به همین نام گسترش می‌یابند. کد نهایی (contained.c) و سند orgmode آن در سایت اصلی موجودند. این سند و کد تحت مجوز GPLv3 منتشر شده‌اند.

ساختار کانتینر

مکانیزم‌های مکمل و هم‌پوشان متعددی کانتینرهای مدرن لینوکس را می‌سازند. به‌طور کلی:

  • namespaces برای گروه‌بندی اشیای هسته در مجموعه‌های متفاوتی استفاده می‌شوند که درخت‌های فرایندی خاص می‌توانند ببینند. مثلاً namespace مربوط به pid، دیدِ فهرست فرایندها را به فرایندهای داخل همان namespace محدود می‌کند.
  • capabilities برای اعمال محدودیت‌های تقریبی روی کارهایی که uid صفر می‌تواند انجام دهد استفاده می‌شوند.
  • cgroups مکانیزمی برای محدود کردن مصرف منابعی مثل حافظه، I/O دیسک و زمان CPU است.
  • setrlimit مکانیزم دیگری برای محدودکردن مصرف منابع است؛ از cgroups قدیمی‌تر است اما بعضی کارها را می‌تواند بکند که cgroups نمی‌تواند.

همهٔ این‌ها مکانیزم‌های هستهٔ لینوکس هستند. Seccomp، capabilities و setrlimit با سیستمعمل‌ها انجام می‌شوند؛ cgroups از طریق یک فایل‌سیستم قابل دسترسی است.

توجه: user namespaces نسبتاً جدید هستند و نوید یکپارچه‌کردن بخش زیادی از این رفتار را می‌دهند؛ اما کامپایل هسته با پشتیبانی آن‌ها، معنای capabilities را در کل سیستم تغییر می‌دهد و تاکنون باگ‌های جدی افزایش امتیاز (privilege escalation) زیادی از آن‌ها بیرون آمده است. مقالهٔ Understanding and Hardening Linux Containers توضیح می‌دهد که به دلیل حساسیتِ user namespace، چندین آسیب‌پذیری جدی در آن کشف شده و همچنان کشف می‌شود. نویسنده فقط در صورت در دسترس بودن از user namespace استفاده می‌کند (و از ایجاد user namespace تو در تو جلوگیری می‌کند).

cint choose_hostname(char *buff, size_t len)
{
	static const char *suits[] = { "swords", "wands", "pentacles", "cups" };
	static const char *minor[] = {
		"ace", "two", "three", "four", "five", "six", "seven", "eight",
		"nine", "ten", "page", "knight", "queen", "king"
	};
	static const char *major[] = {
		"fool", "magician", "high-priestess", "empress", "emperor",
		"hierophant", "lovers", "chariot", "strength", "hermit",
		"wheel", "justice", "hanged-man", "death", "temperance",
		"devil", "tower", "star", "moon", "sun", "judgment", "world"
	};
	struct timespec now = {0};
	clock_gettime(CLOCK_MONOTONIC, &now);
	size_t ix = now.tv_nsec % 78;
	if (ix < sizeof(major) / sizeof(*major)) {
		snprintf(buff, len, "%05lx-%s", now.tv_sec, major[ix]);
	} else {
		ix -= sizeof(major) / sizeof(*major);
		snprintf(buff, len,
			 "%05lxc-%s-of-%s",
			 now.tv_sec,
			 minor[ix % (sizeof(minor) / sizeof(*minor))],
			 suits[ix / (sizeof(minor) / sizeof(*minor))]);
	}
	return 0;
}

Namespaces

clone سیستمعملی است که پشت fork() و امثال آن است؛ و کلید همهٔ این ماجرا هم هست. از نظر مفهومی می‌خواهیم فرایندی با ویژگی‌های متفاوت از والدش بسازیم: باید بتواند / متفاوتی را mount کند، hostname خودش را تنظیم کند و کارهای دیگر انجام دهد. همهٔ این‌ها با پاس دادن پرچم‌ها به clone مشخص می‌شوند.

فرایند فرزند باید پیام‌هایی به والد بفرستد؛ پس یک socketpair مقداردهی می‌شود و فقط یک سر آن در اختیار فرزند قرار می‌گیرد:

c	if (socketpair(AF_LOCAL, SOCK_SEQPACKET, 0, sockets)) {
		fprintf(stderr, "socketpair failed: %m\n");
		goto error;
	}
	if (fcntl(sockets[0], F_SETFD, FD_CLOEXEC)) {
		fprintf(stderr, "fcntl failed: %m\n");
		goto error;
	}
	config.fd = sockets[1];

اول باید جا برای پشته آماده شود؛ بعداً execve پشته را دوباره می‌چیند، پس این فقط موقتی است:

c	#define STACK_SIZE (1024 * 1024)

	char *stack = 0;
	if (!(stack = malloc(STACK_SIZE))) {
		fprintf(stderr, "=> malloc failed, out of memory?\n");
		goto error;
	}

همچنین cgroup برای این درخت فرایند آماده می‌شود (بعداً بیشتر). سپس mount ها، pidها، ساختارهای دادهٔ IPC، دستگاه‌های شبکه و hostname/نام دامنه namespace می‌شوند:

فرایند والد، user namespace فرزند را پیکربندی می‌کند و سپس تا خروجِ درخت فرایند فرزند مکث می‌کند. اگر فرزند user namespace بسازد، والد نگاشت‌های uid و gid را در /proc/<pid>/uid_map و gid_map می‌نویسد:

c#define USERNS_OFFSET 10000
#define USERNS_COUNT 2000

int handle_child_uid_map (pid_t child_pid, int fd)
{
	int uid_map = 0;
	int has_userns = -1;
	if (read(fd, &has_userns, sizeof(has_userns)) != sizeof(has_userns)) {
		fprintf(stderr, "couldn't read from child!\n");
		return -1;
	}
	if (has_userns) {
		char path[PATH_MAX] = {0};
		for (char **file = (char *[]) { "uid_map", "gid_map", 0 }; *file; file++) {
			if (snprintf(path, sizeof(path), "/proc/%d/%s", child_pid, *file)
			    > sizeof(path)) {
				fprintf(stderr, "snprintf too big? %m\n");
				return -1;
			}
			fprintf(stderr, "writing %s...", path);
			if ((uid_map = open(path, O_WRONLY)) == -1) {
				fprintf(stderr, "open failed: %m\n");
				return -1;
			}
			if (dprintf(uid_map, "0 %d %d\n", USERNS_OFFSET, USERNS_COUNT) == -1) {
				fprintf(stderr, "dprintf failed: %m\n");
				close(uid_map);
				return -1;
			}
			close(uid_map);
		}
	}
	if (write(fd, & (int) { 0 }, sizeof(int)) != sizeof(int)) {
		fprintf(stderr, "couldn't write: %m\n");
		return -1;
	}
	return 0;
}

فرایند فرزند به والد اطلاع می‌دهد که آیا باید نگاشت‌های uid و gid تنظیم شوند؛ در صورت موفقیت setgroups، setresgid و setresuid را انجام می‌دهد (هر سه لازم‌اند، چون دو مکانیزم گروهی جدا در لینوکس وجود دارد):

cint userns(struct child_config *config)
{
	fprintf(stderr, "=> trying a user namespace...");
	int has_userns = !unshare(CLONE_NEWUSER);
	if (write(config->fd, &has_userns, sizeof(has_userns)) != sizeof(has_userns)) {
		fprintf(stderr, "couldn't write: %m\n");
		return -1;
	}
	int result = 0;
	if (read(config->fd, &result, sizeof(result)) != sizeof(result)) {
		fprintf(stderr, "couldn't read: %m\n");
		return -1;
	}
	if (result) return -1;
	if (has_userns) {
		fprintf(stderr, "done.\n");
	} else {
		fprintf(stderr, "unsupported? continuing.\n");
	}
	fprintf(stderr, "=> switching to uid %d / gid %d...", config->uid, config->uid);
	if (setgroups(1, & (gid_t) { config->uid }) ||
	    setresgid(config->uid, config->uid, config->uid) ||
	    setresuid(config->uid, config->uid, config->uid)) {
		fprintf(stderr, "%m\n");
		return -1;
	}
	fprintf(stderr, "done.\n");
	return 0;
}

و اینجاست که فرایند فرزندِ حاصل از clone به اینجا می‌رسد. همهٔ مراحل setup انجام می‌شود و سپس executable بارگذاری می‌شود. ترتیب مهم است: نمی‌توانیم بدون capabilities خاصی mount ها را تغییر دهیم، نمی‌توانیم بعد از محدودکردن سیستمعمل‌ها unshare کنیم و غیره:

cint child(void *arg)
{
	struct child_config *config = arg;
	if (sethostname(config->hostname, strlen(config->hostname))
	    || mounts(config)
	    || userns(config)
	    || capabilities()
	    || syscalls()) {
		close(config->fd);
		return -1;
	}
	if (close(config->fd)) {
		fprintf(stderr, "close failed: %m\n");
		return -1;
	}
	if (execve(config->argv[0], config->argv, NULL)) {
		fprintf(stderr, "execve failed! %m.\n");
		return -1;
	}
	return 0;
}

Capabilities

capabilities ویژگیِ «root بودن» را در لینوکس ریزدانه می‌کنند. مفید است که امتیازات را محفظه‌بندی کنیم؛ مثلاً یک فرایند بتواند دستگاه‌های شبکه تخصیص دهد (CAP_NET_ADMIN) اما نتواند همهٔ فایل‌ها را بخواند (CAP_DAC_OVERRIDE). اینجا برای حذف موارد ناخواسته استفاده می‌شوند.

اما همهٔ «root بودن» به capabilities تقسیم نمی‌شود؛ مثلاً نوشتن در بخش‌هایی از procfs حتی بعد از حذف capabilities برای root مجاز است. این بخشی از دلیل نیاز به محدودیت‌های دیگر در کنار capabilities است.

نحوهٔ حذف capabilities هم مهم است. man 7 capabilities الگوریتمی دارد: در execve(2) هسته مجموعه‌های capability جدید را محاسبه می‌کند. ما می‌خواهیم P'(ambient) و P(inheritable) خالی باشند و P'(permitted) و P(effective) فقط شامل موارد بالا باشند. این با دو کار به‌دست می‌آید:

  • پاک‌کردن مجموعهٔ inheritable خودمان (این مجموعهٔ ambient را هم پاک می‌کند، چون طبق man هیچ capability نمی‌تواند ambient باشد مگر اینکه هم permitted باشد هم inheritable).
  • پاک‌کردن bounding set. این کار capabilities فایل‌هایی را که هنگام execve به‌دست می‌آوریم محدود می‌کند.
نکته: اگر فقط مجموعه‌های effective، permitted و inheritable خودمان را حذف کنیم، در فرزندِ execve شده امتیازاتِ file-capability ها دوباره به‌دست می‌آیند؛ این همان‌طور است که bash می‌تواند ping را صدا بزند.

Capabilities حذف‌شده

cint capabilities()
{
	fprintf(stderr, "=> dropping capabilities...");

	int drop_caps[] = {
		CAP_AUDIT_CONTROL,
		CAP_AUDIT_READ,
		CAP_AUDIT_WRITE,
		CAP_BLOCK_SUSPEND,
		CAP_DAC_READ_SEARCH,
		CAP_FSETID,
		CAP_IPC_LOCK,
		CAP_MAC_ADMIN,
		CAP_MAC_OVERRIDE,
		CAP_MKNOD,
		CAP_SETFCAP,
		CAP_SYSLOG,
		CAP_SYS_ADMIN,
		CAP_SYS_BOOT,
		CAP_SYS_MODULE,
		CAP_SYS_NICE,
		CAP_SYS_RAWIO,
		CAP_SYS_RESOURCE,
		CAP_SYS_TIME,
		CAP_WAKE_ALARM
	};

	size_t num_caps = sizeof(drop_caps) / sizeof(*drop_caps);
	fprintf(stderr, "bounding...");
	for (size_t i = 0; i < num_caps; i++) {
		if (prctl(PR_CAPBSET_DROP, drop_caps[i], 0, 0, 0)) {
			fprintf(stderr, "prctl failed: %m\n");
			return 1;
		}
	}
	fprintf(stderr, "inheritable...");
	cap_t caps = NULL;
	if (!(caps = cap_get_proc())
	    || cap_set_flag(caps, CAP_INHERITABLE, num_caps, drop_caps, CAP_CLEAR)
	    || cap_set_proc(caps)) {
		fprintf(stderr, "failed: %m\n");
		if (caps) cap_free(caps);
		return 1;
	}
	cap_free(caps);
	fprintf(stderr, "done.\n");
	return 0;
}

دلایل حذف هر یک (خلاصه):

  • CAP_AUDIT_CONTROL/_READ/_WRITE: دسترسی به سیستم audit هسته. CAP_AUDIT_READ به‌ویژه namespace نمی‌شود و ممکن است اطلاعات مهمی داشته باشد.
  • CAP_BLOCK_SUSPEND: جلوگیری از suspend شدن سیستم؛ suspend namespace نمی‌شود.
  • CAP_DAC_READ_SEARCH: اجازهٔ خواندن فایل‌های دلخواه از طریق open_by_handle_at (در سال ۲۰۱۴ برای فرار از Docker استفاده شد!).
  • CAP_FSETID: بدون user namespace می‌تواند فایل setuid را بدون حذف بیت setuid تغییر دهد — خطرناک!
  • CAP_IPC_LOCK: قفل‌کردن بیشتر حافظهٔ فرایند از حد مجاز (DoS).
  • CAP_MAC_ADMIN/OVERRIDE: سیستم‌های MAC مثل AppArmor و SELinux؛ namespace نمی‌شوند.
  • CAP_MKNOD: ساخت فایل‌های دستگاه برای سخت‌افزار واقعی (مثلاً بازسازی دیسک سخت و خواندن/نوشتن آن!).
  • CAP_SETFCAP: افزودن capability به یک executable و اجرای آن؛ امنیت سیستم را تضعیف می‌کند.
  • CAP_SYSLOG: افشای آدرس‌های هسته و دور زدن KASLR.
  • CAP_SYS_ADMIN: رفتارهای بسیار؛ بیشتر آن‌ها را نمی‌خواهیم.
  • CAP_SYS_BOOT: راه‌اندازی مجدد سیستم و بارگذاری هسته‌های جدید! قطعاً نمی‌خواهیم.
  • CAP_SYS_MODULE: بارگذاری/حذف ماژول‌های هسته.
  • CAP_SYS_NICE: اولویت بالاتر از حد مجاز — DoS بقیهٔ سیستم.
  • CAP_SYS_RAWIO: دسترسی کامل به حافظهٔ میزبان و پورت‌های I/O.
  • CAP_SYS_RESOURCE: دور زدن محدودیت‌های سراسری هسته.
  • CAP_SYS_TIME: تنظیم زمان؛ namespace نمی‌شود.
  • CAP_WAKE_ALARM: دخالت در suspend.

Capabilities حفظ‌شده

پیگیری مواردی که حذف نمی‌شوند هم مهم است:

  • CAP_DAC_OVERRIDE: بر خلاف CAP_DAC_READ_SEARCH، به‌تنهایی اجازهٔ خواندن خارج از mount namespace را نمی‌دهد (DAC یعنی کنترل دسترسی اختیاریِ مجوزهای عادی یونیکس).
  • CAP_FOWNER، CAP_LEASE، CAP_LINUX_IMMUTABLE: همه روی فایل‌های داخل mount namespace عمل می‌کنند.
  • CAP_SYS_PACCT: فقط روی فرایندِ فراخواننده عمل می‌کند؛ ما از accounting استفاده نمی‌کنیم.
  • CAP_IPC_OWNER: فقط توسط توابعی استفاده می‌شود که به IPC namespace احترام می‌گذارند.
  • CAP_NET_ADMIN/BIND_SERVICE/RAW: چون شبکه با بریج مجازی و network namespace جدا می‌شود، مشکلی نیست.
  • CAP_SYS_PTRACE و CAP_KILL: ptrace و سیگنال‌ها نمی‌توانند از pid namespace عبور کنند.
  • CAP_SETUID/SETGID: رفتارهای آن‌ها محدود به namespace است.
  • CAP_SETPCAP: فقط اجازهٔ افزودن capabilitiesای را می‌دهد که فرایند از قبل مؤثرانه دارد.
  • CAP_SYS_CHROOT و CAP_SYS_TTYCONFIG: در چیدمان ما آسیب‌ناپذیرند.

Mount ها

فرایند فرزند در mount namespace خودش است، پس می‌توانیم چیزهایی را که نباید ببیند unmount کنیم. روش:

  1. یک دایرکتوری موقت و یکی داخلش بساز.
  2. آرگومان کاربر را به دایرکتوری موقت bind mount کن.
  3. pivot_root بزن: bind mount را ریشه کن و ریشهٔ قدیمی را روی دایرکتوری موقت داخلی mount کن.
  4. ریشهٔ قدیمی را unmount کن و دایرکتوری موقت داخلی را حذف کن.

اول همه‌چیز با MS_PRIVATE دوباره mount می‌شود؛ بیشتر یک راحتی است تا bind mount خارج از namespace ما نامرئی باشد:

cint mounts(struct child_config *config)
{
	fprintf(stderr, "=> remounting everything with MS_PRIVATE...");
	if (mount(NULL, "/", NULL, MS_REC | MS_PRIVATE, NULL)) {
		fprintf(stderr, "failed! %m\n");
		return -1;
	}
	fprintf(stderr, "remounted.\n");

	fprintf(stderr, "=> making a temp directory and a bind mount there...");
	char mount_dir[] = "/tmp/tmp.XXXXXX";
	if (!mkdtemp(mount_dir)) {
		fprintf(stderr, "failed making a directory!\n");
		return -1;
	}

	if (mount(config->mount_dir, mount_dir, NULL, MS_BIND | MS_PRIVATE, NULL)) {
		fprintf(stderr, "bind mount failed!\n");
		return -1;
	}

	char inner_mount_dir[] = "/tmp/tmp.XXXXXX/oldroot.XXXXXX";
	memcpy(inner_mount_dir, mount_dir, sizeof(mount_dir) - 1);
	if (!mkdtemp(inner_mount_dir)) {
		fprintf(stderr, "failed making the inner directory!\n");
		return -1;
	}
	fprintf(stderr, "done.\n");

	fprintf(stderr, "=> pivoting root...");
	if (pivot_root(mount_dir, inner_mount_dir)) {
		fprintf(stderr, "failed!\n");
		return -1;
	}
	fprintf(stderr, "done.\n");

	char *old_root_dir = basename(inner_mount_dir);
	char old_root[sizeof(inner_mount_dir) + 1] = { "/" };
	strcpy(&old_root[1], old_root_dir);

	fprintf(stderr, "=> unmounting %s...", old_root);
	if (chdir("/")) {
		fprintf(stderr, "chdir failed! %m\n");
		return -1;
	}
	if (umount2(old_root, MNT_DETACH)) {
		fprintf(stderr, "umount failed! %m\n");
		return -1;
	}
	if (rmdir(old_root)) {
		fprintf(stderr, "rmdir failed! %m\n");
		return -1;
	}
	fprintf(stderr, "done.\n");
	return 0;
}

pivot_root سیستمعملی است که mount در / را با mount دیگری جابه‌جا می‌کند. Glibc برایش wrapper ندارد؛ پس خودمان آن را تعریف می‌کنیم:

cint pivot_root(const char *new_root, const char *put_old)
{
	return syscall(SYS_pivot_root, new_root, put_old);
}

نویسنده از بسته‌بندی/بازکردن کانتینرها عمداً اجتناب می‌کند؛ این‌جا محل خوبی برای آسیب‌پذیری است. او به کاربر تکیه می‌کند که دایرکتوری mount شده حاوی فایل‌ها یا hard link های حساس نباشد.

سیستمعمل‌ها

سیستمعمل‌هایی که ثابت شده باعث ضرر یا فرار از sandbox می‌شوند، بلاک‌لیست می‌شوند. باز هم: این بهترین روش نیست، اما گویاترین است. مستندات seccomp داکر و پروفایل seccomp پیش‌فرض آن منابع معقولی برای سیستمعمل‌های خطرناک هستند.

c#define SCMP_FAIL SCMP_ACT_ERRNO(EPERM)

int syscalls()
{
	scmp_filter_ctx ctx = NULL;
	fprintf(stderr, "=> filtering syscalls...");
	if (!(ctx = seccomp_init(SCMP_ACT_ALLOW))

	/* جلوگیری از ساخت executable های setuid/setgid جدید: بدون user namespace
	   فرایندِ محصور می‌تواند یک باینری setuid بسازد که هر کاربری برای
	   روت شدن استفاده کند! */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(chmod), 1,
				SCMP_A1(SCMP_CMP_MASKED_EQ, S_ISUID, S_ISUID))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(chmod), 1,
				SCMP_A1(SCMP_CMP_MASKED_EQ, S_ISGID, S_ISGID))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(fchmod), 1,
				SCMP_A1(SCMP_CMP_MASKED_EQ, S_ISUID, S_ISUID))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(fchmod), 1,
				SCMP_A1(SCMP_CMP_MASKED_EQ, S_ISGID, S_ISGID))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(fchmodat), 1,
				SCMP_A2(SCMP_CMP_MASKED_EQ, S_ISUID, S_ISUID))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(fchmodat), 1,
				SCMP_A2(SCMP_CMP_MASKED_EQ, S_ISGID, S_ISGID))

	/* شروع user namespace های جدید به فرایندهای محصور امکان به‌دست‌آوردن
	   capabilities جدید (هرچند محدود) می‌دهد؛ جلوگیری می‌کنیم. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(unshare), 1,
				SCMP_A0(SCMP_CMP_MASKED_EQ, CLONE_NEWUSER, CLONE_NEWUSER))
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(clone), 1,
				SCMP_A0(SCMP_CMP_MASKED_EQ, CLONE_NEWUSER, CLONE_NEWUSER))

	/* TIOCSTI به فرایندهای محصور اجازه می‌دهد در ترمینال کنترل‌کننده بنویسند. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(ioctl), 1,
				SCMP_A1(SCMP_CMP_MASKED_EQ, TIOCSTI, TIOCSTI))

	/* سیستم keyring هسته namespace نمی‌شود. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(keyctl), 0)
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(add_key), 0)
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(request_key), 0)

	/* قبل از لینوکس 4.8، ptrace کاملاً seccomp را می‌شکست. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(ptrace), 0)

	/* تخصیص گره‌های NUMA: احتمالاً برای DoS کاربرد دارد. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(mbind), 0)
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(migrate_pages), 0)
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(move_pages), 0)
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(set_mempolicy), 0)

	/* userfaultfd می‌تواند اجرای هسته را با fault صفحه متوقف کند؛
	   بخش مهمی از برخی اکسپلویت‌های هسته. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(userfaultfd), 0)

	/* perf_event_open در صورت تنظیم نبودن paranoid می‌تواند آدرس‌های هسته
	   و حافظهٔ مقداردهی‌نشده را فاش کند. */
	    || seccomp_rule_add(ctx, SCMP_FAIL, SCMP_SYS(perf_event_open), 0)

	/* PR_SET_NO_NEW_PRIVS را روشن نمی‌کنیم؛ نامش کمی مبهم است: به‌طور خاص
	   جلوی اجرای باینری‌های setuid/setcap با امتیازات اضافه را می‌گیرد. */
	    || seccomp_attr_set(ctx, SCMP_FLTATR_CTL_NNP, 0)

	    || seccomp_load(ctx)) {
		if (ctx) seccomp_release(ctx);
		fprintf(stderr, "failed: %m\n");
		return 1;
	}
	seccomp_release(ctx);
	fprintf(stderr, "done.\n");
	return 0;
}

سیستمعمل‌هایی که توسط پالیسی پیش‌فرض داکر غیرفعال‌اند اما این کد اجازه می‌دهد (خلاصه): _sysctl، alloc_hugepages، bdflush و چند مورد دیگر که در لینوکس مدرن وجود ندارند؛ موارد وابسته به CAP_SYS_TIME/CAP_SYS_ADMIN/CAP_SYS_BOOT که آن‌ها را رها کرده‌ایم؛ توابع قدیمی‌ای مثل oldstat که معادل مدرن با همان ویژگی‌های امنیتی دارند؛ موارد خاص IA-64/PowerPC/Sparc که فقط روی آن معماری‌ها موجودند؛ و preadv2/pwritev2 که فقط امتداد scatter/gather هستند.

منابع (Resources)

می‌خواهیم از فرایندهای فرزند بدرفتار جلوگیری کنیم که به بقیهٔ سیستم سرویس‌دهی نکنند (DoS). cgroups به‌طور خاص برای محدودکردن حافظه و زمان CPU مفید است؛ محدودکردن تعداد pid و استفادهٔ I/O هم مفید است.

فایل‌سیستم‌های cgroup و cgroup2 رابط‌های متعارف سیستم cgroup هستند. cgroup2 کمی متفاوت است و روی سیستم نویسنده فعال نشده؛ پس از نسخهٔ اول استفاده شده است.

cgroup namespace ها کمی متفاوت از mount namespace ها هستند: باید cgroup را قبل از ورود به cgroup namespace بسازیم؛ بعد از آن، آن cgroup داخل namespace مثل cgroup ریشه رفتار می‌کند.

تنظیمات اعمال‌شده (به ازای hostname):

  • memory/$hostname/memory.limit_in_bytes — حداکثر ۱ گیگابایت حافظه در فضای کاربر.
  • memory/$hostname/memory.kmem.limit_in_bytes — حداکثر ۱ گیگابایت حافظهٔ هسته.
  • cpu/$hostname/cpu.shares = 256 — سهم‌های CPU از ۱۰۲۴ هستند؛ این یعنی حداکثر یک‌چهارم زمان CPU در سیستم شلوغ.
  • pids/$hostname/pid.max = 64 — حداکثر ۶۴ pid برای فرایند و فرزندانش.
  • blkio/$hostname/weight — وزنی پایین‌تر از بقیهٔ سیستم.
c#define MEMORY "1073741824"
#define SHARES "256"
#define PIDS "64"
#define WEIGHT "10"
#define FD_COUNT 64

struct cgrp_control {
	char control[256];
	struct cgrp_setting {
		char name[256];
		char value[256];
	} **settings;
};
struct cgrp_setting add_to_tasks = {
	.name = "tasks",
	.value = "0"
};

struct cgrp_control *cgrps[] = {
	& (struct cgrp_control) {
		.control = "memory",
		.settings = (struct cgrp_setting *[]) {
			& (struct cgrp_setting) {
				.name = "memory.limit_in_bytes",
				.value = MEMORY
			},
			& (struct cgrp_setting) {
				.name = "memory.kmem.limit_in_bytes",
				.value = MEMORY
			},
			&add_to_tasks,
			NULL
		}
	},
	& (struct cgrp_control) {
		.control = "cpu",
		.settings = (struct cgrp_setting *[]) {
			& (struct cgrp_setting) {
				.name = "cpu.shares",
				.value = SHARES
			},
			&add_to_tasks,
			NULL
		}
	},
	& (struct cgrp_control) {
		.control = "pids",
		.settings = (struct cgrp_setting *[]) {
			& (struct cgrp_setting) {
				.name = "pids.max",
				.value = PIDS
			},
			&add_to_tasks,
			NULL
		}
	},
	& (struct cgrp_control) {
		.control = "blkio",
		.settings = (struct cgrp_setting *[]) {
			& (struct cgrp_setting) {
				.name = "blkio.weight",
				.value = PIDS
			},
			&add_to_tasks,
			NULL
		}
	},
	NULL
};

نوشتن در فایل‌سیستم cgroup نسخهٔ ۱ این‌طور کار می‌کند: در هر کنترلر با mkdir یک cgroup با نام دلخواه می‌سازیم (مثلاً /sys/fs/cgroup/memory/$hostname)، داخل آن فایل‌های تکی را با echo مقداردهی می‌کنیم، و یک pid به tasks اضافه می‌کنیم تا درخت فرایند وارد cgroup شود («0» مقدار ویژه‌ای است یعنی فرایندِ در حال نوشتن).

cint resources(struct child_config *config)
{
	fprintf(stderr, "=> setting cgroups...");
	for (struct cgrp_control **cgrp = cgrps; *cgrp; cgrp++) {
		char dir[PATH_MAX] = {0};
		fprintf(stderr, "%s...", (*cgrp)->control);
		if (snprintf(dir, sizeof(dir), "/sys/fs/cgroup/%s/%s",
			     (*cgrp)->control, config->hostname) == -1) {
			return -1;
		}
		if (mkdir(dir, S_IRUSR | S_IWUSR | S_IXUSR)) {
			fprintf(stderr, "mkdir %s failed: %m\n", dir);
			return -1;
		}
		for (struct cgrp_setting **setting = (*cgrp)->settings; *setting; setting++) {
			char path[PATH_MAX] = {0};
			int fd = 0;
			if (snprintf(path, sizeof(path), "%s/%s", dir,
				     (*setting)->name) == -1) {
				fprintf(stderr, "snprintf failed: %m\n");
				return -1;
			}
			if ((fd = open(path, O_WRONLY)) == -1) {
				fprintf(stderr, "opening %s failed: %m\n", path);
				return -1;
			}
			if (write(fd, (*setting)->value, strlen((*setting)->value)) == -1) {
				fprintf(stderr, "writing to %s failed: %m\n", path);
				close(fd);
				return -1;
			}
			close(fd);
		}
	}
	fprintf(stderr, "done.\n");

	/* سقف سخت تعداد فایل‌دیسکریپتورها را هم پایین می‌آوریم؛ چون این عدد،
	   مثل تعداد pidها، به‌ازای هر کاربر است و نمی‌خواهیم فرایند درون
	   کانتینر همهٔ آن‌ها را اشغال کند. */
	fprintf(stderr, "=> setting rlimit...");
	if (setrlimit(RLIMIT_NOFILE,
		      & (struct rlimit) {
			.rlim_max = FD_COUNT,
			.rlim_cur = FD_COUNT,
		})) {
		fprintf(stderr, "failed: %m\n");
		return 1;
	}
	fprintf(stderr, "done.\n");
	return 0;
}

تمیزکاری cgroup هم ساده است: اول فرایند contained را به tasks ریشه برمی‌گردانیم؛ بعد که فرایند فرزند تمام شده و خروج از pid namespace فرزندانش را SIGKILL می‌کند، tasks خالی است و می‌توان با خیال راحت rmdir کرد:

cint free_resources(struct child_config *config)
{
	fprintf(stderr, "=> cleaning cgroups...");
	for (struct cgrp_control **cgrp = cgrps; *cgrp; cgrp++) {
		char dir[PATH_MAX] = {0};
		char task[PATH_MAX] = {0};
		int task_fd = 0;
		if (snprintf(dir, sizeof(dir), "/sys/fs/cgroup/%s/%s",
			     (*cgrp)->control, config->hostname) == -1
		    || snprintf(task, sizeof(task), "/sys/fs/cgroup/%s/tasks",
				(*cgrp)->control) == -1) {
			fprintf(stderr, "snprintf failed: %m\n");
			return -1;
		}
		if ((task_fd = open(task, O_WRONLY)) == -1) {
			fprintf(stderr, "opening %s failed: %m\n", task);
			return -1;
		}
		if (write(task_fd, "0", 2) == -1) {
			fprintf(stderr, "writing to %s failed: %m\n", task);
			close(task_fd);
			return -1;
		}
		close(task_fd);
		if (rmdir(dir)) {
			fprintf(stderr, "rmdir %s failed: %m", dir);
			return -1;
		}
	}
	fprintf(stderr, "done.\n");
	return 0;
}

شبکه (Networking)

شبکهٔ کانتینر توضیح بیشتری می‌خواهد که در این فضا نمی‌گنجد. معمولاً این‌طور کار می‌کند:

  1. یک دستگاه bridge ساخته می‌شود.
  2. یک جفت ethernet مجازی ساخته می‌شود و یک سرش به bridge وصل می‌شود.
  3. سر دیگر داخل network namespace قرار می‌گیرد.
  4. برای دسترسی به شبکهٔ بیرون، میزبان باید ارسال (و احتمالاً NAT) بسته‌ها را فعال کند.

داشتن چند فرایند محصور که یک bridge مشترک دارند یعنی از دید میزبان هر دو در یک LAN هستند؛ پس جعل ARP یک مشکل همیشگی با کانتینرهایی است که این‌طور کار می‌کنند. روش متعارف انجام این کار از C، رابط rtnetlink است؛ احتمالاً استفاده از ip link ... آسان‌تر است. مصرف شبکه را هم می‌توان با کنترلر cgroup به نام net_prio محدود کرد.
