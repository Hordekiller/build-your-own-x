۱. دربارهٔ آموزش

توماس پالف (خالق وبلاگlessmilk.com و مجموعهٔ «یک بازی یک هفته») یک نسخهٔ سادهٔ Flappy Bird با Phaser 2.4 نوشته است — فقط ۶۵ خط کد — و قدم‌به‌قدم توضیحش می‌دهد: گرانش، پرش با کلیک، لوله‌های متحرک و برخورد. نسخهٔ به‌روزشده برای Phaser 3 هم در lessmilk.com/phaser-game-tutorial وجود دارد.

نکته: این بهترین آموزش اول برای Phaser است: کل ساختار یک بازی (سه صحنه: Boot/Play/GameOver) را در کمترین کد ممکن می‌بینید.

۳. کد بازی (کامل)

js// کل بازی در ۶۵ خط!
var game = new Phaser.Game(400, 490, Phaser.AUTO, 'gameDiv');

// صحنهٔ اصلی: خالی است چون همه‌چیز در حالت Play ساخته می‌شود
var mainState = {
  preload: function() {
    game.load.image('bird', 'assets/bird.png');
    game.load.image('pipe', 'assets/pipe.png');
  },

  create: function() {
    // گرانش: پرنده هر لحظه به پایین می‌افتد
    game.physics.startSystem(Phaser.Physics.ARCADE);

    this.bird = game.add.sprite(100, 245, 'bird');
    game.physics.arcade.enable(this.bird);
    this.bird.body.gravity.y = 1000;

    // پرش با کلیک/لمس
    var spaceKey = game.input.keyboard.addKey(Phaser.Keyboard.SPACEBAR);
    spaceKey.onDown.add(this.jump, this);

    // گروه لوله‌ها + تایمر ساخت
    this.pipes = game.add.group();
    this.timer = game.time.events.loop(1500, this.addRowOfPipes, this);

    this.score = 0;
    this.labelScore = game.add.text(20, 20, "0",
      { font: "30px Arial", fill: "#ffffff" });
  },

  update: function() {
    // برخورد پرنده با لوله‌ها → پایان بازی
    game.physics.arcade.overlap(this.bird, this.pipes, this.restartGame, null, this);

    if (this.bird.y < 0 || this.bird.y > 490) {
      this.restartGame();
    }
  },

  jump: function() {
    this.bird.body.velocity.y = -350; // ضربهٔ رو به بالا
  },

  addRowOfPipes: function() {
    // یک لولهٔ تصادفی (فاصلهٔ بین ۱ تا ۵)
    var hole = Math.floor(Math.random() * 5) + 1;

    for (var i = 0; i < 10; i++) {
      if (i != hole && i != hole + 1) {
        var pipe = game.add.sprite(400, i * 49, 'pipe');
        game.physics.arcade.enable(pipe);
        pipe.body.velocity.x = -200; // حرکت به چپ
        pipe.checkWorldBounds = true;
        pipe.outOfBoundsKill = true;
        this.pipes.add(pipe);
      }
    }
    this.score += 1;
    this.labelScore.text = this.score;
  },

  restartGame: function() {
    game.state.start('main'); // شروع دوباره
  },
};

game.state.add('main', mainState);
game.state.start('main');

همین! یک بازی کامل Flappy Bird که فقط index.html را باز کنید اجرا می‌شود.
