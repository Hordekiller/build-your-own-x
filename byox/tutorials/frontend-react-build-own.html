React خودت را بساز — بازسازی گام‌به‌گام با معماری فیبر (pomb.us)

⚙ زبان: JavaScript ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمهٔ فارسی و مرور مقالهٔ «Build your own React» از رودریگو پومبو (Rodrigo Pombo) در pomb.us است — یک راهنمای کامل و خودکفا که React را بر اساس معماری React 16.8 (Fiber + Hooks) از صفر بازسازی می‌کند.
↗ مشاهده منبع اصلی

۱. نمای کلی: هشت مرحله

مقاله در هشت گام یک React کامل می‌سازد — همان معماری واقعی React 16.8: createElement ← render ← رندر همزمان (Concurrent Mode) ← Fiber ← مراحل Render و Commit ← Reconciliation ← کامپوننت‌های تابعی ← Hooks. در پایان، یک کد واحد دارید که همهٔ گام‌ها را در خودش دارد؛ خلاصهٔ تمام کد در انتهای مقاله است.

توجه: کد این مقاله با React 16.8 جفت می‌شود — از propTypes و کلاس‌ها خبری نیست؛ همهٔ چیز با تابع و hook.

۲. گام ۱ و ۲: createElement و render

درخت مجازی با createElement ساخته می‌شود و render آن را به DOM تبدیل می‌کند. ویژگی‌های خاص: همه‌چیز — حتی متن‌ها — نود هستند و render نودها را بازگشتی می‌سازد:

jsfunction createElement(type, props, ...children) {
  return {
    type,
    props: {
      ...props,
      children: children.map((child) =>
        typeof child === 'object' ? child : createTextElement(child)
      ),
    },
  };
}

function createTextElement(text) {
  return { type: 'TEXT_ELEMENT', props: { nodeValue: text, children: [] } };
}

function createDom(fiber) {
  const dom =
    fiber.type === 'TEXT_ELEMENT'
      ? document.createTextNode('')
      : document.createElement(fiber.type);

  updateDom(dom, {}, fiber.props);
  return dom;
}

function render(element, container) {
  const rootFiber = {
    dom: container,
    props: { children: [element] },
    alternate: currentRoot, // برای مقایسه در رندر بعدی
  };
  deletions = [];
  wipRoot = rootFiber;
  nextUnitOfWork = rootFiber;
}

۳. گام ۳ و ۴: رندر همزمان و Fiber

مشکل رندر بازگشتی: کل درخت یک‌جا و بدون توقف رندر می‌شود و ترد را می‌بندد. راه‌حل: شکستن کار به واحدهای کوچک و اجرای تکه‌تکه با requestIdleCallback. ساختار دادهٔ Fiber همین واحد کار را نگه می‌دارد:

jslet nextUnitOfWork = null;
let wipRoot = null;      // ریشهٔ درخت در حال ساخت
let currentRoot = null;  // درخت کامیت‌شدهٔ قبلی
let deletions = [];      // نودهای حذف‌شده

// حلقهٔ کار: هر بار مرورگر بیکار بود، یک واحد کار
function workLoop(deadline) {
  let shouldYield = false;
  while (nextUnitOfWork && !shouldYield) {
    nextUnitOfWork = performUnitOfWork(nextUnitOfWork);
    shouldYield = deadline.timeRemaining() < 1;
  }
  // وقتی همهٔ کارها تمام شد، کل درخت را کامیت کن
  if (!nextUnitOfWork && wipRoot) {
    commitRoot();
  }
  requestIdleCallback(workLoop);
}
requestIdleCallback(workLoop);

// هر fiber: { type, props, dom, parent, child, sibling, alternate, effectTag }
function performUnitOfWork(fiber) {
  const isFunctionComponent = fiber.type instanceof Function;
  if (isFunctionComponent) {
    updateFunctionComponent(fiber); // کامپوننت‌های تابعی
  } else {
    updateHostComponent(fiber);     // نودهای DOM
  }
  // انتخاب واحد کار بعدی: بچه ← خواهر ← پدر
  if (fiber.child) return fiber.child;
  let nextFiber = fiber;
  while (nextFiber) {
    if (nextFiber.sibling) return nextFiber.sibling;
    nextFiber = nextFiber.parent;
  }
}

۴. گام ۵ و ۶: مراحل Render/Commit و Reconciliation

درخت «در حال ساخت» (wip) در مرحلهٔ render ساخته می‌شود و تنها وقتی کامل شد، یک‌باره در مرحلهٔ commit به DOM اعمال می‌شود. مقایسهٔ درخت جدید با قبلی (از طریق alternate) در reconcileChildren انجام می‌شود:

jsfunction reconcileChildren(wipFiber, elements) {
  let index = 0;
  let oldFiber = wipFiber.alternate && wipFiber.alternate.child;
  let prevSibling = null;

  while (index < elements.length || oldFiber != null) {
    const element = elements[index];
    let newFiber = null;

    const sameType = oldFiber && element && element.type === oldFiber.type;

    if (sameType) {
      // همان نوع: به‌روزرسانی props
      newFiber = {
        type: oldFiber.type,
        props: element.props,
        dom: oldFiber.dom,
        parent: wipFiber,
        alternate: oldFiber,
        effectTag: 'UPDATE',
      };
    } else if (element) {
      // عنصر جدید: ساخت نود
      newFiber = {
        type: element.type,
        props: element.props,
        dom: null,
        parent: wipFiber,
        alternate: null,
        effectTag: 'PLACEMENT',
      };
    } else if (oldFiber) {
      // نود حذف‌شده
      oldFiber.effectTag = 'DELETION';
      deletions.push(oldFiber);
    }

    // حرکت روی خواهر و بچه‌ها ...
    if (oldFiber) oldFiber = oldFiber.sibling;
    if (index === 0) wipFiber.child = newFiber;
    else prevSibling.sibling = newFiber;
    prevSibling = newFiber;
    index++;
  }
}

// کامیت: اعمال effectTag ها روی DOM — یک پیمایش کامل
function commitRoot() {
  deletions.forEach(commitWork); // حذف‌ها اول
  commitWork(wipRoot.child);
  currentRoot = wipRoot;
  wipRoot = null;
}

function commitWork(fiber) {
  if (!fiber) return;
  let domParentFiber = fiber.parent;
  while (!domParentFiber.dom) {
    domParentFiber = domParentFiber.parent; // کامپوننت‌ها dom ندارند
  }
  const domParent = domParentFiber.dom;

  if (fiber.effectTag === 'PLACEMENT' && fiber.dom != null) {
    domParent.appendChild(fiber.dom);
  } else if (fiber.effectTag === 'UPDATE' && fiber.dom != null) {
    updateDom(fiber.dom, fiber.alternate.props, fiber.props);
  } else if (fiber.effectTag === 'DELETION') {
    commitDeletion(fiber, domParent);
  }

  commitWork(fiber.child);
  commitWork(fiber.sibling);
}

۵. گام ۷ و ۸: کامپوننت‌های تابعی و Hooks

کامپوننت تابعی همان fiber ای است که type آن تابع است — به‌جای ساخت DOM، تابع را با props صدا می‌زنیم تا بچه‌ها را بگیرد. Hook ها هم با یک متغیر سراسری wipFiber و hookIndex کار می‌کنند — این همان چیزی است که قانون «هوک‌ها را شرطی صدا نزن» را توضیح می‌دهد:

jslet wipFiber = null;
let hookIndex = null;

function updateFunctionComponent(fiber) {
  wipFiber = fiber;
  hookIndex = 0;
  wipFiber.hooks = [];
  // اجرای تابع کامپوننت برای گرفتن بچه‌ها
  const children = [fiber.type(fiber.props)];
  reconcileChildren(fiber, children);
}

function useState(initial) {
  const oldHook =
    wipFiber.alternate &&
    wipFiber.alternate.hooks &&
    wipFiber.alternate.hooks[hookIndex];

  const hook = {
    state: oldHook ? oldHook.state : initial,
    queue: [],
  };

  // اجرای همهٔ اکشن‌های صف‌شده
  const actions = oldHook ? oldHook.queue : [];
  actions.forEach((action) => {
    hook.state = typeof action === 'function' ? action(hook.state) : action;
  });

  const setState = (action) => {
    hook.queue.push(action);
    // برنامه‌ریزی رندر دوباره از ریشه
    wipRoot = {
      dom: currentRoot.dom,
      props: currentRoot.props,
      alternate: currentRoot,
    };
    nextUnitOfWork = wipRoot;
    deletions = [];
  };

  wipFiber.hooks.push(hook);
  hookIndex++;
  return [hook.state, setState];
}

نکتهٔ زیبا: چون useState مقدار قبلی را از alternate می‌خواند، با هر رندر، state حفظ می‌شود و فقط مقادیر تغییرکرده اعمال می‌شوند.

۶. جمع‌بندی

  • React = درخت فیبر + حلقهٔ کار + مقایسه + commit؛ هیچ جادویی نیست.
  • رندر همزمان با شکستن کار به واحدهای کوچک و requestIdleCallback ممکن می‌شود.
  • Hooks فقط آرایه‌ای از state هستند که با شاخص نگه داشته می‌شوند — و به همین دلیل ترتیب صدا زدنشان حیاتی است.
نکتهٔ تکمیلی: اگر این مسیر را دوست داشتید، مقالهٔ «Didact» از همان نویسنده (گام‌به‌گام با مخزن کامل کد) و ویدیوی «Building React From Scratch» از پال او شنسِی را هم ببینید.
کپیرایت: مقالهٔ «Build your own React» اثر رودریگو پومبو (Rodrigo Pombo) است و در pomb.us منتشر شده. این صفحه فقط ترجمهٔ فارسی و مرور آن است و تمام حقوق محتوا و کدها متعلق به نویسندهٔ اصلی است. برای مشاهدهٔ اصل مطلب به سایت اصلی مراجعه کنید.
