مقدمه

بیاموزید چگونه یک ابزار خط فرمان (CLI) در گو بسازید که مخازن گیت محلی شما را پویش می‌کند و یک نمودار مشارکت‌ها (contributions) به سبک گیت‌هاب رسم می‌کند؛ با استفاده از فلگ‌های خط فرمان برای افزودن پوشه‌ها.

چند سال پیش یک برنامهٔ دسکتاپی با Electron + Meteor.js + کتابخانهٔ gitlog ساختم که مخازن گیت محلی‌ام را پویش می‌کرد و نمودار مشارکت‌های خوبی به من می‌داد؛ مشابه نموداری که در گیت‌هاب (GitHub) نمایش داده می‌شود:

آن زمان قبل از اینکه تک‌تک برنامه‌ها از Electron استفاده کنند بود و من واقعاً از این رویکرد خوشم نمی‌آمد؛ چون حجم اپلیکیشن تولیدشده حدود ۵۰ برابر بزرگ‌تر از برنامهٔ مبتنی بر MacGap (مبتنی بر WebKit) بود. به هر حال، خروجی این‌طور به نظر می‌رسید؛ با رابط کاربری شبیه گیت‌هاب:

این ابزار برایم مفید بود؛ چون همهٔ پروژه‌هایم روی گیت‌هاب نیستند (بعضی در BitBucket یا GitLab هستند)، اما همهٔ کدی که روی آن کار می‌کنم روی لپ‌تاپم است؛ بنابراین وقتی صحبت از مشارکت‌ها می‌شود، لپ‌تاپ من «منبع حقیقت واحد» (single source of truth) است.

این اپلیکیشن هنوز اجرا می‌شود، اما هرگز برای عموم منتشر نشده است.

امروز تصمیم گرفتم این قابلیت را به‌صورت یک دستور کنسول گو (Go console command) منتقل کنم؛ چون هنوز این مفهوم را دوست دارم.

قدم‌های اول

کار را به ۲ بخش تقسیم کردم:

  1. به‌دست آوردن فهرستی از پوشه‌هایی که باید پویش شوند
  2. تولید آمار (stats)

هر دو کار را با یک دستور انجام می‌دهم؛ با استفاده از تجزیهٔ فلگ‌های خط فرمان گو (command line flags). وقتی فلگ -add داده شود، دستور یک پوشه به فهرست اضافه می‌کند. اجرای دستور بدون فلگ، نمودار را تولید می‌کند. بازهٔ زمانی داده‌ها را به ۶ ماه اخیر محدود می‌کنم تا حجم زیادی داده یک‌جا به کاربر تحویل داده نشود.

بیایید یک اسکلت ساده برای این جداسازی دغدغه‌ها (separation of concerns) بنویسیم:

gopackage main

import (
    "flag"
)

// scan با گرفتن یک مسیر، آن و زیرپوشه‌هایش را
// در جست‌وجوی مخازن گیت پویش می‌کند
func scan(path string) {
    print("scan")
}

// stats نمودار خوبی از مشارکت‌های گیت شما تولید می‌کند
func stats(email string) {
    print("stats")
}

func main() {
    var folder string
    var email string
    flag.StringVar(&folder, "add", "", "add a new folder to scan for Git repositories")
    flag.StringVar(&email, "email", "your@email.com", "the email to scan")
    flag.Parse()

    if folder != "" {
        scan(folder)
        return
    }

    stats(email)
}

بخش ۱: به‌دست آوردن فهرست پوشه‌هایی که باید پویش شوند

الگوریتمی که برای این بخش اول دنبال می‌کنم بسیار ساده است:

این بخش از برنامه به ۲ زیربخش تقسیم می‌شود. در زیربخش اول، پوشهٔ داده‌شده به‌عنوان آرگومان را به‌صورت بازگشتی (recursive) در جست‌وجوی مخازن پویش می‌کنم. فهرست پوشه‌های مخازن را در فایلی در دایرکتوری خانه (home) ذخیره می‌کنم که .gogitlocalstats نام دارد.

ببینیم scan() چگونه پر می‌شود. اساساً ۳ خط کد است؛ جدا از تولید خروجی:

go// scan یک پوشهٔ جدید را برای مخازن گیت پویش می‌کند
func scan(folder string) {
    fmt.Printf("Found folders:\n\n")
    repositories := recursiveScanFolder(folder)
    filePath := getDotFilePath()
    addNewSliceElementsToFile(filePath, repositories)
    fmt.Printf("\n\nSuccessfully added\n\n")
}

این گردش کار است:

  1. یک برش (slice) از رشته‌ها را از recursiveScanFolder() می‌گیریم
  2. مسیر فایل نقطه‌داری (dot file) را که قرار است در آن بنویسیم به‌دست می‌آوریم
  3. محتوای برش را در فایل می‌نویسیم

بیایید با مورد ۱) شروع کنیم؛ یعنی پویش پوشه. گزینه‌های مختلفی برای پویش یک پوشه با گو وجود دارد. از filepath.Walk استفاده نمی‌کنم؛ چون وارد تک‌تک پوشه‌ها می‌شود. با ioutil.Readdir کنترل بیشتری داریم. از پوشه‌های vendor و node_modules رد می‌شوم؛ چون می‌توانند تعداد بسیار زیادی پوشه داشته باشند که به آن‌ها علاقه‌ای ندارم. از پوشه‌های .git هم رد می‌شوم؛ اما اگر یکی را پیدا کردم، آن را به برشم اضافه می‌کنم:

go// scanGitFolders فهرستی از زیرپوشه‌های `folder` را برمی‌گرداند که به `.git` ختم می‌شوند.
// پوشهٔ پایهٔ مخزن را برمی‌گرداند؛ یعنی والدِ پوشهٔ .git.
// با پاس دادن یک برش موجودِ `folders`، به‌صورت بازگشتی در زیرپوشه‌ها جست‌وجو می‌کند.
func scanGitFolders(folders []string, folder string) []string {
    // حذف `/` پایانی
    folder = strings.TrimSuffix(folder, "/")

    f, err := os.Open(folder)
    if err != nil {
        log.Fatal(err)
    }
    files, err := f.Readdir(-1)
    f.Close()
    if err != nil {
        log.Fatal(err)
    }

    var path string

    for _, file := range files {
        if file.IsDir() {
            path = folder + "/" + file.Name()
            if file.Name() == ".git" {
                path = strings.TrimSuffix(path, "/.git")
                fmt.Println(path)
                folders = append(folders, path)
                continue
            }
            if file.Name() == "vendor" || file.Name() == "node_modules" {
                continue
            }
            folders = scanGitFolders(folders, path)
        }
    }

    return folders
}

این تابع صریحاً از ورود به پوشه‌هایی به نام vendor یا node_modules خودداری می‌کند؛ چون این پوشه‌ها می‌توانند بسیار حجیم باشند و معمولاً مخازن گیت خود را آن‌جا نمی‌گذارید؛ پس می‌توانیم با خیال راحت از آن‌ها صرف‌نظر کنیم.

نکته: رد شدن از vendor و node_modules پویش را به‌شدت سریع‌تر می‌کند؛ چون این پوشه‌ها می‌توانند هزاران زیرپوشه داشته باشند.

همان‌طور که می‌بینید این یک تابع بازگشتی است و توسط این تابع دیگر شروع می‌شود؛ که برای شروع، یک برش خالی از رشته‌ها به آن پاس می‌دهد:

go// recursiveScanFolder جست‌وجوی بازگشتی مخازن گیتی را شروع می‌کند
// که در زیردرخت `folder` قرار دارند
func recursiveScanFolder(folder string) []string {
    return scanGitFolders(make([]string, 0), folder)
}

قسمت ۲) گردش کار، به‌دست آوردن مسیر فایل نقطه‌داری است که پایگاه دادهٔ مسیرهای مخازن ما را در خود دارد:

go// getDotFilePath فایل نقطه‌دار فهرست مخازن را برمی‌گرداند.
// اگر آن و پوشهٔ اطرافش وجود نداشته باشند، آن‌ها را می‌سازد.
func getDotFilePath() string {
    usr, err := user.Current()
    if err != nil {
        log.Fatal(err)
    }

    dotFile := usr.HomeDir + "/.gogitlocalstats"

    return dotFile

این تابع از تابع Current پکیج os/user برای گرفتن کاربر فعلی استفاده می‌کند؛ که یک struct است که به این صورت تعریف شده:

go// User نمایانگر یک حساب کاربری است.
type User struct {
    // Uid شناسهٔ کاربر است.
    // در سیستم‌های POSIX، این عددی دهدهی است که uid را نشان می‌دهد.
    // در ویندوز، این یک شناسهٔ امنیتی (SID) در قالب رشته است.
    // در Plan 9، این محتوای /dev/user است.
    Uid string
    // Gid شناسهٔ گروه اصلی است.
    // در سیستم‌های POSIX، این عددی دهدهی است که gid را نشان می‌دهد.
    // در ویندوز، این یک SID در قالب رشته است.
    // در Plan 9، این محتوای /dev/user است.
    Gid string
    // Username نام ورود (login) است.
    Username string
    // Name نام واقعی یا نمایشی کاربر است.
    // ممکن است خالی باشد.
    // در سیستم‌های POSIX، این اولین (یا تنها) ورودی در فهرست فیلدهای GECOS است.
    // در ویندوز، این نام نمایشی کاربر است.
    // در Plan 9، این محتوای /dev/user است.
    Name string
    // HomeDir مسیر دایرکتوری خانهٔ کاربر است (اگر داشته باشد).
    HomeDir string
}

ما به ویژگی HomeDir علاقه داریم تا مسیر کامل فایل نقطه‌دارمان را بگیریم:

godotFile := usr.HomeDir + "/.gogitlocalstats"

خب، حالا فهرستی از مخازن داریم، فایلی برای نوشتن آن‌ها، و قدم بعدی برای scan() این است که آن‌ها را بدون افزودن خطوط تکراری ذخیره کند.

روند این است:

  1. مخازن موجودِ ذخیره‌شده در فایل را به یک برش تجزیه کنیم
  2. آیتم‌های جدید را بدون افزودن موارد تکراری به برش اضافه کنیم
  3. برش را در فایل ذخیره کنیم و محتوای موجود را بازنویسی کنیم

این کارِ addNewSliceElementsToFile() است:

go// addNewSliceElementsToFile با گرفتن برشی از رشته‌ها که مسیرها را نشان می‌دهند، آن‌ها را
// در فایل‌سیستم ذخیره می‌کند
// addNewSliceElementsToFile با گرفتن برشی از رشته‌ها که مسیرها را نشان می‌دهند، آن‌ها را
// در فایل‌سیستم ذخیره می‌کند
func addNewSliceElementsToFile(filePath string, newRepos []string) {
    existingRepos := parseFileLinesToSlice(filePath)
    repos := joinSlices(newRepos, existingRepos)
    dumpStringsSliceToFile(repos, filePath)
}

اولین کاری که این تابع انجام می‌دهد، فراخوانی parseFileLinesToSlice() است؛ که یک رشتهٔ مسیر فایل می‌گیرد و برشی از رشته‌ها با محتوای فایل برمی‌گرداند. چیز خاصی نیست:

go// parseFileLinesToSlice با گرفتن رشتهٔ مسیر یک فایل، محتوای
// هر خط را می‌گیرد و آن را به برشی از رشته‌ها تجزیه می‌کند.
func parseFileLinesToSlice(filePath string) []string {
    f := openFile(filePath)
    defer f.Close()

    var lines []string
    scanner := bufio.NewScanner(f)
    for scanner.Scan() {
        lines = append(lines, scanner.Text())
    }
    if err := scanner.Err(); err != nil {
        if err != io.EOF {
            panic(err)
        }
    }

    return lines
}

این تابع openFile() را صدا می‌زند؛ که با گرفتن رشتهٔ مسیر فایل، فایل را باز می‌کند و آن را برمی‌گرداند.

go// openFile فایل واقع‌شده در `filePath` را باز می‌کند. اگر وجود نداشته باشد، آن را می‌سازد.
func openFile(filePath string) *os.File {
    f, err := os.OpenFile(filePath, os.O_APPEND|os.O_WRONLY, 0755)
    if err != nil {
        if os.IsNotExist(err) {
            // فایل وجود ندارد
            _, err = os.Create(filePath)
            if err != nil {
                panic(err)
            }
        } else {
            // خطای دیگر
            panic(err)
        }
    }

    return f
}

در این حالت سعی می‌کند فایل نقطه‌دار ما را باز کند. اگر خطایی رخ دهد و خطا به ما بگوید فایل وجود ندارد (با استفاده از os.IsNotExist())، فایل را با os.Create() می‌سازیم تا بتوانیم پر کردنش را با مخازن پویش‌شده شروع کنیم. توصیفگر فایلِ باز (file descriptor) را برمی‌گرداند.

addNewSliceElementsToFile() بعد از گرفتن توصیفگر فایل، بلافاصله f.Close() را معوق (defer) می‌کند تا بعد از پایان کار تابع، فایل بسته شود. سپس parseFileLinesToSlice() را صدا می‌زند؛ یک تابع کمکی که هر خط از یک فایل را به یک برش رشته‌ای تجزیه می‌کند.

سپس joinSlices() با گرفتن ۲ برش، محتوای اولی را به دومی اضافه می‌کند؛ فقط اگر آن محتوا قبلاً وجود نداشته باشد. این کار از خطوط تکراری جلوگیری می‌کند.

بگذارید ببینیم:

go// joinSlices عناصر برش `new` را به برش `existing`
// اضافه می‌کند؛ فقط اگر قبلاً آن‌جا نباشند
func joinSlices(new []string, existing []string) []string {
    for _, i := range new {
        if !sliceContains(existing, i) {
            existing = append(existing, i)
        }
    }
    return existing
}

// sliceContains اگر `slice` شامل `value` باشد true برمی‌گرداند
func sliceContains(slice []string, value string) bool {
    for _, v := range slice {
        if v == value {
            return true
        }
    }
    return false
}

آخرین چیز فراخوانی dumpStringsSliceToFile() است؛ که با گرفتن یک برش از رشته‌ها و یک مسیر فایل، آن برش را با هر رشته در یک خط جدید در فایل می‌نویسد:

go// dumpStringsSliceToFile محتوا را در فایل واقع در مسیر `filePath` می‌نویسد (محتوای موجود را بازنویسی می‌کند)
func dumpStringsSliceToFile(repos []string, filePath string) {
    content := strings.Join(repos, "\n")
    ioutil.WriteFile(filePath, []byte(content), 0755)
}

import (
    "bufio"
    "fmt"
    "io"
    "io/ioutil"
    "log"
    "os"
    "os/user"
    "strings"
)

// getDotFilePath فایل نقطه‌دار فهرست مخازن را برمی‌گرداند.
// اگر آن و پوشهٔ اطرافش وجود نداشته باشند، آن‌ها را می‌سازد.
func getDotFilePath() string {
    usr, err := user.Current()
    if err != nil {
        log.Fatal(err)
    }

    dotFile := usr.HomeDir + "/.gogitlocalstats"

}

            }

            panic(err)
        }
    }

    return f
}

// هر خط را می‌گیرد و آن را به برشی از رشته‌ها تجزیه می‌کند.

func sliceContains(slice []string, value string) bool {
    for _, v := range slice {
        if v == value {
            return true
        }
    }
    return false
}

// joinSlices عناصر برش `new` را به برش `existing`
// اضافه می‌کند؛ فقط اگر قبلاً آن‌جا نباشند

// dumpStringsSliceToFile محتوا را در فایل واقع در مسیر `filePath` می‌نویسد (محتوای موجود را بازنویسی می‌کند)

// addNewSliceElementsToFile با گرفتن برشی از رشته‌ها که مسیرها را نشان می‌دهند، آن‌ها را

    fmt.Printf("\n\n Successfully added\n\n")
}

// scanGitFolders فهرستی از زیرپوشه‌های `folder` را برمی‌گرداند که به `.git` ختم می‌شوند.
// پوشهٔ پایهٔ مخزن را برمی‌گرداند؛ یعنی والدِ پوشهٔ .git.
// با پاس دادن یک برش موجودِ `folders`، به‌صورت بازگشتی در زیرپوشه‌ها جست‌وجو می‌کند.
func scanGitFolders(folders []string, folder string) []string {
    // حذف `/` پایانی
    folder = strings.TrimSuffix(folder, "/")

    f, err := os.Open(folder)
    if err != nil {
        log.Fatal(err)
    }
    files, err := f.Readdir(-1)
    f.Close()
    if err != nil {
        log.Fatal(err)
    }

    var path string

    for _, file := range files {
        if file.IsDir() {
            path = folder + "/" + file.Name()
            if file.Name() == ".git" {
                path = strings.TrimSuffix(path, "/.git")
                fmt.Println(path)
                folders = append(folders, path)
                continue
            }
            if file.Name() == "vendor" || file.Name() == "node_modules" {
                continue
            }
            folders = scanGitFolders(folders, path)
        }
    }

    return folders
}

go// fillCommits با گرفتن یک مخزن که در `path` پیدا شده، commitها را می‌گیرد و
// آن‌ها را در map به نام `commits` می‌گذارد؛ پس از اتمام آن را برمی‌گرداند
func fillCommits(email string, path string, commits map[int]int) map[int]int {
    // ساخت شیء مخزن گیت از مسیر
    repo, err := git.PlainOpen(path)
    if err != nil {
        panic(err)
    }
    // گرفتن ارجاع HEAD
    ref, err := repo.Head()
    if err != nil {
        panic(err)
    }
    // گرفتن تاریخچهٔ commitها با شروع از HEAD
    iterator, err := repo.Log(&git.LogOptions{From: ref.Hash()})
    if err != nil {
        panic(err)
    }
    // پیمایش commitها
    offset := calcOffset()
    err = iterator.ForEach(func(c *object.Commit) error {
        daysAgo := countDaysSinceDate(c.Author.When) + offset

        if c.Author.Email != email {
            return nil

            commits[daysAgo]++
        }

        return nil
    })
    if err != nil {
        panic(err)
    }

    return commits

daysInLastSixMonths یک ثابت (constant) است که به این صورت تعریف می‌شود: const daysInLastSixMonths = 183.

outOfRange هم یک ثابت است؛ به این صورت تعریف می‌شود: const outOfRange = 99999؛ که برخلاف daysInLastSixMonths معنای واقعی ندارد. به‌عنوان مقدار بازگشتی countDaysSinceDate() تنظیم می‌شود؛ وقتی commit قدیمی‌تر از ۶ ماه باشد؛ یعنی بازهٔ تحلیل داده‌های ما.

object توسط پکیج go-git ارائه می‌شود؛ با ایمپورت کردن gopkg.in/src-d/go-git.v4/plumbing/object.

من یک offset (انحراف) به محاسبهٔ «daysAgo» اضافه می‌کنم؛ به خاطر نحوهٔ کار نمودارِ شبیه گیت‌هاب: هر سطر نمایانگر یک نام روز است (از یکشنبه شروع می‌شود) و هر سطر یک هفته را نشان می‌دهد. هفتهٔ جاری را با «دادهٔ جعلی» (fake data) پر می‌کنم.

countDaysSinceDate() برمی‌گرداند که commit چند روز پیش انجام شده است. تاریخ جاری را به دقیقاً شروع روز (00:00:00) بازنشانی می‌کنم تا ساعت‌ها بخشی از معادله نباشند. منطقهٔ زمانی (timezone) از سیستم استنتاج می‌شود.

go// getBeginningOfDay با گرفتن یک time.Time، زمان شروع همان روز را محاسبه می‌کند
func getBeginningOfDay(t time.Time) time.Time {
    year, month, day := t.Date()
    startOfDay := time.Date(year, month, day, 0, 0, 0, 0, t.Location())
    return startOfDay
}

// countDaysSinceDate شمارش می‌کند که از تاریخِ داده‌شدهٔ `date` چند روز گذشته است
func countDaysSinceDate(date time.Time) int {
    days := 0
    now := getBeginningOfDay(time.Now())
    for date.Before(now) {
        date = date.Add(time.Hour * 24)
        days++
        if days > daysInLastSixMonths {
            return outOfRange
        }
    }
    return days
}

تولید ستون‌ها

go// buildCols یک map با ردیف‌ها و ستون‌های آمادهٔ چاپ روی صفحه تولید می‌کند
func buildCols(keys []int, commits map[int]int) map[int]column {
    cols := make(map[int]column)
    col := column{}

    for _, k := range keys {
        week := int(k / 7) //26,25...1
        dayinweek := k % 7 // 0,1,2,3,4,5,6

        if dayinweek == 0 { //بازنشانی
            col = column{}
        }

        col = append(col, commits[k])

        if dayinweek == 6 {
            cols[week] = col
        }
    }

    return cols
}

buildCols() برش کلیدهایی را می‌گیرد که در sortMapIntoSlice() تولید کردیم و map را هم می‌گیرد. یک map جدید می‌سازد؛ به‌جای استفاده از روزها به‌عنوان کلید، از هفته‌ها استفاده می‌کند. نوع column به‌صورت برشی از اعداد صحیح تعریف شده است: type column []int.

هفته با تقسیم اندیس روز بر ۷ تعیین می‌شود و این‌که کدام روز از هفته است با عملیات پیمانه (modulo) k % 7 به‌راحتی به‌دست می‌آید. وقتی روز هفته یکشنبه باشد، یک ستون جدید می‌سازیم و آن را پر می‌کنیم؛ و وقتی شنبه باشد، هفته را به map ستون‌ها اضافه می‌کنیم.

func printMonths() {

		} else {

		}

                if i == 0 && j == calcOffset()-1 {
                    printCell(col[j], true)
                    continue
                } else {
                    if len(col) > j {
                        printCell(col[j], false)
                        continue
                    }
                }
            }
            printCell(0, false)
        }
        fmt.Printf("\n")
    }
}

// printMonths نام ماه‌ها را در خط اول چاپ می‌کند؛ با تشخیص زمانی که
// ماه بین جابه‌جایی هفته‌ها عوض شده است
func printMonths() {
    week := getBeginningOfDay(time.Now()).Add(-(daysInLastSixMonths * time.Hour * 24))
    month := week.Month()
    fmt.Printf("         ")
    for {
        if week.Month() != month {
            fmt.Printf("%s ", week.Month().String()[:3])
            month = week.Month()
        } else {
            fmt.Printf("    ")
        }

        week = week.Add(7 * time.Hour * 24)
        if week.After(time.Now()) {
            break
        }
    }
    fmt.Printf("\n")
}

// printDayCol با گرفتن شمارهٔ روز (۰ یعنی یکشنبه) نام روز را چاپ می‌کند؛
// با در میان‌گذاشتن ردیف‌ها (فقط ۲، ۴، ۶ را چاپ می‌کند)
func printDayCol(day int) {
    out := "     "
    switch day {
    case 1:
        out = " Mon "
    case 3:
        out = " Wed "
    case 5:
        out = " Fri "
    }

    fmt.Printf(out)
}
