۱. داستان مقاله

برنان سال‌ها قبل یک تتریس نوشته بود و این مقاله روایت بازنویسی دوباره آن است: چه چیزی در نسخهٔ اول بد بود، چه درس‌هایی از توسعهٔ آن گرفت و چطور نسخهٔ دوم را با معماری بهتر و کد تمیزتر نوشت. نتیجهٔ کار دو فایل است: tetris.c (منطق کامل بازی — بدون هیچ وابستگی به رابط کاربری) و main.c (لایهٔ نمایش با ncurses).

نکته: جداسازی «منطق بازی» از «رندر» دقیقاً همان معماری است که بازی‌های حرفه‌ای استفاده می‌کنند — می‌توانید رابط کاربری را عوض کنید (مثلاً گرافیکی) بدون دست‌زدن به منطق.

۲. ساختار کد

منطق تتریس شامل: صفحهٔ بازی (board)، قطعات (tetrominoها)، چرخش، برخورد، پاک‌کردن خط و سیستم امتیاز:

c// tetris.c — منطق خالص بازی
// صفحهٔ بازی: ماتریس ۲۰×۱۰
static int board[BOARD_ROWS][BOARD_COLS];

// تعریف ۷ قطعه با ماتریس‌های ۴×۴ (هر قطعه شامل چرخش‌هایش)
typedef struct {
  int shape[4][4];
} Tetromino;

static const Tetromino pieces[] = {
  // I, O, T, S, Z, J, L
};

// بررسی برخورد قطعه با صفحه یا قطعات قبلی
static int collides(const Tetromino *piece, int row, int col) {
  for (int r = 0; r < 4; r++)
    for (int c = 0; c < 4; c++)
      if (piece->shape[r][c] && (board[row + r][col + c] || out_of_bounds))
        return 1;
  return 0;
}

// قرار دادن قطعه روی صفحه
static void place_piece(const Tetromino *piece, int row, int col) {
  for (int r = 0; r < 4; r++)
    for (int c = 0; c < 4; c++)
      if (piece->shape[r][c]) board[row + r][col + c] = 1;
}

// پاک‌کردن خط‌های کامل — از پایین به بالا
static int clear_lines(void) {
  int cleared = 0;
  for (int r = BOARD_ROWS - 1; r >= 0; r--) {
    if (row_full(r)) {
      shift_rows_down(r);
      cleared++;
      r++; // دوباره همان سطر را بررسی کن
    }
  }
  return cleared;
}

۳. رابط کاربری با ncurses

لایهٔ نمایش (main.c) فقط از ncurses برای رسم صفحه، خواندن کلیدها (حرکت، چرخش، فرود سریع) و نمایش امتیاز استفاده می‌کند — و در نهایت هر ۱۰۰ خط پاک‌شده، یک سطح جدید اضافه می‌کند که سرعت بازی بیشتر می‌شود.

c// main.c — نمایش و ورودی
#include <curses.h>
#include "tetris.h"

int main(void) {
  initscr();      // شروع ncurses
  cbreak();       // خواندن کلید بدون Enter
  noecho();
  keypad(stdscr, TRUE);
  nodelay(stdscr, TRUE); // ورودی غیرمسدودکننده

  Game *game = game_new();
  while (!game->over) {
    int key = getch();
    // تبدیل کلیدها به اقدام‌های بازی
    if (key == KEY_LEFT)  game_move(game, LEFT);
    if (key == KEY_RIGHT) game_move(game, RIGHT);
    if (key == KEY_UP)    game_rotate(game);
    if (key == KEY_DOWN)  game_tick(game);
    // ...
    draw(game);           // رسم صفحه، قطعهٔ فعلی و بعدی
    usleep(100000 / game->level);
  }

  endwin();
  return 0;
}

برای اجرا: make و سپس ./tetris — وابستگی‌ها: یک ترمینال و کتابخانهٔ ncurses.

۴. جمع‌بندی

  • تتریس بهترین «اولین بازی» است: منطق ساده ولی کامل (قطعه، چرخش، برخورد، پاک‌کردن خط).
  • معماری دوم‌لایه (منطق/نمایش) ارزش خودش را در بازنویسی نشان می‌دهد.
  • بازنویسی پروژهٔ قبلی، یکی از بهترین تمرین‌های یادگیری است.
