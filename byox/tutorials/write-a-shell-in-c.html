شل را با C بنویس — Write a Shell in C

⚙ زبان: C ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمه فارسی مقالهٔ «Write a shell in C» از استفن برنان (Stephen Brennan، ژانویه ۲۰۱۵) است — مشهورترین آموزش ساخت شل از صفر؛ کد کامل هر بخش به‌همراه تست خطای اجباری (leak checker) و ساختار دادهٔ خط فرمان ارائه می‌شود.

مقدمه — شل چیست؟

شل فرایندها را اجرا و به آن‌ها آموزش می‌دهد که stdin و stdout شان کجاست. و برای آنکه این کار را انجام دهد، یک زبان (اسکریپت‌نویسی) دارد. پس شل دو کار انجام می‌دهد: اجرای برنامه‌ها با fork() و execvp()، و ایجاد زبان که همان «گزاره‌ها» (statements) و «توکن‌ها» (tokens) است.

در این آموزش یک شل مینیمال در C می‌سازیم: خواندن خط (read_line)، تجزیهٔ خط به آرگومان‌ها (split_line)، اجرای دستور (execute — همین اول این کار دشوار را انجام می‌دهیم و بعد سراغ ساختارها می‌رویم)، و چرخهٔ اصلی که این سه را به هم می‌دوزد. سپس به‌ترتیب: builtin ها، یافتن مسیر (PATH)، cd، فرایندهای پس‌زمینه و Ctrl-C، و نهایتاً راه‌اندازی کامل — و در هر مرحله با valgrind --leak-check=yes تست می‌کنیم که حافظه نشت نکند.

خواندن خط فرمان

اول از getline() استفاده می‌کنیم، اما بعداً هنگام دیدن کاراکترهای توکن‌ساز، ساختار ویژه‌ای می‌سازیم که نه‌تنها خط کامل، بلکه هر توکن را جدا نگه می‌دارد تا نهایتاً اجرای builtin ها و پس‌زمینه ممکن شود:

c/* ساختار خط فرمان و نهادهای آن */
typedef struct {
    char *buffer;
    size_t buf_len;
    int argc;
    char **argv;
    char *program_name; /* argv[0] بدون slash، برای پیام‌های خطا */
    int background;     /* آیا این دستور باید پس‌زمینه اجرا شود؟ */
    int stdin, stdout;  /* ریدایرکت؟ */
} lsh_cmd;
c/* ساده‌ترین حالت: خواندن خط از stdin */
char *lsh_read_line(void) {
    char *line = NULL;
    ssize_t bufsize = 0;
    if (getline(&line, &bufsize, stdin) == -1) {
        if (feof(stdin)) exit(EXIT_SUCCESS); /* Ctrl-D */
        else { perror("readline"); exit(EXIT_FAILURE); }
    }
    return line;
}

تجزیهٔ خط به آرگومان

حالت ساده با جداکنندهٔ فاصله و توکن‌بندی با strtok:

c#define LSH_TOK_BUFSIZE 64
#define LSH_TOK_DELIM " \t\r\n\a"

char **lsh_split_line(char *line) {
    int bufsize = LSH_TOK_BUFSIZE, position = 0;
    char **tokens = malloc(bufsize * sizeof(char*));
    char *token;

    if (!tokens) { fprintf(stderr, "lsh: allocation error\n"); exit(EXIT_FAILURE); }

    token = strtok(line, LSH_TOK_DELIM);
    while (token != NULL) {
        tokens[position] = token;
        position++;
        if (position >= bufsize) { /* رشد بافر */ }
        token = strtok(NULL, LSH_TOK_DELIM);
    }
    tokens[position] = NULL;
    return tokens;
}

و در نسخهٔ نهایی، یک توکن‌ساز با جداسازی کاراکترهای خاص >، >>، <، |، & (پس‌زمینه) و ساخت شیء lsh_cmd که ریدایرکت و پس‌زمینه را هم دربرمی‌گیرد.

اجرا — قلب شل

فرایند فرزند با fork() ساخته می‌شود؛ فرزند ریدایرکت را اعمال و برنامه را با execvp اجرا می‌کند؛ اگر شل در حالت عادی باشد، waitpid منتظر می‌ماند (این همان «خط فرمان دوباره می‌آید» است) و اگر پس‌زمینه باشد، شل بلافاصله اعلان بعدی را چاپ می‌کند:

cint lsh_launch(char **args) {
    pid_t pid, wpid;
    int status;

    pid = fork();
    if (pid == 0) {
        /* فرزند: exec هرگز برنمی‌گردد مگر در خطا */
        if (execvp(args[0], args) == -1) perror("lsh");
        exit(EXIT_FAILURE);
    } else if (pid < 0) {
        perror("lsh");
    } else {
        do {
            wpid = waitpid(pid, &status, WUNTRACED);
        } while (!WIFEXITED(status) && !WIFSIGNALED(status));
    }
    return 1;
}

Builtin ها و یافتن برنامه‌ها

وظایف شل (مثل cd، exit، help) باید بدون ساخت فرایند اجرا شوند؛ پس فهرستی از نام‌ها با تابع‌های متناظر نگه می‌داریم (lsh_builtin_str[] و lsh_builtin_func[]). برای یافتن برنامه‌ها هم در مسیرهای PATH می‌گردیم: اول بررسی می‌کنیم نام شامل / هست یا نه؛ اگر نه، با strchr هر مسیرِ PATH را با /name می‌سازیم و با stat وجودش را می‌سنجیم. execvp این کار را برای ما می‌کند، اما دیدن کدِ جست‌وجوی PATH به فهم عمیق‌تر کمک می‌کند: مسیرها را با : از getenv("PATH") جدا کنید، کنار هم قرار دهید و با strdup کپسوله کنید.

پیاده‌سازی cd ساختن دلچسبی است: chdir() فقط در فرایند فعلی مؤثر است، پس باید یک builtin باشد — یعنی نمی‌تواند یک برنامهٔ عادی باشد، چون chdir یک فرایند فرزند را به‌جای خود شل عوض می‌کند!

cint lsh_cd(char **args) {
    if (args[1] == NULL) {
        fprintf(stderr, "lsh: expected argument to \"cd\"\n");
    } else if (chdir(args[1]) != 0) {
        perror("lsh");
    }
    return 1;
}

پس‌زمینه و Ctrl-C

پس‌زمینه یعنی اگر دستور با & تمام شود (lsh->background)، برای فرزند setpgid(0,0) می‌کنیم تا مستقل از ترمینال شود (فشردن Ctrl-C، شل را نکشد چون سیگنال فقط به فرایندهای همان فرایندگروه می‌رود) و waitpid(pid, &status, WNOHANG) برای تعیین اینکه فرایند هنوز زنده است یا نه. نشت‌یاب valgrind --leak-check=yes ./lsh را در هر مرحله اجرا می‌کنیم و مطمئن می‌شویم همهٔ malloc ها آزاد می‌شوند (free_list برای ساختارهای command).

کد کامل و چرخهٔ اصلی

cint main(int argc, char **argv) {
    /* بارگذاری پیکربندی و اجرای دستورات راه‌اندازی */
    lsh_loop();
    return EXIT_SUCCESS;
}

void lsh_loop(void) {
    char *line; char **args; int status;
    do {
        printf("> ");
        line = lsh_read_line();
        args = lsh_split_line(line);
        status = lsh_execute(args);
        free(line); free(args);
    } while (status);
}

(در نسخهٔ کامل، lsh_execute دستورات builtin را می‌شناسد و lsh_launch را صدا می‌زند؛ loop هم روی ساختار lsh_cmd کار می‌کند.) کل کد حدود ۵۰۰ خط C خالص است و در github.com/brenns10/lsh در دسترس است. برای یادگیری بیشتر، ادامهٔ مقاله سری دوم همین آموزش است.
