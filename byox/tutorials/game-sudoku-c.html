۱. دربارهٔ آموزش

سودوکو یک «بازی فکری» کلاسیک است و ساختن حل‌کنندهٔ آن یک تمرین عالی برای یادگیری بازگشت (recursion) و الگوریتم backtracking در C است. در این پلی‌لیست، برنامه‌ای ساخته می‌شود که هر جدول سودوکو را از ورودی می‌خواند و آن را حل می‌کند — با توضیح خط‌به‌خط منطق.

نکته: backtracking همان تکنیک پایه‌ای است که در حل پازل‌ها، مسیریابی (مثل ربات ماز) و حتی موتورهای شطرنج استفاده می‌شود — یادگیری آن با سودوکو بسیار ملموس است.

۳. الگوریتم اصلی

قلب برنامه تابع بازگشتی حل است — همان‌طور که در سری پیاده می‌شود:

c// جدول سودوکو ۹×۹ — خانهٔ خالی با ۰ نمایش داده می‌شود
int board[9][9];

// بررسی اینکه آیا گذاشتن num در (row, col) مجاز است
int is_safe(int row, int col, int num) {
  // بررسی سطر و ستون
  for (int i = 0; i < 9; i++) {
    if (board[row][i] == num || board[i][col] == num) return 0;
  }
  // بررسی خانهٔ ۳×۳
  int box_r = (row / 3) * 3, box_c = (col / 3) * 3;
  for (int i = 0; i < 3; i++)
    for (int j = 0; j < 3; j++)
      if (board[box_r + i][box_c + j] == num) return 0;
  return 1;
}

// حل بازگشتی با backtracking
int solve(void) {
  for (int row = 0; row < 9; row++) {
    for (int col = 0; col < 9; col++) {
      if (board[row][col] == 0) {         // خانهٔ خالی پیدا شد
        for (int num = 1; num <= 9; num++) {
          if (is_safe(row, col, num)) {
            board[row][col] = num;        // فرض کن
            if (solve()) return 1;        // ادامه بده — حل شد؟
            board[row][col] = 0;          // نشد → برگرد (backtrack)
          }
        }
        return 0;                         // هیچ عددی کار نکرد
      }
    }
  }
  return 1; // همهٔ خانه‌ها پر شدند — حل شد
}

این الگوریتم ساده برای تقریباً همهٔ سودوکوهای معمول فوراً جواب می‌دهد. در ویدیوها همچنین خواندن ورودی از فایل، نمایش جدول و بهینه‌سازی‌های ساده (مثل شروع از خانهٔ با کمترین گزینه) پوشش داده می‌شود.
