پشته TCP/IP را کدنویسی کنیم — قسمت ۱: اترنت و ARP

⚙ زبان: C ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمه فارسی «Let's code a TCP/IP stack» از Sami Niiranen است — ساخت پشته شبکه از صفر در C: در قسمت اول، دستگاه مجازی TUN/TAP، تجزیه فریم اترنت و پروتکل ARP پیاده‌سازی می‌شود.

۱. چرا پشته TCP/IP خودمان را بنویسیم؟

هر روز از TCP/IP استفاده می‌کنیم، اما پروتکل‌های آن معمولاً در دل هسته سیستم‌عامل دفن شده‌اند. این سری آموزشی آن‌ها را از گور بیرون می‌آورد: یک پشته شبکه واقعی از صفر در C که مستقیماً با دستگاه شبکه مجازی TUN/TAP هسته لینوکس کار می‌کند. شروع با لایه ۲ (فریم اترنت) و پروتکل ARP، سپس در قسمت‌های بعدی IP و ICMP و TCP/UDP.

۲. دستگاه TUN/TAP

TAP یک دستگاه شبکه مجازی هسته است که به اپلیکیشن‌ها اجازه می‌دهد فریم‌های اترنت (لایه ۲) خام را بدون نیاز به سخت‌افزار شبکه خواندن و نوشتن کنند — دقیقاً همان چیزی که برای یک پشته کاربردی لازم است. تنظیم دستگاه:

c#include <linux/if_tun.h>
#include <linux/if.h>

int tun_alloc(char *dev) {
    int fd = open("/dev/net/tun", O_RDWR);
    struct ifreq ifr;

    memset(&ifr, 0, sizeof(ifr));
    ifr.ifr_flags = IFF_TAP | IFF_NO_PI; // TAP: فریم اترنت، بدون هدر PI

    if (*dev) {
        strncpy(ifr.ifr_name, dev, IFNAMSIZ); // نام دستگاه مثل tap0
    }
    if (ioctl(fd, TUNSETIFF, (void *)&ifr) < 0) {
        perror("TUNSETIFF");
        exit(1);
    }
    return fd;
}

در انتهای مقاله، دستگاه با ip addr add و ip link set up آدرس می‌گیرد تا هسته بسته‌ها را به پشته ما برساند.

۳. ساختار فریم اترنت

فریم اترنت از سربرگ لایه ۲ + محتوا شروع می‌شود. با ساختارهای «بسته‌بندی‌شده» (packed) نمایش داده می‌شود تا کامپایلر padding اضافه نکند:

cstruct eth_hdr {
    unsigned char dmac[6];   // آدرس مک مقصد
    unsigned char smac[6];   // آدرس مک مبدا
    uint16_t ethertype;      // نوع پروتکل بار (آخرین ۲ بایت: IP یا ARP...)
    unsigned char payload[]; // داده
} __attribute__((packed));

struct arp_hdr {
    uint16_t htype;              // نوع سخت‌افزار (۱ = اترنت)
    uint16_t ptype;              // نوع پروتکل (0x0800 = IPv4)
    unsigned char hlen;          // طول آدرس سخت‌افزار (۶)
    unsigned char plen;          // طول آدرس پروتکل (۴)
    uint16_t opcode;             // 1 = درخواست ARP، 2 = پاسخ ARP
    unsigned char sender_mac[6]; // مک فرستنده
    uint32_t sender_ip;          // IP فرستنده
    unsigned char target_mac[6]; // مک هدف
    uint32_t target_ip;          // IP هدف
} __attribute__((packed));

۴. بازکردن فریم‌ها

حلقه اصلی پشته: خواندن فریم از توصیف‌کننده TAP و بازکردن سربرگ‌ها با پرش از روی بایت‌ها (شماره بایت ۰-۱۱ سربرگ اترنت است، ۱۲ به بعد بار):

cint tunfd = tun_alloc("tap0");

for (;;) {
    unsigned char buf[4096];
    ssize_t len = read(tunfd, buf, sizeof(buf));
    if (len < 0) { perror("read"); break; }

    struct eth_hdr *eth = (struct eth_hdr *)buf;

    if (ntohs(eth->ethertype) == ETH_P_ARP) {
        // از روی بار فریم (بعد از سربرگ اترنت) سربرگ ARP را بخوان
        struct arp_hdr *arp = (struct arp_hdr *)(buf + sizeof(struct eth_hdr));
        process_arp(arp);
    }
    /* قسمت‌های بعدی:ethertype == IPv4 و... */
}

۵. پاسخ به ARP

وقتی درخواست «چه کسی این IP را دارد؟» می‌رسد، پشته باید پاسخ ساخت و فرستاد. چون IP مقصد قابل پرس‌وجو است، پاسخ به‌صورت کلاسیک ساخته می‌شود: مبدا/مقصد مک و IP با هم عوض می‌شوند، آدرس مک خود پشته در sender_mac می‌رود و opcode به ۲ (پاسخ) تغییر می‌کند. همین پاسخ مستقیم با write() به TAP نوشته می‌شود تا هسته آن را به شبکه برساند. نگهداری «جدول ARP» برای ترجمه IP→MAC هم بخشی از همین مقاله است.

نکته: آزمایش همه‌چیز با یک ماشین مجازی راحت انجام می‌شود؛ مقاله با نگاشت آدرس‌ها و پیوسته‌های تکراری، اشکال‌زدایی را عملی می‌کند.

۶. ادامه سری و منابع

قسمت‌های بعدی این سری، پروتکل IP (شامل محاسبه Checksum)، ICMP و در نهایت TCP/UDP را اضافه می‌کنند. مقاله برای مطالعه بیشتر به جسم سبک‌وزن lwIP (استفاده‌شده در بسیاری از سیستم‌های تعبیه‌شده)، کتاب کلاسیک «TCP/IP Illustrated» و مطالب دانشگاهی درباره ساختار پشته‌ها ارجاع می‌دهد.

کپیرایت: مقالات «Let's code a TCP/IP stack» اثر سامی نیرانن (Sami Niiranen) است. این صفحه فقط ترجمه فارسی قسمت اول آن است و تمام حقوق محتوا و کدها متعلق به نویسنده اصلی است. برای مشاهده اصل مطلب به سایت اصلی مراجعه کنید.
