Writing a Bootloader — نوشتن بوت‌لودر از صفر (الکس پارکر)

⚙ زبان: C + Assembly 💽 x86 + QEMU ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمهٔ فارسی و مرور مقالهٔ «Writing a Bootloader» از الکس پارکر (Alex Parker) در 3zanders.co.uk است — یک بوت‌لودر کامل دو مرحله‌ای که با اسمبلی و C نوشته می‌شود و یک هستهٔ ساده را بارگذاری می‌کند.
↗ مشاهده منبع اصلی

۱. چرا این آموزش خاص است؟

بیشتر آموزش‌ها یا با GRUB بوت می‌کنند (تا نوشتن بوت‌لودر را دور بزنند) یا فقط یک بوت‌سکتور نمایشی می‌سازند. این مقاله راه میانه را می‌رود: یک بوت‌لودر واقعی دو مرحله‌ای می‌نویسیم که در مرحلهٔ اول (اسمبلی) مرحلهٔ دوم را از دیسک می‌خواند، به حالت ۳۲-بیتی می‌رود، و در مرحلهٔ دوم (C) هسته را بارگذاری و اجرا می‌کند. نتیجه: درک کامل زنجیرهٔ «BIOS ← بوت‌لودر ← هسته».

۲. مرحلهٔ ۱: بوت‌سکتور اسمبلی

بوت‌سکتور ۵۱۲ بایتی اول دیسک، سه کار انجام می‌دهد: پیامی چاپ می‌کند، سکتورهای مرحلهٔ دوم را با سرویس int 0x13 از دیسک می‌خواند و سپس با jmp به آن منتقل می‌شود. خواندن دیسک در حالت ۱۶-بیتی با وقفهٔ BIOS انجام می‌شود:

nasm; خواندن یک سکتور از دیسک با سرویس BIOS (int 0x13)
read_sector:
    mov ah, 0x02        ; تابع: خواندن سکتور
    mov al, 1           ; تعداد سکتورها: یک سکتور
    mov ch, 0           ; سیلندر ۰
    mov cl, 2           ; سکتور ۲ (اولی بوت‌لودر بود)
    mov dh, 0           ; هد ۰
    mov dl, 0x80        ; دیسک اول (هارد)
    mov bx, 0x7E00      ; آدرس بارگذاری در حافظه
    int 0x13            ; فراخوانی BIOS
    jc read_sector      ; در صورت خطا، دوباره تلاش کن
    ret

۳. مرحلهٔ ۲: ورود به حالت ۳۲-بیتی

مرحلهٔ دوم هم‌چنان در اسمبلی است و حالا باید GDT را بارگذاری کند، بیت PE را در cr0 روشن کند و با یک far jump به حالت محافظت‌شده برود. این همان گام حساسی است که تقریباً در تمام پروژه‌های سیستم‌عامل مشترک است:

nasm; ورود به حالت محافظت‌شده
lgdt [gdt_descriptor]
mov eax, cr0
or eax, 1
mov cr0, eax

jmp 0x08:protected       ; far jump: بازنویسی کامل بخش‌ها

[bits 32]
protected:
    mov ax, 0x10
    mov ds, ax
    mov ss, ax
    mov esp, 0x90000     ; پشته‌ی جدید
    ...

۴. مرحلهٔ ۲: هستهٔ C

بعد از ورود به حالت ۳۲-بیتی، کنترل به کد C سپرده می‌شود. مقاله برای سادگی، هسته را با یک فایل Makefile می‌سازد و در QEMU اجرا می‌کند تا پیام نهایی را ببینیم:

bash# ساخت بوت‌لودر و هسته و اجرا
make
qemu-system-i386 -fda os.img

۵. چه چیزهایی یاد می‌گیرید؟

  • سرویس‌های BIOS برای چاپ و خواندن دیسک.
  • ساختار یک بوت‌لودر دو مرحله‌ای (که در دنیای واقعی هم همین الگوست).
  • گذار از حالت ۱۶-بیتی به ۳۲-بیتی با GDT.
  • همکاری اسمبلی و C در یک پروژهٔ bare-metal.
نکتهٔ تکمیلی: این آموزش بهترین پل بین «بوت‌سکتور نمایشی» (مثل آموزش جو برگرون) و «کرنل با GRUB» (مثل Kernel 101) است — هر سه در همین مجموعه ترجمه شده‌اند.
