مقدمه

من دو آموزش ساخت ابزار خط فرمان نوشته‌ام برای ساخت gololcat و gocowsay. در هر دوی آن‌ها از fortune به‌عنوان تولیدکنندهٔ ورودی استفاده کردم.

در این مقاله، سه‌گانهٔ لوله‌کشی (pipe) را با gofortune کامل می‌کنم.

اول از همه، fortune چیست؟ همان‌طور که ویکی‌پدیا می‌گوید، Fortune یک برنامهٔ ساده است که یک پیام شبه‌تصادفی (pseudorandom) را از یک پایگاه‌داده از نقل‌قول‌ها (quotations) نمایش می‌دهد.

اساساً، یک تولیدکنندهٔ نقل‌قول تصادفی (random quote generator).

تاریخچه‌ای بسیار طولانی دارد که به نسخهٔ ۷ یونیکس (Unix Version 7) در سال ۱۹۷۹ برمی‌گردد. هنوز هم سرحال و فعال است. بسیاری از توزیع‌های لینوکس آن را از قبل نصب می‌کنند و در OSX می‌توانید آن را با brew install fortune نصب کنید.

در برخی سیستم‌ها، هنگام استفاده از شل‌ها (shells) به‌عنوان پیام خوش‌آمدگویی یا پیام خداحافظی استفاده می‌شود.

ویکی‌پدیا همچنین می‌گوید:

بسیاری از افراد تصمیم می‌گیرند خروجی fortune را به دستور cowsay لوله‌کشی (pipe) کنند تا کمی طنز به دیالوگ اضافه شود.

این دقیقاً منم! به‌جز اینکه من از دستور gocowsay خودم استفاده می‌کنم.

بس است با مقدمه؛ بیایید یک کلون fortune با Go بسازیم.

برنامه چه کاری انجام می‌دهد

در این‌جا تجزیهٔ کاری که برنامهٔ ما انجام می‌دهد آمده است:

یافتن مسیر پوشهٔ fortunes

موقعیت پوشهٔ fortunes به سیستم و توزیع بستگی دارد و یک پرچم ساخت (build flag) است. می‌توانستم آن را هاردکد کنم یا از یک متغیر محیطی (environment variable) استفاده کنم؛ اما به‌عنوان یک تمرین، یک کار کثیف انجام می‌دهم و مستقیماً از خود fortune می‌پرسم؛ با اجرای آن با پرچم -f، که خروجی زیر را تولید می‌کند:

خط اول خروجی شامل مسیر پوشهٔ fortunes است.

gopackage main

import (
	"fmt"
	"os/exec"
)

func main() {
	out, err := exec.Command("fortune", "-f").CombinedOutput()
	if err != nil {
		panic(err)
	}

	fmt.Println(string(out))
}

این قطعه‌کد خروجی را دقیقاً همان‌طور که دریافت کردم بازتولید می‌کند. به نظر می‌رسد fortune -f خروجی را به stderr می‌نویسد؛ به همین دلیل از CombinedOutput استفاده کردم تا هم stdout و هم stderr را دریافت کنم.

دریافت فقط خط اول

اما من فقط خط اول را می‌خواهم. چطور این کار را انجام دهم؟ این کد تمام خروجی stderr را خط‌به‌خط چاپ می‌کند:

gopackage main

import (
	"bufio"
	"fmt"
	"os/exec"
)

func main() {
	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	for outputStream.Scan() {
	 	fmt.Println(outputStream.Text())
	}
}

برای گرفتن فقط خط اول، حلقهٔ for را حذف می‌کنم و فقط خط اول را اسکن می‌کنم:

gopackage main

import (
	"bufio"
	"fmt"
	"os/exec"
)

func main() {
	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	fortuneCommand.Start()
	outputStream := bufio.NewScanner(pipe)
	outputStream.Scan()
	fmt.Println(outputStream.Text())
}

استخراج مسیر

حالا آن خط را برمی‌داریم و مسیر را از آن استخراج می‌کنیم.

در سیستم من خط اول خروجی این است: 100.00% /usr/local/Cellar/fortune/9708/share/games/fortunes. بیایید یک زیررشته (substring) بسازیم که از اولین رخداد کاراکتر / شروع می‌شود:

goline := outputStream.Text()
path := line[strings.Index(line, "/"):]

حالا مسیر fortunes را دارم. می‌توانم فایل‌های موجود در آن‌جا را فهرست کنم. فایل‌های دودویی .dat و فایل‌های متن ساده وجود دارند. من فایل‌های دودویی و کل پوشهٔ off/ را که حاوی fortuneهای توهین‌آمیز (offensive) است کنار می‌گذارم.

فهرست کردن فایل‌ها

اول فایل‌ها را فهرست کنیم. من از متد Walk بستهٔ path/filepath برای پیمایش درخت فایل‌ها از root استفاده می‌کنم. آن را به‌جای ioutil.ReadDir() استفاده می‌کنم، چون ممکن است پوشه‌های تو در توی fortunes داشته باشیم. در WalkFunc به نام visit، فایل‌های .dat را با filepath.Ext() کنار می‌گذارم، فایل‌های پوشه (مثلاً /off، اما نه فایل‌های موجود در زیرپوشه‌ها) و تمام fortuneهای توهین‌آمیز را که به‌طور مناسب زیر /off قرار دارند کنار می‌گذارم و مقدار هر فایل باقی‌مانده را چاپ می‌کنم.

gofunc visit(path string, f os.FileInfo, err error) error {
	if strings.Contains(path, "/off/") {
		return nil
	}
	if filepath.Ext(path) == ".dat" {
		return nil
	}
	if f.IsDir() {
		return nil
	}
	files = append(files, path)
	return nil
}

func main() {
	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	fortuneCommand.Start()
	outputStream := bufio.NewScanner(pipe)
	outputStream.Scan()
	line := outputStream.Text()
	root := line[strings.Index(line, "/"):]

	err = filepath.Walk(root, visit)
	if err != nil {
		panic(err)
	}
}

بیایید این مقادیر را در یک برش (slice) قرار دهیم تا بعداً بتوانم یکی را به‌صورت تصادفی انتخاب کنم: یک برش files از رشته‌ها تعریف می‌کنم و در تابع visit() به آن اضافه می‌کنم. در پایان main() تعداد فایل‌هایی را که به‌دست آوردم چاپ می‌کنم.

gopackage main

import (
	"bufio"
    "log"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

var files []string

func visit(path string, f os.FileInfo, err error) error {
    if err != nil {
        log.Fatal(err)
    }
	if strings.Contains(path, "/off/") {
		return nil
	}
	if filepath.Ext(path) == ".dat" {
		return nil
	}
	if f.IsDir() {
		return nil
	}
	files = append(files, path)
	return nil
}

func main() {
	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	fortuneCommand.Start()
	outputStream := bufio.NewScanner(pipe)
	outputStream.Scan()
	line := outputStream.Text()
	root := line[strings.Index(line, "/"):]

	err = filepath.Walk(root, visit)
	if err != nil {
		panic(err)
	}

	println(len(files))
}

خواندن نقل‌قول تصادفی از فایل

چیزی که الان کم دارم، اسکن کردن fortuneهای داخل یک فایل و چاپ یک‌تای آن به‌صورت تصادفی است. در هر فایل، نقل‌قول‌ها با یک % که روی یک خط به‌تنهایی نشسته است از هم جدا می‌شوند. به‌راحتی می‌توانم این الگو را تشخیص دهم و هر نقل‌قول را در یک آرایه اسکن کنم:

gofile, err := os.Open(randomFile)
if err != nil {
    panic(err)
}
defer file.Close()

b, err := ioutil.ReadAll(file)
if err != nil {
    panic(err)
}

quotes := string(b)

quotesSlice := strings.Split(quotes, "%")
j := randomInt(1, len(quotesSlice))

fmt.Print(quotesSlice[j])

این کار واقعاً کارآمد نیست، چون کل فایل fortune را در یک برش اسکن می‌کنم و بعد یک مورد تصادفی انتخاب می‌کنم؛ اما کار می‌کند:

بنابراین، این نسخهٔ نهایی کلون fortune بسیار سادهٔ ماست. بخش‌های زیادی از دستور اصلی fortune را ندارد، اما شروعی است.

gopackage main

import (
	"bufio"
	"fmt"
	"io/ioutil"
    "log"
	"math/rand"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"
)

var files []string

// یک عدد صحیح بزرگ‌تر یا مساوی min و کوچک‌تر از max برمی‌گرداند
func randomInt(min, max int) int {
	return min + rand.Intn(max-min)
}

func visit(path string, f os.FileInfo, err error) error {
    if err != nil {
        log.Fatal(err)
    }
	if strings.Contains(path, "/off/") {
		return nil
	}
	if filepath.Ext(path) == ".dat" {
		return nil
	}
	if f.IsDir() {
		return nil
	}
	files = append(files, path)
	return nil
}

func main() {
	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	fortuneCommand.Start()
	outputStream := bufio.NewScanner(pipe)
	outputStream.Scan()
	line := outputStream.Text()
	root := line[strings.Index(line, "/"):]

	err = filepath.Walk(root, visit)
	if err != nil {
		panic(err)
	}

	rand.Seed(time.Now().UnixNano())
	i := randomInt(1, len(files))
	randomFile := files[i]

	file, err := os.Open(randomFile)
	if err != nil {
		panic(err)
	}
	defer file.Close()

	b, err := ioutil.ReadAll(file)
	if err != nil {
		panic(err)
	}

	quotes := string(b)

	quotesSlice := strings.Split(quotes, "%")
	j := randomInt(1, len(quotesSlice))

	fmt.Print(quotesSlice[j])
}

جمع‌بندی

برای جمع‌بندی، visit را به‌عنوان یک آرگومان تابع درون‌خطی (inline) برای filepath.Walk منتقل می‌کنم و files را به‌جای یک متغیر سراسری فایل، به یک متغیر محلی داخل main() تبدیل می‌کنم:

gopackage main

import (
	"bufio"
	"fmt"
	"io/ioutil"
    "log"
	"math/rand"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"
)

// یک عدد صحیح بزرگ‌تر یا مساوی min و کوچک‌تر از max برمی‌گرداند
func randomInt(min, max int) int {
	return min + rand.Intn(max-min)
}

func main() {
	var files []string

	fortuneCommand := exec.Command("fortune", "-f")
	pipe, err := fortuneCommand.StderrPipe()
	if err != nil {
		panic(err)
	}
	fortuneCommand.Start()
	outputStream := bufio.NewScanner(pipe)
	outputStream.Scan()
	line := outputStream.Text()
	root := line[strings.Index(line, "/"):]

	err = filepath.Walk(root, func(path string, f os.FileInfo, err error) error {
        if err != nil {
            log.Fatal(err)
        }
		if strings.Contains(path, "/off/") {
			return nil
		}
		if filepath.Ext(path) == ".dat" {
			return nil
		}
		if f.IsDir() {
			return nil
		}
		files = append(files, path)
		return nil
	})
	if err != nil {
		panic(err)
	}

	rand.Seed(time.Now().UnixNano())
	i := randomInt(1, len(files))
	randomFile := files[i]

	file, err := os.Open(randomFile)
	if err != nil {
		panic(err)
	}
	defer file.Close()

	b, err := ioutil.ReadAll(file)
	if err != nil {
		panic(err)
	}

	quotes := string(b)

	quotesSlice := strings.Split(quotes, "%")
	j := randomInt(1, len(quotesSlice))

	fmt.Print(quotesSlice[j])
}
