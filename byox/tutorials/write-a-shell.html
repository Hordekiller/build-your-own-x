شل را با C بنویس (Write a shell in C — dash)

⚙ زبان: C ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمه فارسی مقالهٔ «Write a shell in C» از دَنیش پرکاش (Danish Prakash، ژانویه ۲۰۱۸) است — نوشتن شل «dash» در C با کد کامل، از صفر تا اجرای دستورها؛ نویسنده نسخهٔ پیشرفته‌تر (با پایپ، تاریخچه و ...) را هم در گیت‌هاب منتشر کرده.
↗ مشاهده منبع اصلی

مقدمه

تا به حال فکر کرده‌اید آن ترمینال چطور کار می‌کند؟ (سؤال دقیق‌تر: آن شل چطور کار می‌کند؟) نویسنده هم همین سؤال را داشت و بعد از «چند جست‌وجو و کلی مطالعه» فهمید: اول یک شل پایه در C نوشت و بعد چند قابلیت دیگر رویش سوار کرد. شل به زبان ساده «ابزاری است که از طریق آن با سیستمعامل تعامل می‌کنید» — تعریفی گنگ اما گویا. نیازمندی‌ها: دانش C، یک gcc و یک ویرایشگر متن.

حلقهٔ اصلی شل

این تابع، موتور محرک حلقه است. اول از همه یک prompt («») کاربر را از آمادگی شل باخبر می‌کند:

cvoid loop() {
    char *line;
    char **args;
    int status = 1;

    do {
        printf("> ");
        line = read_line();
        args = split_line(line);
        status = dash_execute(args);
        free(line);
        free(args);
    } while (status);
}

status مقدار بازگشتی توابع اجرای دستور را نگه می‌دارد و پایان حلقه را تعیین می‌کند: وقتی کاربر exit را وارد کند، تابع exit مقدار 0 برمی‌گرداند و حلقه می‌شکند. دو خط آخرِ do-while با free() حافظهٔ دو اشاره‌گر را آزاد می‌کنند — آزادسازی صریح حافظه در C ضروری و عادت خوبی است.

خواندن دستورها — read_line

چون نمی‌توانید طول دستورِ کاربر را از قبل بدانید، تخصیص ایستا بی‌معنی است: حافظه را به‌صورت پویا تخصیص می‌دهیم و هر وقت لازم شد بیشتر می‌کنیم:

cchar *read_line() {
    int buffsize = 1024;
    int position = 0;
    char *buffer = malloc(sizeof(char) * buffsize);
    int c;

    if (!buffer) {                          /* malloc ناکام */
        fprintf(stderr, "dash: Allocation error\n");
        exit(EXIT_FAILURE);
    }

    while (1) {
        c = getchar();
        if (c == EOF || c == '\n') {
            buffer[position] = '\0';
            return buffer;
        } else {
            buffer[position] = c;
        }
        position++;

        if (position >= buffsize) {         /* رشد بافر */
            buffsize += 1024;
            buffer = realloc(buffer, buffsize);
            if (!buffer) {
                fprintf(stderr, "dash: Allocation error\n");
                exit(EXIT_FAILURE);
            }
        }
    }
}

وقتی EOF یا \n رسید، تهی‌پایان‌گر برمی‌گردد؛ بررسی‌های if(!buffer) هم اطمینان می‌دهند که در صورت شکست تخصیص (NULL از malloc/realloc) خطا بگیریم.

توکن‌سازی ورودی — split_line

دستورِ خوانده‌شده را توکن (تکه‌تکه) می‌کنیم تا اجرایش آسان شود. کار با تابع استاندارد strtok است: فراخوانی اول token اول را برمی‌گرداند و هر فراخوانی بعدی با ورودی NULL از همان‌جا ادامه می‌دهد:

cstr1 = strtok("this is it!", " ");
// str1 -> "this"
str1 = strtok(NULL, " ");
// str1 -> "is"
str1 = strtok(NULL, " ");
// str1 -> "it!"
c#define TOK_DELIM " \t\r\n"
#define TK_BUFF_SIZE 1024

char **split_line(char *line) {
    int buffsize = TK_BUFF_SIZE, position = 0;
    char **tokens = malloc(buffsize * sizeof(char *));
    char *token;

    if (!tokens) {
        fprintf(stderr, "dash: Allocation error\n");
        exit(EXIT_FAILURE);
    }
    token = strtok(line, TOK_DELIM);
    while (token != NULL) {
        tokens[position] = token;
        position++;
        if (position >= buffsize) {
            buffsize += TK_BUFF_SIZE;
            tokens = realloc(tokens, buffsize * sizeof(char *));
            if (!tokens) { /* ... */ }
        }
        token = strtok(NULL, TOK_DELIM);
    }
    tokens[position] = NULL;
    return tokens;
}

خروج — dash_exit و اجرای دستور — dash_execute

یک تابع پیش‌پاافتاده که 0 برمی‌گرداند تا حلقه بسته شود؛ و در dash_execute اول بررسی می‌کنیم که آیا دستور exit است (برای خوانایی، تابع جدا گرفته شده). اگر نه، fork() فرایند فعلی را کپی می‌کند («فرزند» که دقیقاً مثل والد است جز شناسهٔ فرایندش) و مقدار بازگشتی می‌گوید کجاییم: 0 یعنی فرزند، و pid فرزند یعنی والد:

cint dash_execute(char **args) {
    pid_t cpid;
    int status;

    if (strcmp(args[0], "exit") == 0) {
        return dash_exit(args);
    }

    cpid = fork();

    if (cpid == 0) {
        /* فرزند: execvp تصویر فرایند را با برنامهٔ موردنظر عوض می‌کند؛
           در شکست -1 برمی‌گرداند */
        if (execvp(args[0], args) < 0)
            printf("dash: command not found: %s\n", args[0]);
        exit(EXIT_FAILURE);

    } else if (cpid < 0) {
        printf("Error forking\n");
    } else {
        waitpid(cpid, &status, WUNTRACED);
    }
    return 1;
}

int dash_exit(char **args) {
    return 0;
}

waitpid هم تضمین می‌کند فرزند قبل از برگشت شل به اعلان، کارش را تمام کند.

کد کامل و نتیجه‌گیری

کل کد بالا — شامل پروتوتایپ‌ها، تعریف TOK_DELIM، ماکروهای رنگیِ خطا و main() که فقط loop() را صدا می‌زند — یک شل پایهٔ کامل می‌سازد. این تمرینی «یادگیری»، نه محصول: «احتمالاً هرگز نمی‌خواهید از شلی این‌قدر پایه استفاده کنید، اما حالا می‌دانید شل‌های محبوب‌تان زیر پوست چطور کار می‌کنند.»
