ساخت برنامهٔ خط فرمان خودت از صفر با Zig

مقدمه

رابط‌های خط فرمان (Command-Line Interfaces / CLI) ابزارهایی ضروری برای توسعه‌دهندگان، مدیران سیستم و کاربران حرفه‌ای هستند. در این آموزش، ساخت یک اپ CLI قدرتمند و انعطاف‌پذیر در Zig را از پایه (from scratch) پیش می‌گیریم. در پایان، یک چارچوب (framework) CLI خواهید داشت که از فرمان‌ها، گزینه‌ها، مدیریت خطا و سازگاری چندسکویی (cross-platform) پشتیبانی می‌کند.

زبان برنامه‌نویسی Zig عملکرد زبان‌های سطح پایین را با ویژگی‌های ایمنی مدرن ترکیب می‌کند و آن را انتخابی عالی برای توسعهٔ برنامه‌های CLI می‌سازد. ما یک چارچوب CLI قابل‌استفاده‌مجدد می‌سازیم که می‌تواند:

  • فرمان‌ها (commands) را تجزیه و مدیریت کند
  • گزینه‌های کوتاه و بلند (short and long-form options) همراه با مقدار پردازش کند
  • گزینه‌های اجباری (required options) را اعمال کند
  • گزینه‌های اختیاری (optional options) را مدیریت کند
  • سازگاری چندسکویی (ویندوز، لینوکس، مک‌اواس) فراهم کند

رویکرد ما از یک طراحی تمیز و ماژولار پیروی می‌کند که به‌راحتی می‌توانید آن را در پروژه‌های خودتان به‌کار ببرید.

پیش‌نیازها (Prerequisites)

  • آشنایی اولیه با نحو (syntax) زبان Zig
  • کامپایلر Zig نصب‌شده (نسخهٔ ۰٫۱۱٫۰ یا بالاتر توصیه می‌شود)
  • یک ویرایشگر متن یا IDE با پشتیبانی از Zig

گام ۱: انواع اصلی CLI (Core CLI Types)

ابتدا، ساختارهای دادهٔ اصلی CLI خود را در cli.zig تعریف می‌کنیم. برای نمایش فرمان‌ها و گزینه‌ها به انواع (types) داده نیاز داریم:

zigconst std = @import("std");
const builtin = @import("builtin");

pub const MAX_COMMANDS: u8 = 10;
pub const MAX_OPTIONS: u8 = 20;

const Byte = u8;
const Slice = []const Byte;
const Slices = []const Slice;

/// ساختاری برای نمایش نوع فرمان.
pub const command = struct {
    name: Slice,                     // نام فرمان
    func: fnType,                    // تابع اجرای فرمان
    req: Slices = &.{},              // گزینه‌های اجباری
    opt: Slices = &.{},              // گزینه‌های اختیاری
    const fnType = *const fn ([]const option) bool;
};

/// ساختاری برای نمایش نوع گزینه.
pub const option = struct {
    name: Slice,                     // نام گزینه
    func: ?fnType = null,            // تابع اجرای گزینه
    short: Byte,                     // فرم کوتاه، مثلاً -n|-N
    long: Slice,                     // فرم بلند، مثلاً --name
    value: Slice = "",               // مقدار گزینه
    const fnType = *const fn (Slice) bool;
};

/// خطاهای احتمالی هنگام اجرای CLI
pub const Error = error{
    NoArgsProvided,
    UnknownCommand,
    UnknownOption,
    MissingRequiredOption,
    UnexpectedArgument,
    CommandExecutionFailed,
    TooManyCommands,
    TooManyOptions,
};

گام ۲: پیاده‌سازی تجزیه‌کنندهٔ فرمان (Command Parser Implementation)

حالا، عملکرد اصلی را می‌سازیم که آرگومان‌های خط فرمان را تجزیه (parse) می‌کند و فرمان‌ها را اجرا می‌کند:

zig/// اجرای برنامهٔ CLI را آغاز می‌کند.
pub fn start(commands: []const command, options: []const option, debug: bool) !void {
    if (commands.len > MAX_COMMANDS) {
        return error.TooManyCommands;
    }
    if (options.len > MAX_OPTIONS) {
        return error.TooManyOptions;
    }

    // ایجاد یک تخصیص‌دهندهٔ همه‌کاره (general-purpose allocator) برای مدیریت حافظه هنگام اجرا
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

    // دریافت آرگومان‌های خط فرمان به روشی چندسکویی
    const args = try std.process.argsAlloc(allocator);
    defer std.process.argsFree(allocator, args);

    try startWithArgs(commands, options, args, debug);
}

/// برنامهٔ CLI را با آرگومان‌های داده‌شده آغاز می‌کند.
pub fn startWithArgs(commands: []const command, options: []const option, args: anytype, debug: bool) !void {
    if (args.len < 2) {
        if(debug) std.debug.print("No command provided by user!\n", .{});
        return Error.NoArgsProvided;
    }

    // استخراج نام فرمان (دومین آرگومان پس از نام برنامه)
    const command_name = args[1];
    var detected_command: ?command = null;

    // جست‌وجو در فهرست فرمان‌های موجود برای یافتن تطابق
    for (commands) |cmd| {
        if (std.mem.eql(u8, cmd.name, command_name)) {
            detected_command = cmd;
            break;
        }
    }

    // اگر فرمان تطبیقی یافت نشد، خطا برگردان
    if (detected_command == null) {
        if(debug) std.debug.print("Unknown command: {s}\n", .{command_name});
        return Error.UnknownCommand;
    }

    // دریافت فرمان تطبیق‌یافته از متغیر اختیاری
    const cmd = detected_command.?;

    if(debug) std.debug.print("Detected command: {s}\n", .{cmd.name});

    // تخصیص حافظه برای گزینه‌های تشخیص‌داده‌شده بر اساس آرگومان‌های باقی‌مانده
    var detected_options: [MAX_OPTIONS]option = undefined;
    var detected_len: usize = 0;
    var i: usize = 2;

    // تجزیهٔ گزینه‌ها برای دریافت مقادیرشان
    while (i < args.len) {
        const arg = args[i];

        if (std.mem.startsWith(u8, arg, "-")) {
            const option_name = if (std.mem.startsWith(u8, arg[1..], "-")) arg[2..] else arg[1..];
            var matched_option: ?option = null;

            for (options) |opt| {
                if (std.mem.eql(u8, option_name, opt.long) or (option_name.len == 1 and option_name[0] == opt.short)) {
                    matched_option = opt;
                    break;
                }
            }

            if (matched_option == null) {
                if(debug) std.debug.print("Unknown option: {s}\n", .{arg});
                return Error.UnknownOption;
            }

            var opt = matched_option.?;

            // تشخیص مقدار گزینه
            if (i + 1 < args.len and !std.mem.startsWith(u8, args[i + 1], "-")) {
                opt.value = args[i + 1];
                i += 1;
            } else {
                opt.value = "";
            }

            if (detected_len >= MAX_OPTIONS) {
                return error.TooManyOptions;
            }

            detected_options[detected_len] = opt;
            detected_len += 1;
        } else {
            if(debug) std.debug.print("Unexpected argument: {s}\n", .{arg});
            return Error.UnexpectedArgument;
        }

        i += 1;
    }

    // برش دادن گزینه‌های تشخیص‌داده‌شده به تعداد واقعی آن‌ها
    const used_options = detected_options[0..detected_len];

    // اطمینان از ارائهٔ همهٔ گزینه‌های اجباری فرمان تشخیص‌داده‌شده
    for (cmd.req) |req_option| {
        var found = false;

        for (used_options) |opt| {
            if (std.mem.eql(u8, req_option, opt.name)){
                found = true; break;
            }
        }

        if (!found) {
            if(debug) std.debug.print("Missing required option: {s}\n", .{req_option});
            return Error.MissingRequiredOption;
        }
    }

    // اجرای تابع مرتبط با فرمان با گزینه‌های تشخیص‌داده‌شده
    if (!cmd.func(used_options)) {
        return Error.CommandExecutionFailed;
    } else {
        // اجرای توابع گزینه‌ها
        for (used_options) |opt| {
            if(opt.func == null) continue;

            const result = opt.func.?(opt.value);

            if (!result) {
                if(debug) std.debug.print("Option function execution failed: {s}\n", .{opt.name});
                return Error.CommandExecutionFailed;
            }
        }
    }

    // اگر اجرا به این نقطه برسد، فرمان با موفقیت اجرا شده است
    if(debug) std.debug.print("Command executed successfully: {s}\n", .{cmd.name});
}

گام ۳: ساخت مدیریت‌کننده‌های فرمان (Creating Command Handlers)

در ادامه، چند مدیریت‌کنندهٔ فرمان (command handler) را در commands.zig پیاده‌سازی می‌کنیم:

zigconst std = @import("std");
const cli = @import("cli.zig");

pub const methods = struct {
    pub const commands = struct {
        // مدیریت‌کنندهٔ فرمان «hello»
        pub fn helloFn(_options: []const cli.option) bool {
            std.debug.print("Hello, ", .{});

            // جست‌وجو برای گزینهٔ «name»
            for (_options) |opt| {
                if (std.mem.eql(u8, opt.name, "name")) {
                    if (opt.value.len > 0) {
                        std.debug.print("{s}", .{opt.value});
                    } else {
                        std.debug.print("World", .{});
                    }
                    break;
                }
            }

            std.debug.print("!\n", .{});
            return true;
        }

        // مدیریت‌کنندهٔ فرمان «help»
        pub fn helpFn(_: []const cli.option) bool {
            std.debug.print(
                "Usage: my-cli <command> [options]\n" ++
                "Commands:\n" ++
                "  hello    Greet someone\n" ++
                "  help     Show this help message\n" ++
                "" ++
                "Options for hello:\n" ++
                "  -n, --name <value>    Name to greet\n"
                , .{}
            );
            return true;
        }
    };

    pub const options = struct {
        // مدیریت‌کنندهٔ گزینهٔ «name»
        pub fn nameFn(_: []const u8) bool {
            // منطق مخصوص این گزینه می‌تواند اینجا بیاید
            return true;
        }
    };
};

گام ۴: برنامهٔ اصلی (Main Application)

حالا، همه‌چیز را در main.zig به هم متصل می‌کنیم:

zigconst std = @import("std");
const cli = @import("cli.zig");
const cmd = @import("commands.zig");

pub fn main() !void {
    // تعریف فرمان‌های موجود
    const commands = [_]cli.command{
        cli.command{
            .name = "hello",
            .func = &cmd.methods.commands.helloFn,
            .opt = &.{"name"},  // «name» برای فرمان hello اختیاری است
        },
        cli.command{
            .name = "help",
            .func = &cmd.methods.commands.helpFn,
        },
    };

    // تعریف گزینه‌های موجود
    const options = [_]cli.option{
        cli.option{
            .name = "name",
            .short = 'n',
            .long = "name",
            .func = &cmd.methods.options.nameFn,
        },
    };

    // اجرای برنامهٔ CLI
    try cli.start(&commands, &options, true);
}

zigconst Build = @import("std").Build;

pub fn build(b: *Build) void {
    const exe_mod           = b.createModule(.{
        .root_source_file   = b.path("src/main.zig"),
        .target             = b.standardTargetOptions(.{}),
        .optimize           = b.standardOptimizeOption(.{}),
    });

    const exe               = b.addExecutable(.{
        .name               = "cli",
        .root_module        = exe_mod,
    });

    b.installArtifact(exe);

    const exe_tests         = b.addTest(.{
        .root_module        = exe_mod,
    });

    const run_exe_tests     = b.addRunArtifact(exe_tests);

    const test_step         = b.step("test", "Run unit tests");
    test_step.dependOn(&run_exe_tests.step);
}

۱. افزودن گزینه‌های اجباری (Adding Required Options)

بیایید فرمان hello خود را طوری تغییر دهیم که یک گزینهٔ greeting (سلام‌گویی) اجباری داشته باشد:

        .name = "name",
        .short = 'n',
        .long = "name",
        .func = &cmd.methods.options.nameFn,

مدیریت‌کنندهٔ فرمان hello را در commands.zig به‌روزرسانی کنید:

zigpub fn helloFn(options: []const cli.option) bool {
    var greeting: []const u8 = undefined;
    var name: []const u8 = "World";

    // استخراج گزینه‌ها
    for (options) |opt| {
        if (std.mem.eql(u8, opt.name, "greeting")) {
            greeting = opt.value;
        } else if (std.mem.eql(u8, opt.name, "name")) {
            if (opt.value.len > 0) {
                name = opt.value;
            }
        }
    }

    std.debug.print("{s}, {s}!\n", .{greeting, name});
    return true;
}

۳. افزودن رنگ و خروجی استایل‌دار (Adding Colors and Styled Output)

برای کاربرپسندتر کردن CLI خود، افزودن رنگ و استایل را در نظر بگیرید:

zig// به cli.zig اضافه کنید
pub const Color = enum {
    Reset,
    Black,
    Red,
    Green,
    Yellow,
    Blue,
    Magenta,
    Cyan,
    White,

    pub fn ansiCode(self: Color) []const u8 {
        return switch (self) {
            .Reset => "\x1b[0m",
            .Black => "\x1b[30m",
            .Red => "\x1b[31m",
            .Green => "\x1b[32m",
            .Yellow => "\x1b[33m",
            .Blue => "\x1b[34m",
            .Magenta => "\x1b[35m",
            .Cyan => "\x1b[36m",
            .White => "\x1b[37m",
        };
    }
};

pub fn printColored(color: Color, comptime fmt: []const u8, args: anytype) void {
    std.debug.print("{s}" ++ fmt ++ "{s}", .{color.ansiCode()} ++ args ++ .{Color.Reset.ansiCode()});
}

۴. نشانگرهای پیشرفت (Progress Indicators)

برای فرمان‌های طولانی‌مدت، یک نشانگر پیشرفت (progress indicator) اضافه کنید:

zig// در cli.zig
pub const Spinner = struct {
    frames: []const []const u8,
    current: usize = 0,
    message: []const u8,
    timer: std.time.Timer,

    pub fn init(message: []const u8) !Spinner {
        return Spinner{
            .frames = &[_][]const u8{ "⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏" },
            .message = message,
            .timer = try std.time.Timer.start(),
            .current = 0,
        };
    }

    pub fn tick(self: *Spinner) void {
        const stdout = std.io.getStdOut().writer();
        _ = stdout.print("\r{s} {s}", .{self.frames[self.current], self.message}) catch {};
        self.current = (self.current + 1) % self.frames.len;
    }

    pub fn stop(self: *Spinner, message: []const u8) void {
        const stdout = std.io.getStdOut().writer();
        _ = stdout.print("\r✓ {s}\n", .{message}) catch {};
    }
};

// کاربرد در commands.zig
pub fn longRunningCommandFn(_: []const cli.option) bool {
    var spinner = cli.Spinner.init("Processing...") catch |err| {
        std.debug.print("Failed to initialize spinner: {}\n", .{err});
        return false;
    };

    // شبیه‌سازی کار
    var i: usize = 0;
    while (i < 50) : (i += 1) {
        spinner.tick();
        std.time.sleep(100 * std.time.ns_per_ms);
    }

    spinner.stop("Done processing!");
    return true;
}

نتیجه‌گیری (Conclusion)

  • تجزیه و اجرای فرمان‌ها
  • گزینه‌های کوتاه و بلند
  • گزینه‌های اجباری و اختیاری
  • مدیریت خطا
  • سازگاری چندسکویی
  • خروجی رنگی
  • نشانگرهای پیشرفت

این پایه، همه‌چیزی را که برای ساخت ابزارهای خط فرمان پیشرفته در Zig نیاز دارید فراهم می‌کند. می‌توانید این چارچوب را گسترش دهید تا زیرفرمان‌ها (subcommands)، اعلان‌های تعاملی (interactive prompts)، تکمیل خودکار (autocompletion) و موارد دیگر را اضافه کنید.

مراحل بعدی (Next Steps)

این بهبودها را برای رساندن چارچوب CLI خود به سطح بعدی در نظر بگیرید:

  1. حالت تعاملی (Interactive Mode): افزودن پشتیبانی از اعلان‌ها و منوهای تعاملی
  2. مدیریت پیکربندی (Configuration Management): پیاده‌سازی پیکربندی پایدار با استفاده از فایل‌ها
  3. تکمیل خودکار (Autocomplete): افزودن پشتیبانی تکمیل خودکار شِل (shell autocompletion)
  4. تست‌نویسی (Testing): ساخت تست‌های واحد برای اجزای CLI شما
  5. تولید مستندات (Documentation Generation): تولید خودکار مستندات راهنما و کاربرد

ترکیب عملکرد سطح پایین و ویژگی‌های ایمنی مدرن در Zig، آن را به انتخابی عالی برای برنامه‌های CLI تبدیل می‌کند. با این پایه، می‌توانید ابزارهایی بسازید که سریع، قابل‌اعتماد و کاربرپسند هستند.
