Load Balancer چه کار می‌کند؟

وقتی ترافیک وب زیاد می‌شود، یک سرور کافی نیست. Load balancer درخواست‌ها را بین چند «بک‌اند» (backend) توزیع می‌کند تا پاسخ‌گویی حفظ شود. ساده‌ترین استراتژی، round-robin است: هر درخواست به سرور بعدی می‌رود.

این آموزش با ابزار استاندارد Go (net/http و httputil) یک load balancer می‌سازد که با سلامت سنجی (health check) هم کار می‌کند.

ساختار و reverse proxy

قلب کار، `httputil.ReverseProxy` است که درخواست را به یک سرور مقصد بازنویسی می‌کند. ما چند backend را نگه می‌داریم و با یک نوبت‌دهنده (index) بین آن‌ها می‌چرخیم:

Go// ساختار backends
type Server struct {
    URL          *url.URL
    Alive        bool
    ReverseProxy *httputil.ReverseProxy
}

// نوبت‌دهندهٔ round-robin با lock
type LoadBalancer struct {
    servers         []*Server
    current         uint64
    mu              sync.RWMutex
}

func (lb *LoadBalancer) NextServer() *Server {
    lb.mu.Lock()
    defer lb.mu.Unlock()
    // سرور زندهٔ بعدی را پیدا کن
    for i := 0; i < len(lb.servers); i++ {
        idx := int(lb.current) % len(lb.servers)
        lb.current++
        if lb.servers[idx].Alive {
            return lb.servers[idx]
        }
    }
    return nil
}

اگر هیچ سروری زنده نباشد، باید خطای ۵۰۳ برگردانیم؛ در غیر این صورت همان server برگردانده‌شده، پاسخ می‌دهد.

هندلر درخواست

وقتی درخواست به load balancer برسد، باید به یکی از سرورها برگردد. ساده‌ترین هندلر:

Go// توزیع درخواست روی سرورها
func (lb *LoadBalancer) handleRequest(w http.ResponseWriter, r *http.Request) {
    server := lb.NextServer()
    if server == nil {
        http.Error(w, "هیچ سرور زنده‌ای نیست", http.StatusServiceUnavailable)
        return
    }
    server.ReverseProxy.ServeHTTP(w, r)
}

func main() {
    lb := &LoadBalancer{}
    for _, s := range []string{"http://localhost:8081", "http://localhost:8082"} {
        u, _ := url.Parse(s)
        proxy := httputil.NewSingleHostReverseProxy(u)
        lb.servers = append(lb.servers, &Server{URL: u, Alive: true, ReverseProxy: proxy})
    }
    http.ListenAndServe(":8080", http.HandlerFunc(lb.handleRequest))
}

با این ۳۰ خط، توزیع round-robin روی دو سرور کار می‌کند.

سلامت‌سنجی

در نسخهٔ کامل‌تر، یک تابع در پس‌زمینه هر چند ثانیه یک‌بار با درخواست GET به همهٔ سرورها، وضعیت آن‌ها را چک می‌کند و `Alive` را به‌روزرسانی می‌کند:

Go// بررسی دوره‌ای سلامت سرورها
func (lb *LoadBalancer) healthCheck() {
    for _, s := range lb.servers {
        resp, err := http.Get(s.URL.String() + "/health")
        lb.mu.Lock()
        s.Alive = err == nil && resp != nil && resp.StatusCode == http.StatusOK
        lb.mu.Unlock()
        if resp != nil { resp.Body.Close() }
    }
}

func startHealthCheck(lb *LoadBalancer, every time.Duration) {
    ticker := time.NewTicker(every)
    for range ticker.C { lb.healthCheck() }
}

با این‌کار، اگر یک سرور از کار بیفتد، خودکار از چرخهٔ توزیع حذف می‌شود.
