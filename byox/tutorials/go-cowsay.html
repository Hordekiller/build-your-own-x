مقدمه: cowsay چیست؟

Cowsay یکی از آن اپلیکیشن‌هایی است که بدون آن نمی‌توان زندگی کرد.

این ابزار اساساً تصاویر ASCII از یک گاو را همراه با هر پیامی که به آن بدهید تولید می‌کند؛ در اسکرین‌شات بالا از fortune برای تولید پیام استفاده شده است. اما این ابزار به دنیای گاو محدود نیست؛ می‌تواند پنگوئن، گوزن شاخ‌دار (moose) و بسیاری از حیوانات دیگر را هم چاپ کند.

به نظر می‌رسد این ابزار مفیدی برای انتقال به Go باشد!

ضمناً، من مجوز سادهٔ انگلیسی‌ای که به آن پیوست شده را دوست دارم:

تعریف مسئله: خواندن ورودی از پایپ

بیایید با تعریف مسئله شروع کنیم. می‌خواهیم ورودی را از طریق یک پایپ (pipe) بپذیریم و گاو ما آن را بگوید.

تکرار اول، ورودی کاربر را از پایپ می‌خواند و آن را دوباره چاپ می‌کند. چندان پیچیده نیست.

gopackage main

import (
	"bufio"
	"fmt"
	"io"
	"os"
)

func main() {
	info, _ := os.Stdin.Stat()

	if info.Mode()&os.ModeCharDevice != 0 {
		fmt.Println("The command is intended to work with pipes.")
		fmt.Println("Usage: fortune | gocowsay")
		return
	}

	reader := bufio.NewReader(os.Stdin)
	var output []rune

	for {
		input, _, err := reader.ReadRune()
		if err != nil && err == io.EOF {
			break
		}
		output = append(output, input)
	}

	for j := 0; j < len(output); j++ {
		fmt.Printf("%c", output[j])
	}
}

گاو را نداریم و همچنین باید پیام را با قالب‌بندی مناسب داخل یک بالن (balloon) بپیچیم.

ساخت بالن گفتار

اینجاست اولین تکرار کامل برنامهٔ ما:

gopackage main

import (
	"bufio"
	"fmt"
	"io"
	"os"
	"strings"
	"unicode/utf8"
)

// buildBalloon یک برش از رشته‌ها با حداکثر عرضِ maxwidth می‌گیرد،
// به خط اول و آخر و همچنین به ابتدا/انتهای هر خط حاشیه اضافه می‌کند
// و یک رشته شامل محتوای بالن را برمی‌گرداند
func buildBalloon(lines []string, maxwidth int) string {
	var borders []string
	count := len(lines)
	var ret []string

	borders = []string{"/", "\\", "\\", "/", "|", "<", ">"}

	top := " " + strings.Repeat("_", maxwidth+2)
	bottom := " " + strings.Repeat("-", maxwidth+2)

	ret = append(ret, top)
	if count == 1 {
		s := fmt.Sprintf("%s %s %s", borders[5], lines[0], borders[6])
		ret = append(ret, s)
	} else {
		s := fmt.Sprintf(`%s %s %s`, borders[0], lines[0], borders[1])
		ret = append(ret, s)
		i := 1
		for ; i < count-1; i++ {
			s = fmt.Sprintf(`%s %s %s`, borders[4], lines[i], borders[4])
			ret = append(ret, s)
		}
		s = fmt.Sprintf(`%s %s %s`, borders[2], lines[i], borders[3])
		ret = append(ret, s)
	}

	ret = append(ret, bottom)
	return strings.Join(ret, "\n")
}

// tabsToSpaces همهٔ تب‌های موجود در رشته‌های برشِ lines را به ۴ فاصله
// تبدیل می‌کند تا در شمارشِ runeها ناهم‌ترازی پیش نیاید
func tabsToSpaces(lines []string) []string {
	var ret []string
	for _, l := range lines {
		l = strings.Replace(l, "\t", "    ", -1)
		ret = append(ret, l)
	}
	return ret
}

// calculateMaxWidth با گرفتن یک برش از رشته‌ها، طول بلندترین رشته را برمی‌گرداند
func calculateMaxWidth(lines []string) int {
	w := 0
	for _, l := range lines {
		len := utf8.RuneCountInString(l)
		if len > w {
			w = len
		}
	}

	return w
}

// normalizeStringsLength یک برش از رشته‌ها می‌گیرد و به هر یک به‌اندازه‌ای
// فاصله اضافه می‌کند که همهٔ آن‌ها تعداد rune یکسانی داشته باشند
func normalizeStringsLength(lines []string, maxwidth int) []string {
	var ret []string
	for _, l := range lines {
		s := l + strings.Repeat(" ", maxwidth-utf8.RuneCountInString(l))
		ret = append(ret, s)
	}
	return ret
}

func main() {
	info, _ := os.Stdin.Stat()

	if info.Mode()&os.ModeCharDevice != 0 {
		fmt.Println("The command is intended to work with pipes.")
		fmt.Println("Usage: fortune | gocowsay")
		return
	}

	var lines []string

	reader := bufio.NewReader(os.Stdin)

	for {
		line, _, err := reader.ReadLine()
		if err != nil && err == io.EOF {
			break
		}
		lines = append(lines, string(line))
	}

	var cow = `         \  ^__^
          \ (oo)\_______
	    (__)\       )\/\
	        ||----w |
	        ||     ||
		`

	lines = tabsToSpaces(lines)
	maxwidth := calculateMaxWidth(lines)
	messages := normalizeStringsLength(lines, maxwidth)
	balloon := buildBalloon(messages, maxwidth)
	fmt.Println(balloon)
	fmt.Println(cow)
	fmt.Println()
}

قابل پیکربندی کردن شکل: پرچم -f

حالا بیایید شکل را قابل پیکربندی کنیم و یک استگوسوروس اضافه کنیم:

برنامهٔ اصلی از پرچم -f برای پذیرش یک شکل سفارشی استفاده می‌کند. پس بیایید همین کار را با پردازش یک پرچم خط فرمان انجام دهیم.

برنامهٔ قبلی را به‌طور خلاصه تغییر می‌دهم تا تابع printFigure() معرفی شود:

go// printFigure با گرفتن نام شکل، آن را چاپ می‌کند.
// در حال حاضر `cow` و `stegosaurus` را می‌پذیرد.
func printFigure(name string) {

	var cow = `         \  ^__^
          \ (oo)\_______
	    (__)\       )\/\
	        ||----w |
	        ||     ||
		`

	var stegosaurus = `         \                      .       .
          \                    / ` + "`" + `.   .' "
           \           .---.  <    > <    >  .---.
            \          |    \  \ - ~ ~ - /  /    |
          _____           ..-~             ~-..-~
         |     |   \~~~\\.'                    ` + "`" + `./~~~/
        ---------   \__/                         \__/
       .'  O    \     /               /       \  "
      (_____,    ` + "`" + `._.'               |         }  \/~~~/
       ` + "`" + `----.          /       }     |        /    \__/
             ` + "`" + `-.      |       /      |       /      ` + "`" + `. ,~~|
                 ~-.__|      /_ - ~ ^|      /- _      ` + "`" + `..-'
                      |     /        |     /     ~-.     ` + "`" + `-. _  _  _
                      |_____|        |_____|         ~ - . _ _ _ _ _>

	`

	switch name {
	case "cow":
		fmt.Println(cow)
	case "stegosaurus":
		fmt.Println(stegosaurus)
	default:
		fmt.Println("Unknown figure")
	}
}

و main() را تغییر می‌دهم تا یک پرچم بپذیرد و آن را به printFigure() پاس دهد:

gofunc main() {
	//...

	var figure string
	flag.StringVar(&figure, "f", "cow", "the figure name. Valid values are `cow` and `stegosaurus`")
	flag.Parse()

	//...
	printFigure(figure)
	fmt.Println()
}
