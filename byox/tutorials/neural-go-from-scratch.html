ساخت یک شبکه عصبی از صفر در Go

۱. هدف: طبقه‌بندی گل‌های زنبق (iris)

دیتاست معروف iris سه نوع گل زنبق را بر اساس چهار ویژگی (طول و عرض کاسبرگ و گلبرگ) طبقه‌بندی می‌کند. نویسنده یک شبکهٔ عصبی از صفر می‌نویسد — بدون هیچ کتابخانهٔ ML — تا هم این دادهٔ واقعی را حل کند و هم مکانیک درونی شبکه را نشان دهد: ساختار، مقداردهی، انتشار پیشرو، انتشار بازگشتی و آموزش.

۲. مدل و انواع داده

شبکه به‌صورت struct تعریف می‌شود: وزن‌ها (ماتریس‌ها)، تعداد نورون‌های هر لایه و تابع فعال‌سازی. با نمایش ماتریسی، انتشار پیشرو فقط چند ضرب ماتریسی است:

go// تعریف شبکه: وزن‌ها بین هر جفت لایه + فعال‌سازی
type NeuralNet struct {
    config  NeuralNetConfig
    wHidden *mat.Dense   // وزن‌های لایهٔ پنهان
    wOut    *mat.Dense   // وزن‌های لایهٔ خروجی
    bHidden *mat.Dense   // بایاس لایهٔ پنهان
    bOut    *mat.Dense   // بایاس لایهٔ خروجی
}

// ساختار پیکربندی: ابعاد لایه‌ها، نرخ یادگیری و تعداد عصرها
type NeuralNetConfig struct {
    InputNeurons  int
    OutputNeurons int
    HiddenNeurons int
    NumEpochs     int
    LearningRate  float64
}

۳. مقداردهی تصادفی وزن‌ها

شبکه با وزن‌های تصادفی کوچک شروع می‌کند. تابع NewNetwork با گرفتن پیکربندی، ماتریس‌های وزن را می‌سازد و با توزیع نرمال مقداردهی می‌کند؛ بایاس‌ها نیز به‌صورت تصادفی (یا صفر) مقداردهی می‌شوند. تولیدکنندهٔ اعداد تصادفی seed می‌خورد تا آزمایش‌ها بازتولیدپذیر باشند:

go// ساخت و مقداردهی تصادفی وزن‌ها و بایاس‌ها
func NewNetwork(config NeuralNetConfig) *NeuralNet {
    r := rand.New(rand.NewSource(time.Now().UnixNano()))
    wHidden := mat.NewDense(config.InputNeurons, config.HiddenNeurons, nil)
    bHidden := mat.NewDense(1, config.HiddenNeurons, nil)
    wOut := mat.NewDense(config.HiddenNeurons, config.OutputNeurons, nil)
    bOut := mat.NewDense(1, config.OutputNeurons, nil)

    // مقداردهی تصادفی از توزیع نرمال با میانگین صفر
    for i := 0; i < config.InputNeurons; i++ {
        for j := 0; j < config.HiddenNeurons; j++ {
            wHidden.Set(i, j, r.NormFloat64())
        }
    }
    // ... همین کار برای بقیهٔ ماتریس‌ها
}

۴. انتشار پیشرو (feed forward)

برای هر نمونه: خروجی لایهٔ پنهان = سیگموئید(ورودی × وزن‌پنهان + بایاس‌پنهان)، سپس خروجی نهایی = سیگموئید(پنهان × وزن‌خروجی + بایاس‌خروجی). تابع فعال‌سازی سیگموئید و مشتق آن به‌صورت توابع جدا تعریف می‌شوند تا در backprop هم استفاده شوند:

go// اجرای شبکه روی یک بردار ویژگی‌ها
func (nn *NeuralNet) predict(x *mat.VecDense) mat.Matrix {
    // محاسبهٔ خروجی لایهٔ پنهان و اعمال سیگموئید
    hiddenInputs := mat.NewDense(1, nn.config.HiddenNeurons, nil)
    hiddenInputs.MulVec(nn.wHidden, x)
    hiddenInputs.AddScaled(hiddenInputs, 1, nn.bHidden.RowView(0))
    hiddenOutputs := applySigmoid(hiddenInputs)

    // محاسبهٔ خروجی نهایی و اعمال سیگموئید
    finalInputs := mat.NewDense(1, nn.config.OutputNeurons, nil)
    finalInputs.MulVec(nn.wOut, hiddenOutputs)
    finalInputs.AddScaled(finalInputs, 1, nn.bOut.RowView(0))
    return applySigmoid(finalInputs)
}
