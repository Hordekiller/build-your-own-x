AngularJS ساده‌شدهٔ خودت را در ۲۰۰ خط جاوااسکریپت بساز

⚙ زبان: JavaScript ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمهٔ فارسی و مرور مقالهٔ «Build Your Own Simplified AngularJS in 200 Lines of JavaScript» از مینکو گچف (Minko Gechev) است. او با بازسازی سه سرویس اصلی AngularJS — کامپایلر، تزریق وابستگی و digest cycle — نشان می‌دهد این فریم‌ورک «جادویی» نیست.
↗ مشاهده منبع اصلی

۱. سه ستون AngularJS

مقاله سه بخش اصلی را شناسایی و یکی‌یکی بازسازی می‌کند:

  • DOMCompiler ($compile): دستورها (directiveها) را از HTML پیدا می‌کند و آن‌ها را اجرا می‌کند؛
  • Provider ($provide + $injector): سرویس‌ها را تعریف و تزریق می‌کند؛
  • Scope: مدلی با سلسله‌مراتب که binding دوسویه را ممکن می‌کند.

این‌ها دقیقاً همان چیزی‌اند که AngularJS را AngularJS می‌کند؛ بقیه (routing، filters، form controls) فقط دور این سه محور جمع شده‌اند.

۲. DOMCompiler: پیدا کردن و اجرای دستورها

وقتی AngularJS بارگذاری می‌شود، شروع به «کامپایل» DOM می‌کند: همهٔ عناصر را پیمایش می‌کند، دستورها را از روی نام صفت پیدا می‌کند و آن‌ها را با scope فعلی اجرا می‌کند. پیاده‌سازی سادهٔ ما:

jsconst DOMCompiler = {
  bootstrap() {
    // پیمایش بازگشتی تمام نودهای DOM
    this.compile(document.children[0], Provider.get('$rootScope'));
  },

  compile(el, scope) {
    const dirs = this._getElDirectives(el);
    // اجرای هر دستور با scope
    dirs.forEach((d) => {
      d.scope = scope;
      d.fn(scope, el, d.attrs, d);
    });
    // صفت ng-repeat و امثال آن: نادیده‌گرفتن عنصر تکرارشونده
    if (dirs.some((d) => d.name === 'ng-repeat')) return;
    // پیمایش بچه‌ها
    Array.prototype.slice.call(el.children).forEach((child) => {
      this.compile(child, scope);
    });
  },

  _getElDirectives(el) {
    const attrs = el.attributes;
    const result = [];
    for (let i = 0; i < attrs.length; i++) {
      const attr = attrs[i];
      if (Directive.get(attr.nodeName)) {
        result.push({
          name: attr.nodeName,
          value: attr.nodeValue,
          fn: Directive.get(attr.nodeName),
        });
      }
    }
    return result;
  },
};

دستورها در یک رجیستری (Directive) ذخیره می‌شوند: تابع‌هایی که (scope, element, attrs) می‌گیرند و معمولاً فقط عملیات «یک‌بار» انجام می‌دهند — مثل ثبت یک watch یا ساخت DOM جدید.

۳. Provider و تزریق وابستگی

تزریق وابستگی در AngularJS ساده است: سرویس‌ها با نام ثبت می‌شوند و وقتی تابعی اعلام می‌کند «فلان سرویس را می‌خواهم»، injector آن را پیدا و تحویل می‌دهد. نام آرگومان‌های تابع همان «توضیح نیاز» است:

jsconst Provider = {
  _providers: {},
  _cache: {},

  provide(name, fn) {
    this._providers[name] = fn;
  },

  // تزریق: نام آرگومان‌های تابع را از روی تعریف تابع می‌خوانیم
  get(name) {
    if (this._cache[name]) return this._cache[name];
    const provider = this._providers[name];
    if (!provider) throw new Error('سرویس ناشناخته: ' + name);

    // گرفتن نام پارامترها با تبدیل تابع به رشته و تحلیل متن آن
    const args = this._getParams(provider);
    // ساخت خودکار سرویس‌های وابسته (بازگشتی)
    const deps = args.map((arg) => this.get(arg));
    const instance = provider.apply(null, deps);
    this._cache[name] = instance; // فقط یک نمونه از هر سرویس
    return instance;
  },

  _getParams(fn) {
    // تحلیل رشتهٔ تعریف تابع برای استخراج نام پارامترها
    const fnStr = fn.toString();
    const params = fnStr.match(/^function\s*[^\(]*\(\s*([^\)]*)\)/m);
    if (!params || !params[1]) return [];
    return params[1].split(',').map((p) => p.trim().replace(/[^_$a-zA-Z0-9]/g, ''));
  },
};

همان ترفندی که خود AngularJS (و Angular 2+ با DI کامپایل‌شده) استفاده می‌کرد: «نام پارامترها» در توسعهٔ تولیدی مشکل‌ساز بود (minification) و به همین دلیل Angular بعداً سراغ تزریق صریح (کامنت یا آرایهٔ نام) رفت.

۴. Scope و binding دوسویه

هر scope درخت است: با $new() scope فرزند می‌سازیم (که ویژگی‌های پدر را می‌بیند اما تغییراتش را به پدر برنمی‌گرداند) و با $digest حلقهٔ بررسی تغییرات اجرا می‌شود. دستور ng-bind مقدار scope را به متن عنصر وصل می‌کند:

jsclass Scope {
  constructor() {
    this.$$watchers = [];
  }

  $watch(fn, listener) {
    const watcher = { fn, last: undefined, listener };
    this.$$watchers.push(watcher);
  }

  // حلقهٔ digest: تا وقتی تغییری هست بررسی را تکرار کن (حداکثر ۱۰ بار)
  $digest() {
    let dirty = true;
    let count = 0;
    while (dirty && count < 10) {
      dirty = this._digestOnce();
      count++;
    }
  }

  _digestOnce() {
    let dirty = false;
    this.$$watchers.forEach((watcher) => {
      const newValue = watcher.fn(this);
      if (newValue !== watcher.last) {
        watcher.last = newValue;
        watcher.listener(newValue, watcher.last, this);
        dirty = true;
      }
    });
    return dirty;
  }

  $new() {
    // سلسله‌مراتب scope: برای فرزندها
    const child = new Scope();
    child.$parent = this;
    return child;
  }
}

// دستور ng-bind: اتصال عبارت scope به متن عنصر
Directive.set('ng-bind', (scope, el, attrs) => {
  scope.$watch(() => {
    // ارزیابی عبارت (مثل "name") در scope فعلی
    return eval('scope.' + attrs['ng-bind']);
  }, (value) => {
    el.innerHTML = value;
  });
});

۵. گرد هم آوردن همه: یک برنامهٔ کامل

js// ثبت سرویس‌ها
Provider.provide('$scope', () => new Scope());
Provider.provide('$rootScope', () => new Scope());

// دستور ng-model برای ورودی‌ها (binding دوسویه)
Directive.set('ng-model', (scope, el) => {
  el.addEventListener('input', () => {
    scope.model = el.value;
    scope.$digest(); // با هر تایپ، digest اجرا می‌شود
  });
});

// بوت استرپ: از ریشهٔ سند شروع کن
DOMCompiler.bootstrap();

حالا هر HTML با ng-bind="model" و ng-model روی یک ورودی، رفتار دوسویهٔ AngularJS دارد — همه‌اش در حدود ۲۰۰ خط!

نکتهٔ تکمیلی: چرا AngularJS اینقدر کند بود؟ چون هر تغییر، یک digest کامل کل درخت scope را اجرا می‌کرد (dirty checking). فریم‌ورک‌های جدیدتر مثل Angular و Vue به‌جای آن با «زون‌ها» یا «مشاهدهٔ getter/setter» تغییرات را مستقیم ردیابی می‌کنند.

۶. جمع‌بندی

  • دستورها = توابعی که با scope اجرا می‌شوند؛ ng-repeat، ng-bind و… همه در یک رجیستری ساده‌اند.
  • تزریق وابستگی = خواندن نام آرگومان‌ها از متن تابع.
  • Scope = درختی از watcherها با حلقهٔ digest؛ منبع قدرت و ضعف AngularJS.
