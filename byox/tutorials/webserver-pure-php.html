نوشتن یک وب سرور با PHP خالص

⚙ زبان: PHP 📁 دسته: وب سرور ✅ ترجمه شده 🔗 منبع اصلی
منبع اصلی: این صفحه ترجمه فارسی آموزش «Writing a webserver in pure PHP» از «ماریو دلونا (Mario Deluna)» — ClanCats Station است. نسخه کامل و به‌روز را از سایت اصلی بخوانید.
↗ مشاهده منبع اصلی

مقدمه — خب، این کار تقریباً بی‌فایده است، اما ممکن است!

این آموزش به شما کمک می‌کند بهتر بفهمید یک وب سرور ساده چگونه می‌تواند کار کند و اینکه نوشتن یکی با PHP هیچ مشکلی ندارد. اما استفاده از آن در تولید مثل این است که با چنگال سوپ بخورید؛ پس... فقط، نکنید. بگذارید مختصر توضیح دهم چرا ایدهٔ خوبی نیست:

PHP یک زبان اسکریپتی است که واقعاً برای چنین کارهایی طراحی نشده.

یک وب سرور یک فرایند طولانی‌مدت است و PHP ساختهٔ چنین کاری نیست. همچنین PHP به‌صورت بومی از چندنخی پشتیبانی نمی‌کند (pthreads)، که توسعهٔ یک وب سرور خوب و پرسرعت را سخت یا غیرممکن می‌کند. تخصیص حافظهٔ PHP هم می‌تواند پرهزینه باشد و حتی با وجود سرعت بالای مفسر، سربار یک مفسر را دارد.

اما چیزها ممکن است در آینده تغییر کنند. اگر با برنامه‌نویس‌ها معاشرت کرده باشید، حداقل یکی را دیده‌اید که دربارهٔ بد بودن PHP شوخی می‌کند و دلایل واضحی هم دارد که چرا PHP چنین شوخی صنعتی شده. اما زبان و اکوسیستمش بسیار تکامل یافته است. هنوز هم آدم‌هایی هستند که پشت شوخی‌ها را نمی‌بینند؛ بسیاری از آن‌ها یا حداقل ۱۰ سال است PHP را لمس نکرده‌اند یا کورکورانه نظر دیگران را کپی می‌کنند. من شرط می‌بندم چیزهایی مثل برنامه‌های PHP طولانی‌مدت با وب سرورهای تعبیه‌شده بیشتر و بیشتر دیده خواهند شد.

مبانی — وب سرور چگونه کار می‌کند؟

  1. سرور منتظر اتصالات ورودی می‌ماند.
  2. یک کلاینت به سرور متصل می‌شود.
  3. سرور اتصال را می‌پذیرد و ورودی را پردازش می‌کند.
  4. سرور به کلاینت پاسخ می‌دهد.

کل سورس در GitHub در دسترس است: ClanCatsStation/PHPWebserver

ساختار

ما این را به‌صورت انتزاعی از Request و Response می‌سازیم. راه‌های زیادی برای طراحی یک برنامه هست؛ نویسنده به‌عنوان گام اول، ترجیح می‌دهد بخشی از برنامه را بنویسد که API را مصرف می‌کند و بعد خود API را بنویسد. پس انتظار داریم بتوانیم از آن این‌طور استفاده کنیم:

php// ساخت یک نمونه از سرور
$server = new Server( '127.0.0.1', 80 );

// شروع گوش‌دادن
$server->listen( function( Request $request ) 
{
	return new Response( 'Hello Dude' );
});

راه‌اندازی

بعد فایل اسکریپت (server) را می‌سازیم که راه‌اندازی سرور را بر عهده دارد. پسوند .php را اضافه نمی‌کنیم تا bash بداند چه کند؛ این هدر را اضافه کنید:

php#!/usr/bin/env php

همیشه سرور را به localhost وصل می‌کنیم اما می‌خواهیم پورت به‌عنوان آرگومان خط فرمان قابل تعریف باشد:

php// به آرگومان اول نیازی نداریم
array_shift( $argv );

// آرگومان بعدی باید پورت باشد؛ اگر نبود از 80 استفاده کن
if ( empty( $argv ) )
{
	$port = 80;
} else {
	$port = array_shift( $argv );
}

این امکان را می‌دهد بعداً سرور را این‌طور راه بیندازیم:

Bash$ sudo php server 8008

با ترکیب با آنچه قبلاً تعریف کردیم، چنین فایلی به دست می‌آید:

php#!/usr/bin/env php
listen( function( Request $request ) 
{
	return new Response( 'Hello Dude' );
});

ساخت سوکت

برای bind کردن سوکت به یکی نیاز داریم؛ پس تابع createSocket را می‌سازیم:

phpprotected function createSocket()
{
	$this->socket = socket_create( AF_INET, SOCK_STREAM, 0 );
}
  • آرگومان اول خانوادهٔ پروتکل/دامنه را مشخص می‌کند؛ AF_INET برای پروتکل‌های IPv4 و TCP و UDP است.
  • آرگومان دوم نوع ارتباط سوکت را تعریف می‌کند؛ SOCK_STREAM یک استریم بایت ساده مبتنی بر اتصال تمام‌دوطرفه (full-duplex) است.
  • آرگومان سوم پروتکل را تنظیم می‌کند.

Bind کردن سوکت

تابع socket_bind وقتی خطایی رخ دهد false برمی‌گرداند. چون این نباید هرگز اتفاق بیفتد، با پیام خطای سوکت یک استثنا پرتاب می‌کنیم:

phpprotected function bind()
{
	if ( !socket_bind( $this->socket, $this->host, $this->port ) )
	{
		throw new Exception( 'Could not bind: '.$this->host.':'.$this->port.' - '.socket_strerror( socket_last_error() ) );
	}
}

ساخت و bind سوکت در سازنده

برای سادگی این کار را داخل سازنده انجام می‌دهیم:

phppublic function __construct( $host, $port )
{
	$this->host = $host;
	$this->port = (int) $port;
	
	// ساخت سوکت
	$this->createSocket();
	
	// bind کردن سوکت
	$this->bind();
}

گوش‌دادن به اتصالات

تابع listen حلقهٔ بی‌نهایت اصلی سرور است:

phppublic function listen( $callback )
{
	// بررسی اینکه callback معتبر است؛ اگر نه استثنا پرتاب کن
	if ( !is_callable( $callback ) )
	{
		throw new Exception('The given argument should be callable.');
	}
	
	// این همان چیزی است که فرایند را طولانی، بی‌نهایت و بی‌پایان می‌کند..
	while ( 1 ) 
	{
		// گوش‌دادن به اتصالات
		socket_listen( $this->socket );
		
		// تلاش برای گرفتن منبع سوکت کلاینت
		// اگر false بود خطا داریم؛ اتصال را ببند و رد شو
		if ( !$client = socket_accept( $this->socket ) ) 
		{
			socket_close( $client ); continue;
		}
		
		// ساخت نمونه جدید Request با هدر کلاینت
		// در دنیای واقعی نمی‌توانید حداکثر اندازه را روی 1024 ثابت کنید..
		$request = Request::withHeaderString( socket_read( $client, 1024 ) );
		
		// اجرای callback
		$response = call_user_func( $callback, $request );
		
		// بررسی اینکه واقعاً یک شیء Response دریافت کرده‌ایم
		// اگر نه، یک شیء Response خطای 404 برگردان
		if ( !$response || !$response instanceof Response )
		{
			$response = Response::error( 404 );
		}
		
		// از پاسخ‌مان یک رشته بساز
		$response = (string) $response;
		
		// پاسخ را به سوکت کلاینت بنویس
		socket_write( $client, $response, strlen( $response ) );
		
		// اتصال را ببند تا بتوانیم اتصالات جدید بپذیریم
		socket_close( $client );
	}
}

پارس‌کردن هدر

یک هدر HTTP این‌طور به نظر می‌رسد:

HTTPGET / HTTP/1.1
Host: 127.0.0.1:8008
Connection: keep-alive
Accept: text/html
User-Agent: Chrome/41.0.2272.104
Accept-Encoding: gzip, deflate, sdch
Accept-Language: en-US,en;q=0.8,de;q=0.6

خط اول متد درخواست، uri و پروتکل را نشان می‌دهد و بعد از آن پارامترهای هدر به صورت کلید/مقدار می‌آیند:

phppublic static function withHeaderString( $header )
{
	// رشته را به خط تبدیل کن
	$lines = explode( "\n", $header );
	
	// متد و uri را استخراج کن
	list( $method, $uri ) = explode( ' ', array_shift( $lines ) );
	
	$headers = [];
	
	foreach( $lines as $line )
	{
		// تمیزکردن خط
		$line = trim( $line );
		
		if ( strpos( $line, ': ' ) !== false )
		{
			list( $key, $value ) = explode( ': ', $line );
			$headers[$key] = $value;
		}
	}	
	
	// ساخت شیء درخواست جدید
	return new static( $method, $uri, $headers );
}

سازنده $method، $uri و $headers را می‌گیرد. برای این مثال می‌خواهیم پارامترهای کوئری را هم جدا و پارس کنیم:

phppublic function __construct( $method, $uri, $headers = [] ) 
{
	$this->headers = $headers;
	$this->method = strtoupper( $method );
	
	// جداکردن uri و رشتهٔ پارامترها
	@list( $this->uri, $params ) = explode( '?', $uri );

	// پارس‌کردن پارامترها
	parse_str( $params, $this->parameters );
}

برای دسترسی به داده‌های درخواست، چند getter می‌سازیم؛ getter های header و param اجازهٔ دادن مقدار پیش‌فرض را دارند:

phppublic function method()
{
	return $this->method;
}

public function uri()
{
	return $this->uri;
}

public function header( $key, $default = null )
{
	if ( !isset( $this->headers[$key] ) )
	{
		return $default;
	}
		
	return $this->headers[$key];
}

public function param( $key, $default = null )
{
	if ( !isset( $this->parameters[$key] ) )
	{
		return $default;
	}
		
	return $this->parameters[$key];
}

سازندهٔ Response

پارامترهای عمومی که شیء Response باید پیاده‌سازی کند عبارت‌اند از وضعیت status، بدنهٔ body و سایر هدرها. بدنه باید آرگومان الزامی سازنده باشد و وضعیت و بقیهٔ هدرها اختیاری. سازنده باید مقادیر پیش‌فرضی مثل Date و Server را هم تنظیم کند:

phpprotected $status = 200;
protected $body = '';
protected $headers = [];

public function __construct( $body, $status = null )
{
	if ( !is_null( $status ) )
	{
		$this->status = $status;
	}
	
	$this->body = $body;
	
	// تنظیم هدرهای اولیه
	$this->header( 'Date', gmdate( 'D, d M Y H:i:s T' ) );
	$this->header( 'Content-Type', 'text/html; charset=utf-8' );
	$this->header( 'Server', 'PHPServer/1.0.0 (Whateva)' );
}

تنظیم هدرها

برای افزودن پارامترهای هدر جدید یک setter می‌سازیم:

phppublic function header( $key, $value )
{
	$this->headers[ucfirst($key)] = $value;
}

ساختن رشتهٔ هدر

نمی‌توانیم شیء Response را مستقیم به نویسندهٔ سوکت بدهیم؛ باید از داده‌هایمان یک رشته بسازیم:

phppublic function buildHeaderString()
{
	$lines = [];
		
	// وضعیت پاسخ
	$lines[] = "HTTP/1.1 ".$this->status." ".static::$statusCodes[$this->status];
		
	// افزودن هدرها
	foreach( $this->headers as $key => $value )
	{
		$lines[] = $key.": ".$value;
	}
		
	return implode( " \r\n", $lines )."\r\n\r\n";
}

جادو!

با متد جادویی __toString کل رشتهٔ نوشته‌شده به اتصال باز را برمی‌گردانیم:

phppublic function __toString()
{
	return $this->buildHeaderString().$this->body;
}
