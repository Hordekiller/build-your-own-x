۱. ایدهٔ اصلی: فکر کردن مثل برنامه‌نویس

هدف مقاله فقط «ساختن بازی Snake» نیست؛ بلکه روش فکر کردن یک برنامه‌نویس است: به‌جای نوشتن کد یکجا، مسئله را به قطعات کوچک و قابل‌حل تقسیم کنید: (۱) مار را چگونه نمایش دهیم؟ (۲) چگونه حرکت کند؟ (۳) چگونه غذا بخورد و رشد کند؟ (۴) چگونه ببازیم؟ (۵) چگونه امتیاز بشماریم؟ در انتها همهٔ قطعات در یک فایل snake.html جمع می‌شوند.

۲. نمایش مار: آرایه‌ای از مختصات

مار یک «آرایه از مختصات» روی شبکهٔ ۲۰×۲۰ است؛ سر، اولین عضو و غذا یک نقطهٔ تصادفی است:

js// مار: آرایه‌ای از [x, y]
let snake = [
  [9, 9],  // سر
  [8, 9],  // بدن
  [7, 9],
];

let direction = [1, 0]; // حرکت به راست

// شبکه: canvas 400×400 با سلول‌های ۲۰ پیکسلی
const CELL = 20;
const GRID = 20;

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  snake.forEach(([x, y]) => {
    ctx.fillStyle = 'green';
    ctx.fillRect(x * CELL, y * CELL, CELL - 1, CELL - 1);
  });
  // غذا
  ctx.fillStyle = 'red';
  ctx.fillRect(food[0] * CELL, food[1] * CELL, CELL - 1, CELL - 1);
}

۳. حرکت: unshift و pop

حرکت مار از این قانون ظریف استفاده می‌کند: یک عضو به سر اضافه کن (unshift) و اگر غذا نخورد، عضو آخر را حذف کن (pop) — همین باعث حرکت و رشد می‌شود:

jsfunction gameStep() {
  // سر جدید
  const head = snake[0];
  const newHead = [head[0] + direction[0], head[1] + direction[1]];

  // برخورد با دیوار یا خودش → پایان
  const hitWall = newHead[0] < 0 || newHead[0] >= GRID ||
                  newHead[1] < 0 || newHead[1] >= GRID;
  const hitSelf = snake.some(([x, y]) => x === newHead[0] && y === newHead[1]);
  if (hitWall || hitSelf) { gameOver(); return; }

  snake.unshift(newHead);

  // غذا خورده شد؟
  if (newHead[0] === food[0] && newHead[1] === food[1]) {
    score++;
    food = randomFood();      // غذا جدید
  } else {
    snake.pop();              // بدون رشد: دم حذف می‌شود
  }
  draw();
}

// حلقهٔ بازی
setInterval(gameStep, 100);

۴. ورودی کیبورد

jsdocument.addEventListener('keydown', (e) => {
  // جلوگیری از برعکس رفتن مستقیم
  const keyMap = {
    ArrowUp:    [0, -1],
    ArrowDown:  [0, 1],
    ArrowLeft:  [-1, 0],
    ArrowRight: [1, 0],
  };
  const d = keyMap[e.key];
  if (d) {
    if (d[0] !== -direction[0] || d[1] !== -direction[1]) {
      direction = d;
    }
    e.preventDefault();
  }
});
